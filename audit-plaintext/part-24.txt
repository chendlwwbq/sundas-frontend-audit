export async function uploadImage(input: string | Blob, options?: ImageReadOptions): Promise<UploadedImage> {
    const owner = options?.expectedOwner === undefined ? currentMediaOwner() : options.expectedOwner;
    assertMediaOwner(owner, options?.signal);
    options = { ...options, expectedOwner: owner };
    if (typeof input !== "string") {
        try {
            return await storeImage(input, options);
        } catch (error) {
            if (error instanceof StorageQuotaError) throw error;
            const uploaded = await uploadBlobToServer(input, "image", owner, options?.signal, true, false, options?.onProgress).catch(() => null);
            if (!uploaded) throw error;
            const meta = await loadImageMeta(uploaded.url, options).catch(() => null);
            return { url: uploaded.url, width: meta?.width ?? 0, height: meta?.height ?? 0, bytes: uploaded.bytes, mimeType: uploaded.mime };
        }
    }
    if (isOwnMediaUrl(input)) {
        const meta = await loadImageMeta(input, options);
        assertMediaOwner(owner, options?.signal);
        if (!meta) throw new Error(i18n.t("common.imageReadFailed"));
        return { url: input, width: meta.width, height: meta.height, bytes: 0, mimeType: "" };
    }

    let blob: Blob;
    try {
        blob = await fetchImageBlob(input, options);
    } catch (error) {
        if (options?.signal?.aborted || isNamedError(error, IMAGE_RESPONSE_ERROR) || isNamedError(error, IMAGE_TIMEOUT_ERROR) || !/^https?:\/\//i.test(input)) throw error;
        const meta = await loadImageMeta(input, options, IMAGE_REMOTE_LOAD_TIMEOUT_MS);
        if (!meta) throw error;
        assertMediaOwner(owner, options?.signal);
        return { url: input, width: meta.width, height: meta.height, bytes: 0, mimeType: "" };
    }
    return storeImage(blob, options);
}

async function storeImage(blob: Blob, options?: ImageReadOptions): Promise<UploadedImage> {
    const owner = options?.expectedOwner === undefined ? currentMediaOwner() : options.expectedOwner;
    assertMediaOwner(owner, options?.signal);
    const storageKey = localMediaKey("image", owner, nanoid());
    const url = URL.createObjectURL(blob);
    try {
        const meta = await loadImageMeta(url, options);
        if (!meta) throw new Error(i18n.t("common.imageReadFailed"));
        throwIfAborted(options?.signal);

        const uploaded = await uploadBlobToServer(blob, "image", owner, options?.signal, false, false, options?.onProgress).catch((error) => { if (error instanceof StorageQuotaError) throw error; return null; });
        assertMediaOwner(owner, options?.signal);
        if (uploaded) {
            URL.revokeObjectURL(url);
            return { url: uploaded.url, width: meta.width, height: meta.height, bytes: blob.size, mimeType: blob.type.startsWith("image/") ? blob.type : "" };
        }

        await store.setItem(storageKey, blob);
        assertMediaOwner(owner, options?.signal);
        throwIfAborted(options?.signal);
        objectUrls.set(storageKey, url);
        await storeImagePreview(storageKey, blob);
        assertMediaOwner(owner, options?.signal);
        notifyLocalMediaOnly(owner);
        return { url, storageKey, width: meta.width, height: meta.height, bytes: blob.size, mimeType: blob.type.startsWith("image/") ? blob.type : "" };
    } catch (error) {
        objectUrls.delete(storageKey);
        URL.revokeObjectURL(url);
        await deleteImagePreview(storageKey);
        await store.removeItem(storageKey).catch(() => undefined);
        throw error;
    }
}

async function fetchImageBlob(url: string, options?: ImageReadOptions) {
    const controller = new AbortController();
    let timedOut = false;
    const abort = () => controller.abort();
    if (options?.signal?.aborted) abort();
    else options?.signal?.addEventListener("abort", abort, { once: true });
    const timer = window.setTimeout(() => {
        timedOut = true;
        controller.abort();
    }, IMAGE_DOWNLOAD_TIMEOUT_MS);
    try {
        const response = await fetch(withLocalProxy(url), { signal: controller.signal });
        if (!response.ok) throw namedError(IMAGE_RESPONSE_ERROR);
        return await response.blob();
    } catch (error) {
        if (timedOut) throw namedError(IMAGE_TIMEOUT_ERROR);
        if (options?.signal?.aborted) throw abortReason(options.signal);
        throw error;
    } finally {
        window.clearTimeout(timer);
        options?.signal?.removeEventListener("abort", abort);
    }
}

function loadImageMeta(url: string, options?: ImageReadOptions, timeoutMs = IMAGE_DECODE_TIMEOUT_MS) {
    return new Promise<{ width: number; height: number } | null>((resolve, reject) => {
        if (options?.signal?.aborted) return reject(abortReason(options.signal));
        const image = new Image();
        let settled = false;
        const finish = (value: { width: number; height: number } | null) => {
            if (settled) return;
            settled = true;
            window.clearTimeout(timer);
            options?.signal?.removeEventListener("abort", abort);
            image.onload = null;
            image.onerror = null;
            resolve(value);
        };
        const abort = () => {
            if (settled) return;
            settled = true;
            window.clearTimeout(timer);
            image.onload = null;
            image.onerror = null;
            reject(abortReason(options!.signal!));
        };
        const timer = window.setTimeout(() => finish(null), timeoutMs);
        options?.signal?.addEventListener("abort", abort, { once: true });
        image.onload = () => finish(image.naturalWidth && image.naturalHeight ? { width: image.naturalWidth, height: image.naturalHeight } : null);
        image.onerror = () => finish(null);
        image.src = url;
    });
}

type ImageHydration = { owner?: string; isActive?: () => boolean; cacheMiss?: boolean };

export async function resolveImageUrl(storageKey?: string, fallback = "", hydration?: ImageHydration) {
    if (!storageKey) return fallback;
    const owner = hydration?.owner ?? currentMediaOwner();
    assertLocalMediaOwner(storageKey, owner);
    const assertActive = () => {
        if (hydration?.isActive) {
            if (!hydration.isActive()) throw new DOMException("账户数据加载已中断", "AbortError");
        } else assertMediaOwner(owner);
    };
    assertActive();
    // Refresh a stale cover URL while concurrent refreshes still share one read.
    if (hydration?.cacheMiss && objectUrls.get(storageKey)) objectUrls.delete(storageKey);
    const url = await objectUrls.resolve(storageKey, async () => {
        const blob = await store.getItem<Blob>(storageKey);
        assertActive();
        return blob;
    });
    assertActive();
    return url || fallback;
}

/** 优先返回本地缩略图，没有则回落到原图 URL —— 对齐 BeefTV 的封面解析语义。 */
export async function resolvePreviewImageUrl(storageKey?: string, fallback = "", hydration?: ImageHydration) {
    if (!storageKey) return fallback;
    const owner = hydration?.owner ?? currentMediaOwner();

    if (hydration?.isActive) {
        if (!hydration.isActive()) throw new DOMException("账户数据加载已中断", "AbortError");

}

export async function ensureImagePreview(storageKey?: string) {
    if (!storageKey) return undefined;
    const owner = currentMediaOwner();
    try { assertLocalMediaOwner(storageKey, owner); } catch { return undefined; }
    return previewUrls.resolve(storageKey, async (isCurrent) => {
        const stored = await previewStore.getItem<StoredImagePreview>(storageKey).catch(() => null);
        if (currentMediaOwner() !== owner || !isCurrent()) return undefined;
        if (stored?.version === IMAGE_PREVIEW_VERSION) return stored.blob;
        // Keep the existing serial decoder queue; duplicate requests share this complete operation.
        const result = previewQueue.then(async () => {
            if (currentMediaOwner() !== owner || !isCurrent()) return undefined;
            const original = await getImageBlob(storageKey);
            return original ? createStoredImagePreview(storageKey, original, () => currentMediaOwner() === owner && isCurrent()) : undefined;
        });
        previewQueue = result.catch(() => undefined);
        return result;
    }).catch(() => undefined);
}

export async function cleanupUnusedImages(usedData: unknown) {
    const usedKeys = collectImageStorageKeys(usedData);
    await Promise.all([
        imageLogStore.iterate((value) => {

        videoLogStore.iterate((value) => {
            collectImageStorageKeys(value, usedKeys);
        }),
    ]);
    const unused: string[] = [];
    await store.iterate((_value, key) => {
        if (!usedKeys.has(key)) unused.push(key);
    });
    const orphanPreviews: string[] = [];
    await previewStore.iterate((_value, key) => {
        if (!usedKeys.has(key)) orphanPreviews.push(key);
    });
    await Promise.all([deleteStoredImages(unused), ...orphanPreviews.map(deleteImagePreview)]);
}

export function collectImageStorageKeys(value: unknown, keys = new Set<string>()) {
    if (!value || typeof value !== "object") return keys;
    if ("storageKey" in value && typeof value.storageKey === "string" && value.storageKey.startsWith("image:")) keys.add(value.storageKey);
    Object.values(value).forEach((item) => (Array.isArray(item) ? item.forEach((child) => collectImageStorageKeys(child, keys)) : collectImageStorageKeys(item, keys)));
    return keys;
}

function blobToDataUrl(blob: Blob) {
    return new Promise<string>((resolve, reject) => {
        const reader = new FileReader();
        reader.onload = () => resolve(String(reader.result || ""));
        reader.onerror = () => reject(new Error(i18n.t("common.imageReadFailed")));
        reader.readAsDataURL(blob);
    });
}

export type IndexedDbStoreUsage = { name: string; records: number; bytes: number };
export type IndexedDbDatabaseUsage = { name: string; version: number; bytes: number; stores: IndexedDbStoreUsage[] };
export type LocalStorageUsage = { usage: number; quota: number | null; contentBytes: number; databases: IndexedDbDatabaseUsage[] };

export async function readLocalStorageUsage(): Promise<LocalStorageUsage> {
    const [estimate, database] = await Promise.all([readStorageEstimate(), readDatabaseUsage("sundas")]);
    return { usage: estimate?.usage ?? database.bytes, quota: estimate?.quota ?? null, contentBytes: database.bytes, databases: [database] };
}

async function readStorageEstimate() {
    if (typeof navigator === "undefined" || !navigator.storage?.estimate) return null;
    try {
        return await navigator.storage.estimate();
    } catch {
        return null;
    }
}

function readDatabaseUsage(name: string) {
    return new Promise<IndexedDbDatabaseUsage>((resolve, reject) => {
        const request = indexedDB.open(name);
        request.onerror = () => reject(request.error);
        request.onsuccess = () => {
            const database = request.result;
            const names = Array.from(database.objectStoreNames);
            if (!names.length) {
                database.close();
                resolve({ name, version: database.version, bytes: 0, stores: [] });
                return;
            }
            const transaction = database.transaction(names, "readonly");
            Promise.all(names.map((storeName) => readStoreUsage(transaction.objectStore(storeName))))
                .then((stores) => resolve({ name, version: database.version, bytes: stores.reduce((total, store) => total + store.bytes, 0), stores: stores.sort((a, b) => b.bytes - a.bytes) }))
                .catch(reject)
                .finally(() => database.close());
        };
    });
}

function readStoreUsage(store: IDBObjectStore) {
    return new Promise<IndexedDbStoreUsage>((resolve, reject) => {
        let records = 0;
        let bytes = 0;
        const request = store.openCursor();
        request.onerror = () => reject(request.error);
        request.onsuccess = () => {
            const cursor = request.result;
            if (!cursor) {
                resolve({ name: store.name, records, bytes });
                return;
            }
            records += 1;
            bytes += valueBytes(cursor.value);
            cursor.continue();
        };
    });
}

function valueBytes(value: unknown) {
    if (value instanceof Blob) return value.size;
    if (value instanceof ArrayBuffer) return value.byteLength;
    if (ArrayBuffer.isView(value)) return value.byteLength;
    return new TextEncoder().encode(typeof value === "string" ? value : JSON.stringify(value)).byteLength;
}

/** 返回 true 表示已使用同源授权下载；其他 URL 继续原有下载行为。 */
export function downloadCloudMedia(value: string, filename: string) {
    const url = new URL(value, window.location.origin);
    if (url.origin !== window.location.origin || !url.pathname.startsWith('/media/private/')) return false;
    url.searchParams.set('download', filename);
    const anchor = document.createElement('a');
    anchor.href = url.toString();
    anchor.download = filename;
    anchor.rel = 'noreferrer';
    document.body.appendChild(anchor);
    anchor.click();
    anchor.remove();
    return true;
}

const pendingMediaRefresh = new WeakMap<HTMLMediaElement | HTMLImageElement, string>();

/** 只重取本站私有媒体；同一次失败不循环重试，读取成功后允许之后再次过期恢复。 */
export function refreshExpiredMedia(event: { currentTarget: HTMLMediaElement | HTMLImageElement }) {
    const video = event.currentTarget;
    const src = new URL(video.getAttribute('src') || '', window.location.origin);
    if (src.origin !== window.location.origin || !src.pathname.startsWith('/media/private/') || (video instanceof HTMLMediaElement && video.error && [1, 3].includes(video.error.code))) return;
    src.searchParams.delete('_refresh');
    const source = src.toString();
    if (pendingMediaRefresh.get(video) === source) return;
    pendingMediaRefresh.set(video, source);
    const position = video instanceof HTMLMediaElement ? video.currentTime : 0;
    src.searchParams.set('_refresh', String(Date.now()));
    const refreshedUrl = src.toString();
    if (video instanceof HTMLMediaElement) video.addEventListener('loadedmetadata', () => { if (video.src === refreshedUrl && position > 0) video.currentTime = position; }, { once: true });
    video.addEventListener(video instanceof HTMLMediaElement ? 'loadeddata' : 'load', () => { if (pendingMediaRefresh.get(video) === source) pendingMediaRefresh.delete(video); }, { once: true });
    video.src = refreshedUrl;
    if (video instanceof HTMLMediaElement) video.load();
}

                const now = new Date().toISOString();
                const project: CanvasProject = {
                    id: crypto.randomUUID(),
                    title: source.title || i18n.t("canvas.project.imported"),
                    createdAt: source.createdAt || now,
                    updatedAt: now,
                    nodes: source.nodes || [],
                    connections: source.connections || [],
                    chatSessions: source.chatSessions || [],
                    activeChatId: source.activeChatId || null,
                    backgroundMode: source.backgroundMode || "dots",
                    showImageInfo: source.showImageInfo || false,
                    viewport: source.viewport || initialViewport,
                    ...(source.production ? { production: source.production } : {}),
                    ...(source.directorScenes ? { directorScenes: source.directorScenes } : {}),
                };
                set((state) => ({ projects: [project, ...state.projects] }));
                return project.id;
            },
            openProject: (id) => {
                return get().projects.find((item) => item.id === id) || null;
            },
            renameProject: (id, title) =>
                set((state) => ({
                    projects: state.projects.map((project) => (project.id === id ? { ...project, title: title.trim() || project.title, updatedAt: nextUpdatedAt(project.updatedAt) } : project)),
                })),
            deleteProjects: (ids) =>
                set((state) => {
                    const now = new Date().toISOString();
                    const removing = new Set(ids);
                    const projects = state.projects.filter((project) => !removing.has(project.id));
                    const deletedProjects = [...state.deletedProjects.filter((item) => !removing.has(item.id)), ...ids.map((id) => ({ id, deletedAt: now }))];
                    return { projects, deletedProjects };
                }),
            clearDeletedProjects: (ids) =>
                set((state) => {
                    const removing = new Set(ids);
                    return { deletedProjects: state.deletedProjects.filter((item) => !removing.has(item.id)) };
                }),
            replaceProjects: (projects, deletedProjects = []) => set({ projects, deletedProjects }),
            updateProject: (id, patch) =>
                set((state) => {
                    const current = state.projects.find(project => project.id === id);
                    if (!current || Object.entries(patch).every(([key, value]) => current[key as keyof CanvasProject] === value)) return state;
                    return { projects: state.projects.map((project) => (project.id === id ? { ...project, ...patch, updatedAt: nextUpdatedAt(project.updatedAt) } : project)) };
                }),
        }),
        {
            name: CANVAS_STORE_KEY,
            skipHydration: true,
            storage: canvasStorage,
            partialize: (state) =>
                ({
                    projects: state.projects,
                    deletedProjects: state.deletedProjects,
                }) as StorageValue<CanvasStore>["state"],
            onRehydrateStorage: () => (_state, error) => {
                canvasHydrationError = error;
                if (!error) useCanvasStore.setState({ hydrated: true });
            },
        },
    ),
);

export async function flushCanvasAccount() {
    const previous = useCanvasStore.persist.getOptions().name;
    if (!previous?.trim()) throw new Error("画布缓存标识缺失，已停止保存和切换账户");
    if (saveTimer) clearTimeout(saveTimer);
    saveTimer = null;
    await canvasWrites.flush();
    if (previous !== CANVAS_STORE_KEY && queuedPersistState) {
        await writeCanvas(previous, JSON.stringify({ state: queuedPersistState, version: 0 }));
    }
}

export async function switchCanvasAccount(userId: string | null, alreadyFlushed = false) {
    if (!alreadyFlushed) await flushCanvasAccount();
    switchingCanvas = true;
    try {
        queuedPersistState = null;
        useCanvasStore.setState({ projects: [], deletedProjects: [], hydrated: false });
        useCanvasStore.persist.setOptions({ name: userId ? `${CANVAS_STORE_KEY}:${userId}` : CANVAS_STORE_KEY });
        if (userId) {
            canvasHydrationError = undefined;
            await useCanvasStore.persist.rehydrate();
            if (canvasHydrationError) throw canvasHydrationError;
        }
    } finally {
        switchingCanvas = false;
    }
}

export const useAssetStore = create<AssetStore>()(
    persist(
        (set, get) => ({
            hydrated: false,
            assets: [],
            deletedAssets: [],
            addAsset: (asset) => {
                const now = new Date().toISOString();
                const id = crypto.randomUUID();
                set((state) => ({ assets: [{ ...asset, id, createdAt: now, updatedAt: now } as Asset, ...state.assets] }));
                return id;
            },
            updateAsset: (id, patch) =>
                set((state) => ({
                    assets: state.assets.map((asset) => (asset.id === id ? ({ ...asset, ...patch, updatedAt: nextUpdatedAt(asset.updatedAt) } as Asset) : asset)),
                })),
            removeAsset: (id) =>
                set((state) => {
                    const assets = state.assets.filter((asset) => asset.id !== id);
                    get().cleanupImages({ assets });
                    return { assets, deletedAssets: [...new Set([...state.deletedAssets, id])] };
                }),
            clearDeletedAssets: (ids) => set((state) => ({ deletedAssets: state.deletedAssets.filter((id) => !ids.includes(id)) })),
            replaceAssets: (assets) => set({ assets }),
            // 媒体文件存储仍被所有账户共享，不能按当前账户引用删除其他账户的离线文件。
            cleanupImages: () => {},
        }),
        {
            name: ASSET_STORE_KEY,
            skipHydration: true,
            storage: assetStorage,
            partialize: (state) => ({ assets: state.assets, deletedAssets: state.deletedAssets }) as StorageValue<AssetStore>["state"],
            onRehydrateStorage: () => (_state, error) => {
                assetHydrationError = error;
                if (!error) useAssetStore.setState({ hydrated: true });
            },
        },
    ),
);

export async function flushAssetAccount() {
    await assetWrites.flush();
    const previous = useAssetStore.persist.getOptions().name;
    if (!previous?.trim()) throw new Error("资产缓存标识缺失，已停止保存和切换账户");
    if (previous !== ASSET_STORE_KEY && useAssetStore.getState().hydrated) {
        const { assets, deletedAssets } = useAssetStore.getState();
        await assetWrites.write(() => localForageStorage.setItem(previous, JSON.stringify({ state: { assets, deletedAssets }, version: 0 })));
    }
}

export async function switchAssetAccount(userId: string | null, alreadyFlushed = false) {
    if (!alreadyFlushed) await flushAssetAccount();
    switchingAssets = true;
    try {
        useAssetStore.setState({ assets: [], deletedAssets: [], hydrated: false });
        useAssetStore.persist.setOptions({ name: userId ? `${ASSET_STORE_KEY}:${userId}` : ASSET_STORE_KEY });
        if (userId) {
            assetHydrationError = undefined;
            await useAssetStore.persist.rehydrate();
            if (assetHydrationError) throw assetHydrationError;
        }
    } finally {
        switchingAssets = false;
    }
}

async function hydrateNow(set: (state: Partial<AuthState>) => void) {
    const epoch = authEpoch;
    const safeSet = (state: Partial<AuthState>) => { if (authEpoch === epoch) set(state); };
    safeSet({ ready: false, user: null, wallet: null });
    try {
        let data: { user: AuthUser | null; wallet?: WalletState };
        try { data = await apiRequest<{ user: AuthUser | null; wallet?: WalletState }>("/api/auth/me"); }
        catch { throw new Error("无法确认登录状态，请检查网络后刷新页面"); }
        if (authEpoch !== epoch) return;
        try { await switchCachedAccount(data.user?.id ?? null); }
        catch (error) {
            if (error instanceof Error && error.name === "LocalMediaOwnershipError") throw new Error(error.message);
            throw new Error("本地账户数据加载失败，请检查浏览器存储后刷新页面");
        }
        safeSet({ user: data.user, wallet: data.user ? (data.wallet ?? null) : null });
    } finally { safeSet({ ready: true }); }
}

export const useAuthStore = create<AuthState>((set, get) => ({
    user: null,
    wallet: null,
    ready: false,
    sessionEpoch: 0,

    hydrate: () => authTransitions.write(async () => { authEpoch += 1; set({ sessionEpoch: authEpoch }); await hydrateNow(set); }),

    externalSessionChanged: () => {
        const epoch = ++authEpoch;
        set({ ready: false, user: null, wallet: null, sessionEpoch: epoch });
        return authTransitions.write(async () => { if (authEpoch === epoch) await hydrateNow(set); });
    },

    refreshWallet: async () => {
        const owner = get().user?.id;
        const epoch = authEpoch;
        if (!owner || !get().ready) return;
        const request = ++walletRequest;
        const wallet = await apiRequest<WalletState>("/api/wallet", { headers: { "X-Sundas-User": owner } });
        if (get().ready && get().user?.id === owner && authEpoch === epoch && request === walletRequest) set({ wallet });
    },

    login: (email, password) => authTransitions.write(async () => {
        authEpoch += 1;
        const epoch = authEpoch;
        set({ ready: false, sessionEpoch: authEpoch });
        try {
            try { await switchCachedAccount(null); }
            catch { throw new Error("本地数据保存失败，请检查浏览器存储空间后重试"); }
            if (authEpoch === epoch) set({ user: null, wallet: null });
        } finally { if (authEpoch === epoch) set({ ready: true }); }
        const data = await apiRequest<{ user: AuthUser }>("/api/auth/login", { method: "POST", json: { email, password } });
        broadcastSessionChange();
        try { await hydrateNow(set); }
        catch (error) { throw new Error(`登录已提交，但会话确认失败：${error instanceof Error ? error.message : "请刷新页面"}`); }
        if (get().user?.id !== data.user.id) throw new Error("登录已提交，但会话未能确认，请刷新页面后重试");
        return data.user;
    }),

    register: (email, code, password, inviteCode) => authTransitions.write(async () => {
        authEpoch += 1;
        const epoch = authEpoch;
        set({ ready: false, sessionEpoch: authEpoch });
        try {
            try { await switchCachedAccount(null); }
            catch { throw new Error("本地数据保存失败，请检查浏览器存储空间后重试"); }
            if (authEpoch === epoch) set({ user: null, wallet: null });
        } finally { if (authEpoch === epoch) set({ ready: true }); }
        const data = await apiRequest<{ user: AuthUser }>("/api/auth/register", { method: "POST", json: { email, code, password, inviteCode: inviteCode || undefined } });
        broadcastSessionChange();
        try { await hydrateNow(set); }
        catch (error) { throw new Error(`注册已提交，请勿重复注册；会话确认失败：${error instanceof Error ? error.message : "请刷新页面后登录"}`); }
        if (get().user?.id !== data.user.id) throw new Error("注册已提交，但会话未能确认，请刷新页面后登录");
        return data.user;
    }),

    logout: () => authTransitions.write(async () => {
        const owner = get().user?.id;
        authEpoch += 1;
        const epoch = authEpoch;
        set({ ready: false, sessionEpoch: authEpoch });
        try {
            try { await switchCachedAccount(null); }
            catch { throw new Error("本地数据保存失败，尚未退出，请检查浏览器存储空间后重试"); }
            if (authEpoch !== epoch) return;
            if (authEpoch === epoch) set({ user: null, wallet: null });
            await apiRequest("/api/auth/logout", { method: "POST", headers: owner ? { "X-Sundas-User": owner } : undefined });
            broadcastSessionChange();
        } finally {
            if (authEpoch === epoch) set({ ready: true });
        }
    }),

    sendCode: async (email, purpose) => {
        await apiRequest("/api/auth/send-code", { method: "POST", json: { email, purpose } });
    },

    resetPassword: async (email, code, password) => {
        await apiRequest("/api/auth/reset-password", { method: "POST", json: { email, code, password } });

        await apiRequest("/api/auth/change-password", { method: "POST", headers: expectedUserId ? { "X-Sundas-User": expectedUserId } : undefined, json: { oldPassword, newPassword } });
    },

function initialWidth() {
    if (typeof window === "undefined") return CANVAS_SIDE_PANEL_DEFAULT_WIDTH;
    const stored = Number(localStorage.getItem(WIDTH_KEY));
    if (!stored) return CANVAS_SIDE_PANEL_DEFAULT_WIDTH;
    return Math.min(CANVAS_SIDE_PANEL_MAX_WIDTH, Math.max(CANVAS_SIDE_PANEL_MIN_WIDTH, stored));
}

function initialOpen() {
    if (typeof window === "undefined") return true;
    const stored = localStorage.getItem(OPEN_KEY);
    if (stored === "1") return true;
    if (stored === "0") return false;
    // 首次访问时小屏默认收起，避免移动端面板遮挡画布；桌面端保持展开。
    return !window.matchMedia("(max-width: 767px)").matches;
}

function initialNodeView(): CanvasNodeViewMode {
    if (typeof window === "undefined") return "list";
    return localStorage.getItem(NODE_VIEW_KEY) === "grid" ? "grid" : "list";
}

/** Best-effort default capability for a freshly fetched model name; user can override in the channel editor. */
export function guessCapability(name: string): ModelCapability {
    const value = name.toLowerCase();
    if (VIDEO_KEYWORDS.some((keyword) => value.includes(keyword))) return "video";
    if (AUDIO_KEYWORDS.some((keyword) => value.includes(keyword))) return "audio";
    if (IMAGE_KEYWORDS.some((keyword) => value.includes(keyword))) return "image";
    return "text";
}

function findChannelModel(config: AiConfig, value: string): { channel: ModelChannel; model: ChannelModel } | null {
    const decoded = decodeChannelModel(value);
    const name = decoded?.model || value;
    const channel = decoded ? config.channels.find((item) => item.id === decoded.channelId) : config.channels.find((item) => item.models.some((model) => model.name === name));
    const model = channel?.models.find((item) => item.name === name);
    return channel && model ? { channel, model } : null;
}

export function modelCapabilityOf(config: AiConfig, value: string): ModelCapability | undefined {
    return findChannelModel(config, value)?.model.capability;

export function modelMatchesCapability(config: AiConfig, value: string, capability?: ModelCapability) {
    if (!capability) return true;
    return modelCapabilityOf(config, value) === capability;

export function resolveModelForCapability(config: AiConfig, currentModel: string | undefined, capability: ModelCapability) {
    const defaultModel = capability === "image" ? config.imageModel : capability === "video" ? config.videoModel : capability === "audio" ? config.audioModel : config.textModel;
    const fallbackModel = capability === "image" ? defaultConfig.imageModel : capability === "video" ? defaultConfig.videoModel : capability === "audio" ? defaultConfig.audioModel : defaultConfig.textModel;
    for (const model of [currentModel, defaultModel]) {
        if (!model) continue;
        const configuredCapability = modelCapabilityOf(config, model);
        // Keep unavailable selections visible so the readiness check blocks the
        // request. Only a known capability switch may choose another model.
        if (!configuredCapability || configuredCapability === capability) return model;
    }
    return fallbackModel;
}

    if (!capability) return config.models;
    return config.channels.flatMap((channel) => channel.models.filter((model) => model.capability === capability).map((model) => encodeChannelModel(channel.id, model.name)));
}

/** Normalize a mixed list of raw model names or model objects into deduped ChannelModel entries. */
export function normalizeChannelModels(models: Array<string | ChannelModel> | undefined): ChannelModel[] {
    const seen = new Set<string>();
    const result: ChannelModel[] = [];
    for (const item of models || []) {
        const name = (typeof item === "string" ? item : item?.name || "").trim();
        if (!name || seen.has(name)) continue;
        seen.add(name);
        const capability = typeof item === "string" ? guessCapability(name) : item.capability || guessCapability(name);
        const script = typeof item === "string" ? undefined : item.script?.trim() || undefined;
        result.push({ name, capability, script });
    }
    return result;
}

export function createModelChannel(channel?: Partial<ModelChannel>): ModelChannel {
    const apiFormat = normalizeApiFormat(channel?.apiFormat);
    return {
        id: channel?.id?.trim() || nanoid(),
        name: channel?.name?.trim() || i18n.t("config.channels.newName"),
        baseUrl: channel?.baseUrl?.trim() || defaultBaseUrlForApiFormat(apiFormat),
        apiKey: channel?.apiKey || "",
        apiFormat,
        models: normalizeChannelModels(channel?.models),
    };
}

export function upsertChannelCredentials(
    config: AiConfig,
    input: { baseUrl?: string | null; apiKey?: string | null },
): ChannelCredentialsImportResult & { config: AiConfig } {
    const rawBaseUrl = input.baseUrl?.trim() || "";
    if (!rawBaseUrl) return { status: "missing-base-url", config };
    if (!isHttpBaseUrl(rawBaseUrl)) return { status: "invalid-base-url", config };

    const baseUrl = normalizeImportedBaseUrl(rawBaseUrl);
    const apiKey = input.apiKey?.trim() || "";
    const matchingIndex = config.channels.findIndex((channel) => normalizedBaseUrlKey(channel.baseUrl) === normalizedBaseUrlKey(baseUrl));

    if (matchingIndex >= 0) {
        const existing = config.channels[matchingIndex];
        if (existing.baseUrl === baseUrl && (!apiKey || existing.apiKey === apiKey)) {
            return { status: "updated", channelName: existing.name, config };
        }
        const updated = { ...existing, baseUrl, ...(apiKey ? { apiKey } : {}) };
        const channels = config.channels.map((channel, index) => (index === matchingIndex ? updated : channel));
        return { status: "updated", channelName: existing.name, config: { ...config, channels } };
    }

    const channel = createModelChannel({
        name: importedChannelName(baseUrl),
        baseUrl,
        apiKey,
        apiFormat: "openai",
        models: [],
    });
    return { status: "created", channelName: channel.name, config: { ...config, channels: [...config.channels, channel] } };
}

function isHttpBaseUrl(baseUrl: string) {
    try {
        const url = new URL(baseUrl);
        return (url.protocol === "http:" || url.protocol === "https:") && Boolean(url.hostname);
    } catch {
        return false;
    }
}

function normalizedBaseUrlKey(baseUrl: string) {
    try {
        return stripTrailingApiVersion(normalizeImportedBaseUrl(baseUrl));
    } catch {
        return stripTrailingApiVersion(baseUrl.trim().replace(/\/+$/, ""));
    }
}

function normalizeImportedBaseUrl(baseUrl: string) {
    const url = new URL(baseUrl.trim());
    url.hash = "";
    return url.toString().replace(/\/+$/, "");
}

function stripTrailingApiVersion(baseUrl: string) {
    return baseUrl.replace(/\/v1$/i, "");
}

function importedChannelName(baseUrl: string) {
    const hostname = new URL(baseUrl).hostname;
    return hostname.replace(/^(?:www|api)\./i, "") || i18n.t("config.channels.newName");
}

export function normalizeModelOptionValue(value: string | undefined, channels: ModelChannel[]) {
    const model = (value || "").trim();
    if (!model) return "";
    const decoded = decodeChannelModel(model);
    if (decoded) {
        const channel = channels.find((item) => item.id === decoded.channelId);
        return channel && channel.models.some((item) => item.name === decoded.model) ? model : "";
    }
    const channel = channels.find((item) => item.models.some((entry) => entry.name === model)) || channels[0];
    return channel && channel.models.some((item) => item.name === model) ? encodeChannelModel(channel.id, model) : model;
}

function normalizeChannels(config: AiConfig) {
    const persistedChannels = Array.isArray(config.channels) ? config.channels : [];
    const channels = persistedChannels.map((channel, index) =>
        createModelChannel({
            ...channel,
            id: channel.id || (index === 0 ? "default" : `channel-${index + 1}`),
            name: channel.name || (index === 0 ? i18n.t("config.channels.defaultName") : i18n.t("config.channels.indexedName", { index: index + 1 })),
            models: normalizeChannelModels(channel.models),
        }),
    );
    if (!channels.length) {
        channels.push(
            createModelChannel({
                id: "default",
                name: i18n.t("config.channels.defaultName"),
                baseUrl: config.baseUrl || defaultConfig.baseUrl,
                apiKey: config.apiKey || "",
                apiFormat: config.apiFormat || defaultConfig.apiFormat,
                models: normalizeChannelModels([config.model, config.imageModel, config.videoModel, config.textModel, config.audioModel].map(modelOptionName)),
            }),
        );
    }
    return channels;
}

export function buildApiUrl(baseUrl: string, path: string) {
    const normalizedBaseUrl = baseUrl.trim().replace(/\/+$/, "");
    const lowerBaseUrl = normalizedBaseUrl.toLowerCase();
    const apiBaseUrl = lowerBaseUrl.endsWith("/v1") ? normalizedBaseUrl : `${normalizedBaseUrl}/v1`;
    return withLocalProxy(`${apiBaseUrl}${path}`);
}

export function normalizeLocalProxyUrl(value: string) {
    const trimmed = value.trim().replace(/\/+$/, "");
    if (!trimmed) return "";
    return /^https?:\/\//i.test(trimmed) ? trimmed : `http://${trimmed}`;
}

/** Prefix an outgoing request with the local forwarding proxy so the browser is not blocked by CORS. */
export function withLocalProxy(url: string) {
    // Requests to our gateway must retain the browser session and never use a legacy forwarding proxy.
    if (typeof window !== "undefined" && /^https?:\/\//i.test(url) && new URL(url).origin === window.location.origin) return url;
    const { proxyEnabled, proxyUrl } = useConfigStore.getState().config;
    if (!proxyEnabled || !/^https?:\/\//i.test(url)) return url;
    const base = normalizeLocalProxyUrl(proxyUrl);
    if (!base || url.startsWith(`${base}/`)) return url;
    return `${base}/${url}`;
}

import { create } from "zustand";
import { nanoid } from "nanoid";

export type OperationPhase = "preparing" | "transferring" | "processing" | "done" | "error";

export type OperationProgress = {
    id: string;
    title: string;
    detail: string;
    phase: OperationPhase;
    totalBytes?: number;
    loadedBytes?: number;
    totalFiles?: number;
    doneFiles?: number;
    startedAt: number;
    updatedAt: number;
};

type ProgressStore = { entries: Record<string, OperationProgress>; remove: (id: string) => void };

/**
 * 操作进度（sundas 精简版）。
 *
 * BeefTV 这套 store 还带「画布节点任务」聚合与速率采样；sundas 目前只有导演台
 * 导出白膜视频会用到它，所以只保留 startOperation 的进度契约，
 * 供调用方上报进度/完成/失败，UI 侧按需读取 entries。
 */
export const useOperationProgress = create<ProgressStore>((set) => ({
    entries: {},
    remove: (id) =>
        set((state) => {
            const entries = { ...state.entries };
            delete entries[id];
            return { entries };
        }),
}));

export function startOperation(title: string, options: { detail?: string; totalBytes?: number; totalFiles?: number } = {}) {
    const id = nanoid();
    const startedAt = Date.now();
    let finished = false;

    const patch = (next: Partial<OperationProgress>) => {
        if (finished) return;
        useOperationProgress.setState((state) => {
            const previous = state.entries[id];
            if (!previous) return state;
            return { entries: { ...state.entries, [id]: { ...previous, ...next, updatedAt: Date.now() } } };
        });
    };

    useOperationProgress.setState((state) => ({
        entries: {
            ...state.entries,
            [id]: {
                id,
                title,
                detail: options.detail || "正在准备文件",
                phase: "preparing",
                totalBytes: options.totalBytes,
                totalFiles: options.totalFiles,
                startedAt,
                updatedAt: startedAt,
            },
        },
    }));

    const settle = (phase: "done" | "error", detail: string) => {
        patch({ phase, detail });
        finished = true;
        if (phase === "done") window.setTimeout(() => useOperationProgress.getState().remove(id), 4500);
    };

    return {
        update: patch,
        progress: (loaded: number, total?: number) =>
            patch({ loadedBytes: loaded, totalBytes: total, phase: total && loaded >= total ? "processing" : "transferring", detail: total && loaded >= total ? "文件已传输，正在确认与保存" : "正在传输文件" }),
        done: (detail = "处理完成") => settle("done", detail),
        fail: (detail = "处理没有完成，请重试") => settle("error", detail),
        dismiss: () => {
            finished = true;
            useOperationProgress.getState().remove(id);
        },
    };
}

import { create } from "zustand";

// The Agent panel dispatches commands through this store to set workbench prompts and optionally start generation.
// The panel writes model, quality, size, count, and other options to use-config-store, which workbench pages read directly.
// Prompt and run are sent here; pages identify new commands by nonce and call clear after consuming them.

export type WorkbenchCommand = {
    nonce: number;
    taskId?: string;
    prompt?: string;
    run: boolean;
};

export type WorkbenchGenerationTask = {
    id: string;
    kind: "image" | "video";
    status: "queued" | "running" | "succeeded" | "failed";
    prompt?: string;
    createdAt: string;
    updatedAt: string;
    successCount?: number;
    failCount?: number;
    error?: string;
};

type WorkbenchAgentStore = {
    imageCommand: WorkbenchCommand | null;
    videoCommand: WorkbenchCommand | null;
    tasks: WorkbenchGenerationTask[];
    dispatchImage: (command: Omit<WorkbenchCommand, "nonce" | "taskId">) => string | undefined;
    dispatchVideo: (command: Omit<WorkbenchCommand, "nonce" | "taskId">) => string | undefined;
    updateTask: (id: string, patch: Partial<Pick<WorkbenchGenerationTask, "status" | "successCount" | "failCount" | "error">>) => void;
    clearImageCommand: () => void;
    clearVideoCommand: () => void;
    reset: () => void;
};

let nonce = 0;
const nextNonce = () => (nonce += 1);

export const useWorkbenchAgentStore = create<WorkbenchAgentStore>((set) => ({
    imageCommand: null,
    videoCommand: null,
    tasks: [],
    dispatchImage: (command) => {
        const commandNonce = nextNonce();
        const task = command.run ? createTask("image", commandNonce, command.prompt) : undefined;
        set((state) => ({ imageCommand: { ...command, nonce: commandNonce, taskId: task?.id }, tasks: task ? [task, ...state.tasks].slice(0, 30) : state.tasks }));
        return task?.id;
    },
    dispatchVideo: (command) => {
        const commandNonce = nextNonce();
        const task = command.run ? createTask("video", commandNonce, command.prompt) : undefined;
        set((state) => ({ videoCommand: { ...command, nonce: commandNonce, taskId: task?.id }, tasks: task ? [task, ...state.tasks].slice(0, 30) : state.tasks }));
        return task?.id;
    },
    updateTask: (id, patch) => set((state) => ({ tasks: state.tasks.map((task) => (task.id === id ? { ...task, ...patch, updatedAt: new Date().toISOString() } : task)) })),
    clearImageCommand: () => set({ imageCommand: null }),
    clearVideoCommand: () => set({ videoCommand: null }),
    reset: () => set({ imageCommand: null, videoCommand: null, tasks: [] }),
}));

function createTask(kind: "image" | "video", commandNonce: number, prompt?: string): WorkbenchGenerationTask {
    const now = new Date().toISOString();
    return { id: `${kind}-${commandNonce}`, kind, status: "queued", prompt, createdAt: now, updatedAt: now };
}
