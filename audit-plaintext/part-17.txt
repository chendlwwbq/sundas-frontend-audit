function PaymentNotifications() {
    const permissions = useAdminPermissions();
    const apiRequest = useAdminRequest();
    const { message } = AntApp.useApp();
    const [data, setData] = useState<{ items: PendingNotification[]; total: number }>({ items: [], total: 0 });
    const [page, setPage] = useState(1);
    const [reviewFilter, setReviewFilter] = useState("unreviewed");
    const [loading, setLoading] = useState(false);
    const [target, setTarget] = useState<PendingNotification | null>(null);
    const [note, setNote] = useState("");
    const [saving, setSaving] = useState(false);
    const loadSequence = useRef(0);
    async function load() {
        const sequence = ++loadSequence.current;
        setLoading(true);
        try { const result = await apiRequest<{ items: PendingNotification[]; total: number }>(`/api/admin/payment-notifications/unprocessed?page=${page}&pageSize=20&review=${reviewFilter}`); if (sequence === loadSequence.current) setData(result); }
        catch (error) { if (sequence === loadSequence.current) message.error(error instanceof Error ? error.message : String(error)); }
        finally { if (sequence === loadSequence.current) setLoading(false); }
    }
    useEffect(() => { void load(); return () => { loadSequence.current++; }; }, [page, reviewFilter]);
    async function review() {
        if (!target || !note.trim()) { message.warning("请填写核对说明"); return; }
        setSaving(true);
        try {
            await apiRequest(`/api/admin/payment-notifications/${target.id}/review`, { method: "POST", json: { note: note.trim() } });
            message.success("已记录人工核对，不改变支付状态"); setTarget(null); setNote(""); await load();
        } catch (error) { message.error(error instanceof Error ? error.message : String(error)); }
        finally { setSaving(false); }
    }
    const columns: TableColumnsType<PendingNotification> = [
        { title: "收到时间", dataIndex: "createdAt", render: (value: string) => dayjs(value).format("YYYY-MM-DD HH:mm:ss") },

const SUMMARY_PRESETS: { label: string; value: SummaryPreset }[] = [

function weekStart(now: Dayjs) {
    return now.subtract((now.day() + 6) % 7, "day").startOf("day");
}
function rangeForPreset(preset: SummaryPreset): [Dayjs, Dayjs] {
    const now = dayjs();
    if (preset === "24h") return [now.subtract(24, "hour"), now];
    if (preset === "week") return [weekStart(now), now];
    if (preset === "month") return [now.startOf("month"), now];
    if (preset === "lastWeek") return [weekStart(now).subtract(7, "day"), weekStart(now)];
    if (preset === "lastMonth") return [now.subtract(1, "month").startOf("month"), now.startOf("month")];
    return [now.startOf("day"), now];
}
function formatRange(range: [Dayjs, Dayjs]) {
    return `${range[0].format("YYYY-MM-DD HH:mm")} ~ ${range[1].format("YYYY-MM-DD HH:mm")}`;
}
function formatMoney(cents: number) {
    return `¥${(cents / 100).toLocaleString("zh-CN", { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`;
}

            message.error(error instanceof Error ? error.message : String(error));

            setLoading(false);

    }

    useEffect(() => {
        void load();

                title: "替换当前启用内容？",
                content: `启用后会立即停用「${label}」，并记录两条审计日志。`,
                okText: "替换并启用",
                cancelText: "取消",
                onOk: () => resolve(true),
                onCancel: () => resolve(false),

        });
    }

    async function saveItem(draft: ItemDraft) {
        if (draft.placement === "topbar" && !draft.title.trim()) return message.warning("请填写顶部活动条文案");
        if (draft.placement !== "topbar" && draft.placement !== "corner" && !draft.imageUrl) return message.warning("请上传图片");
        if ((draft.placement === "carousel" && !draft.alt.trim()) || ((draft.placement === "popup" || draft.placement === "corner") && !draft.title.trim())) return message.warning(draft.placement === "carousel" ? "请填写图片 alt 文案" : "请填写标题");
        if (draft.actionType !== "none" && !draft.ctaLabel && draft.placement !== "carousel") return message.warning(`请填写按钮文案${draft.placement === "corner" ? "（右下角卡片必须要有按钮，否则用户无法关闭以外的操作）" : ""}`);
        if (draft.actionType === "internal" && !draft.actionValue) return message.warning("请选择站内路径");
        if ((draft.actionType === "external" || draft.actionType === "copy") && !draft.actionValue.trim()) return message.warning("请填写点击目标");
        if (draft.enabled && (draft.placement === "topbar" || draft.placement === "popup" || draft.placement === "corner")) {
            const other = data.items.find((item) => item.placement === draft.placement && item.enabled && item.id !== draft.id);
            if (other && !(await confirmReplace(other.title || other.alt || "当前内容"))) return;
        }
        setSaving(true);
        try {
            const { id, ...payload } = draft;
            await apiRequest(id ? `/api/admin/promo/items/${id}` : "/api/admin/promo/items", {
                method: id ? "PATCH" : "POST",
                json: payload,
            });
            setItemDraft(null);
            await load();
            message.success(id ? "已保存" : "已添加");
        } catch (error) {
            message.error(error instanceof Error ? error.message : String(error));
        } finally {
            setSaving(false);
        }
    }

            const other = data.items.find((entry) => entry.placement === item.placement && entry.enabled && entry.id !== item.id);
            if (other && !(await confirmReplace(other.title || other.alt || "当前内容"))) return;
        }
        try {
            await apiRequest(`/api/admin/promo/items/${item.id}`, { method: "PATCH", json: { enabled: !item.enabled } });

            setData((state) => ({ ...state, items: state.items.map((item) => item.placement === "carousel" ? { ...item, sort: itemOrders.get(item.id) ?? item.sort } : item) }));

            message.error(error instanceof Error ? error.message : String(error));
            await load();
        }

    function moveItem(item: PromoItem, direction: -1 | 1) {
        const list = data.items.filter((entry) => entry.placement === "carousel").sort((a, b) => a.sort - b.sort);
        const index = list.findIndex((entry) => entry.id === item.id);
        const target = index + direction;
        if (target < 0 || target >= list.length) return;
        const next = [...list];
        [next[index], next[target]] = [next[target], next[index]];
        void reorderItems("carousel", next.map((entry) => entry.id));
    }

    function moveContact(contact: PromoContact, direction: -1 | 1) {
        const list = [...data.contacts].sort((a, b) => a.sort - b.sort);
        const index = list.findIndex((entry) => entry.id === contact.id);
        const target = index + direction;
        if (target < 0 || target >= list.length) return;
        const next = [...list];
        [next[index], next[target]] = [next[target], next[index]];
        void reorderItems("contact", next.map((entry) => entry.id));
    }

function AdminImage({ src, alt, className, compact = false }: { src: string; alt: string; className?: string; compact?: boolean }) {
    const [failed, setFailed] = useState(false);
    useEffect(() => setFailed(false), [src]);
    if (!src || failed) {
        return (
            <div className={`grid place-items-center bg-[var(--admin-panel-soft)] text-[var(--admin-muted)] ${className ?? ""}`}>
                {compact ? <ImageOff className="size-4" /> : <span className="flex flex-col items-center gap-1 text-[9px]"><ImageOff className="size-4" />图片不可用</span>}
            </div>
        );
    }
    return <img src={src} alt={alt} className={className} onError={() => setFailed(true)} />;
}

function CarouselPanel({
    items,
    onAdd,
    onEdit,
    onToggle,
    onDelete,
    onMove,
    onReorder,
}: {
    items: PromoItem[];
    onAdd: () => void;
    onEdit: (item: PromoItem) => void;
    onToggle: (item: PromoItem) => void;
    onDelete: (item: PromoItem) => void;
    onMove: (item: PromoItem, direction: -1 | 1) => void;
    onReorder: (kind: "carousel" | "contact", ids: string[]) => Promise<void>;
}) {
    const canManage = useAdminPermissions().includes("promo.manage");
    const [draggingId, setDraggingId] = useState<string | null>(null);

    function drop(target: PromoItem) {
        if (!draggingId || draggingId === target.id) return;
        const from = items.findIndex((item) => item.id === draggingId);
        const to = items.findIndex((item) => item.id === target.id);
        if (from < 0 || to < 0) return;
        const next = [...items];
        const [moved] = next.splice(from, 1);
        next.splice(to, 0, moved);
        setDraggingId(null);
        void onReorder("carousel", next.map((item) => item.id));
    }

    return (
        <AdminPanel
            title="首页轮播"
            description="拖拽调整播放顺序；启用哪些就轮播哪些，没有数量上限。只支持图片，整张图可点击。"
            actions={canManage && <Button type="primary" size="small" icon={<Plus className="size-3.5" />} onClick={onAdd}>新增图片</Button>}
            bodyClassName="space-y-2"
        >
            {items.length ? items.map((item, index) => (
                <div
                    key={item.id}
                    draggable={canManage}
                    onDragStart={() => setDraggingId(item.id)}
                    onDragEnd={() => setDraggingId(null)}
                    onDragOver={(event) => event.preventDefault()}
                    onDrop={() => drop(item)}
                    className={`flex items-center gap-3 rounded-[10px] border bg-[var(--admin-panel)] p-3 transition ${draggingId === item.id ? "border-[var(--admin-ink)] opacity-60" : "border-[var(--admin-line)]"}`}
                >
                    <GripVertical className="size-4 shrink-0 cursor-grab text-[var(--admin-muted)]" />
                    <AdminImage src={item.imageUrl} alt={item.alt} className="h-12 w-20 shrink-0 rounded-md object-cover" />
                    <div className="min-w-0 flex-1">
                        <div className="line-clamp-1 text-sm font-medium text-[var(--admin-ink)]">{item.alt || "未填写 alt"}</div>
                        <div className="mt-1 flex flex-wrap items-center gap-2 text-[11px] text-[var(--admin-muted)]">
                            <span>{index + 1}</span>
                            <span>·</span>
                            <span>{actionSummary(item)}</span>
                        </div>
                    </div>
                    {canManage && <div className="flex shrink-0 items-center gap-1">
                        <Tooltip title="上移"><Button size="small" type="text" icon={<ArrowUp className="size-3.5" />} disabled={index === 0} onClick={() => onMove(item, -1)} /></Tooltip>
                        <Tooltip title="下移"><Button size="small" type="text" icon={<ArrowDown className="size-3.5" />} disabled={index === items.length - 1} onClick={() => onMove(item, 1)} /></Tooltip>
                        <Switch size="small" checked={item.enabled} onChange={() => onToggle(item)} />
                        <Button size="small" type="text" icon={<Pencil className="size-3.5" />} onClick={() => onEdit(item)} aria-label="编辑" />
                        <Button size="small" type="text" danger icon={<Trash2 className="size-3.5" />} onClick={() => onDelete(item)} aria-label="删除" />
                    </div>}
                </div>
            )) : <Empty image={Empty.PRESENTED_IMAGE_SIMPLE} description="还没有轮播图" />}
        </AdminPanel>
    );
}

function ImageUploadField({ label, value, onChange, hint, required, ratio, recommended }: { label: string; value: string; onChange: (url: string) => void; hint?: string; required?: boolean; ratio?: number; recommended?: string }) {
    const scope = useAdminScope();
    const { message } = AntApp.useApp();
    const inputRef = useRef<HTMLInputElement>(null);
    const [busy, setBusy] = useState(false);
    const [actualRatio, setActualRatio] = useState<number | null>(null);
    const [crop, setCrop] = useState<{ file: File; src: string } | null>(null);

    useEffect(() => {
        if (!value) return setActualRatio(null);
        const image = new Image();
        image.onload = () => { if (scope.current()) setActualRatio(image.naturalWidth / image.naturalHeight); };
        image.onerror = () => { if (scope.current()) setActualRatio(null); };
        image.src = value;
    }, [value]);

    useEffect(() => () => {
        if (crop) URL.revokeObjectURL(crop.src);
    }, [crop]);

    const ratioMismatch = Boolean(ratio && actualRatio && Math.abs(actualRatio - ratio) / ratio > 0.1);

    async function upload(file: Blob) {
        if (!scope.owner || !scope.current()) return;
        setBusy(true);
        try {
            const uploaded = await uploadBlobToServer(file, "promo", scope.owner);
            if (!scope.current()) return;
            if (!uploaded) throw new Error("上传失败，请稍后重试");
            onChange(uploaded.url);
            message.success("图片已上传");
            if (crop) URL.revokeObjectURL(crop.src);
            setCrop(null);
        } catch (error) {
            if (scope.current()) message.error(error instanceof Error ? error.message : String(error));
        } finally {
            if (scope.current()) setBusy(false);
        }
    }

    function pick(file: File | undefined) {
        if (!file || !scope.current()) return;
        if (!["image/jpeg", "image/png", "image/webp"].includes(file.type)) return void message.error("只支持 JPG / PNG / WebP");
        if (file.size > 10 * 1024 * 1024) return void message.error("图片不能超过 10MB");
        if (inputRef.current) inputRef.current.value = "";
        if (ratio) {
            setCrop({ file, src: URL.createObjectURL(file) });
            return;
        }
        void upload(file);
    }

    function closeCrop() {
        if (crop) URL.revokeObjectURL(crop.src);
        setCrop(null);
    }

    return (
        <div>
            <div className="mb-1.5 flex items-center gap-1 text-xs font-medium text-[var(--admin-muted)]">
                {label}{required ? <span className="text-[var(--admin-danger)]">*</span> : null}
            </div>
            <div className="flex gap-3 rounded-[9px] border border-dashed border-[var(--admin-line-strong)] bg-[var(--admin-panel-soft)] p-2.5">
                <div className="relative grid h-20 w-32 shrink-0 place-items-center overflow-hidden rounded-md border border-[var(--admin-line)] bg-[var(--admin-panel)] text-[var(--admin-muted)]">
                    {value ? <img src={value} alt="" className="h-full w-full object-cover" /> : <ImageOff className="size-5" />}
                </div>
                <div className="flex min-w-0 flex-col justify-center gap-2">
                    <div className="flex flex-wrap gap-2">
                        <Button size="small" icon={<Upload className="size-3.5" />} loading={busy} onClick={() => inputRef.current?.click()}>{value ? "替换图片" : "上传图片"}</Button>
                        {value ? <Button size="small" type="text" danger icon={<X className="size-3.5" />} onClick={() => onChange("")}>移除</Button> : null}
                    </div>
                    <div className={`text-[10px] leading-4 ${ratioMismatch ? "text-[var(--admin-warning)]" : "text-[var(--admin-muted)]"}`}>
                        {ratioMismatch && actualRatio ? `当前比例约 ${actualRatio.toFixed(2)}:1，建议 ${recommended ?? ""}；上传时会按比例裁切。` : hint || "JPG / PNG / WebP，单图 ≤ 10MB"}
                    </div>
                </div>
            </div>
            <input ref={inputRef} type="file" accept="image/jpeg,image/png,image/webp" className="hidden" onChange={(event) => pick(event.target.files?.[0])} />
            <ImageCropDialog file={crop?.file ?? null} src={crop?.src ?? ""} ratio={ratio ?? 1} busy={busy} onCancel={closeCrop} onConfirm={(file) => void upload(file)} />
        </div>
    );
}

function ImageCropDialog({ file, src, ratio, busy, onCancel, onConfirm }: { file: File | null; src: string; ratio: number; busy: boolean; onCancel: () => void; onConfirm: (file: File) => void }) {
    const [zoom, setZoom] = useState(1);
    const [offset, setOffset] = useState({ x: 0, y: 0 });
    const [natural, setNatural] = useState({ width: 0, height: 0 });
    const [frame, setFrame] = useState({ width: 0, height: 0 });
    const [frameElement, setFrameElement] = useState<HTMLDivElement | null>(null);
    const imageRef = useRef<HTMLImageElement>(null);
    const dragRef = useRef<{ x: number; y: number; offsetX: number; offsetY: number } | null>(null);

    useEffect(() => {
        setZoom(1);
        setOffset({ x: 0, y: 0 });
        setNatural({ width: 0, height: 0 });
    }, [file]);

    useEffect(() => {
        if (!frameElement) return;
        const update = () => setFrame({ width: frameElement.clientWidth, height: frameElement.clientHeight });
        update();
        const observer = new ResizeObserver(update);
        observer.observe(frameElement);
        return () => observer.disconnect();
    }, [frameElement]);

    const baseScale = natural.width && frame.width ? Math.max(frame.width / natural.width, frame.height / natural.height) : 1;
    const scale = baseScale * zoom;
    const renderedWidth = natural.width * scale;
    const renderedHeight = natural.height * scale;
    const maxX = Math.max(0, (renderedWidth - frame.width) / 2);
    const maxY = Math.max(0, (renderedHeight - frame.height) / 2);
    const left = (frame.width - renderedWidth) / 2 + offset.x;
    const top = (frame.height - renderedHeight) / 2 + offset.y;

    useEffect(() => {
        setOffset((current) => ({
            x: Math.max(-maxX, Math.min(maxX, current.x)),
            y: Math.max(-maxY, Math.min(maxY, current.y)),
        }));
    }, [maxX, maxY]);

    function pointerDown(event: React.PointerEvent<HTMLDivElement>) {
        event.currentTarget.setPointerCapture(event.pointerId);
        dragRef.current = { x: event.clientX, y: event.clientY, offsetX: offset.x, offsetY: offset.y };
    }

    function pointerMove(event: React.PointerEvent<HTMLDivElement>) {
        const drag = dragRef.current;
        if (!drag) return;
        setOffset({
            x: Math.max(-maxX, Math.min(maxX, drag.offsetX + event.clientX - drag.x)),
            y: Math.max(-maxY, Math.min(maxY, drag.offsetY + event.clientY - drag.y)),
        });
    }

    function confirm() {
        const image = imageRef.current;
        if (!image || !natural.width || !frame.width) return;
        const sourceX = Math.max(0, (renderedWidth - frame.width) / 2 - offset.x) / scale;
        const sourceY = Math.max(0, (renderedHeight - frame.height) / 2 - offset.y) / scale;
        const sourceWidth = frame.width / scale;
        const sourceHeight = frame.height / scale;
        const outputWidth = Math.min(1200, Math.round(sourceWidth));
        const outputHeight = Math.max(1, Math.round(outputWidth / ratio));
        const canvas = document.createElement("canvas");
        canvas.width = outputWidth;
        canvas.height = outputHeight;
        const context = canvas.getContext("2d");
        if (!context) return;
        context.drawImage(image, sourceX, sourceY, sourceWidth, sourceHeight, 0, 0, outputWidth, outputHeight);
        canvas.toBlob((blob) => {
            if (!blob || !file) return;
            onConfirm(new File([blob], file.name.replace(/\.[^.]+$/, "") + ".jpg", { type: "image/jpeg" }));
        }, "image/jpeg", 0.92);
    }

    return (
        <Modal
            title={ratio === 1 ? "裁切二维码" : "裁切图片"}
            open={Boolean(file)}
            onCancel={onCancel}
            width={540}
            footer={<div className="flex justify-end gap-2"><Button onClick={onCancel}>取消</Button><Button type="primary" loading={busy} onClick={confirm}>裁切并上传</Button></div>}
        >
            <div className="flex flex-col items-center gap-4">
                <div
                    ref={setFrameElement}
                    className="promo-crop-frame"
                    style={{ aspectRatio: ratio }}
                    onPointerDown={pointerDown}
                    onPointerMove={pointerMove}
                    onPointerUp={() => { dragRef.current = null; }}
                    onPointerCancel={() => { dragRef.current = null; }}
                >
                    {src ? (
                        <img
                            ref={imageRef}
                            src={src}
                            alt="待裁切图片"
                            draggable={false}
                            onLoad={(event) => setNatural({ width: event.currentTarget.naturalWidth, height: event.currentTarget.naturalHeight })}
                            style={{ width: renderedWidth || "100%", height: renderedHeight || "auto", left, top }}
                        />
                    ) : null}
                    <div className="promo-crop-grid" />
                </div>
                <div className="w-full">
                    <div className="mb-1 flex items-center justify-between text-[11px] text-[var(--admin-muted)]">
                        <span>拖动图片调整位置</span>
                        <span>缩放 {zoom.toFixed(1)}×</span>
                    </div>
                    <Slider min={1} max={3} step={0.01} value={zoom} onChange={setZoom} />
                </div>
                <p className="m-0 text-center text-[11px] leading-4 text-[var(--admin-muted)]">框内即为最终显示区域，建议保持二维码完整。</p>
            </div>
        </Modal>
    );
}

type Code = { id: string; code: string; credits: number; status: string; expiresAt: string | null; usedAt: string | null; usedBy: string | null };

const STATUS: Record<string, { label: string; tone: AdminBadgeTone }> = {
    unused: { label: "未使用", tone: "positive" },
    used: { label: "已使用", tone: "neutral" },
    void: { label: "已作废", tone: "danger" },
};

export function RedeemTab() {
    const permissions = useAdminPermissions();
    const apiRequest = useAdminRequest();
    const { message, modal } = AntApp.useApp();
    const [batches, setBatches] = useState<Batch[]>([]);
    const [codes, setCodes] = useState<Code[]>([]);
    const [batchId, setBatchId] = useState("");
    const [status, setStatus] = useState("");
    const [page, setPage] = useState(1);
    const [total, setTotal] = useState(0);
    const [loading, setLoading] = useState(false);
    const [activeTab, setActiveTab] = useState("codes");
    const [generating, setGenerating] = useState(false);
    const [generateOpen, setGenerateOpen] = useState(false);
    const [name, setName] = useState("");
    const [credits, setCredits] = useState(100);
    const [count, setCount] = useState(10);
    const [expiresAt, setExpiresAt] = useState<Dayjs | null>(null);

    async function loadBatches() {
        const data = await apiRequest<{ batches: Batch[] }>("/api/admin/redeem/batches");
        setBatches(data.batches);
    }

    async function loadCodes() {
        setLoading(true);
        try {
            const params = new URLSearchParams({ page: String(page), pageSize: "20", status, batchId });
            const data = await apiRequest<{ items: Code[]; total: number }>(`/api/admin/redeem/codes?${params.toString()}`);
            setCodes(data.items);
            setTotal(data.total);
        } catch (error) {
            message.error(error instanceof Error ? error.message : String(error));
        } finally {
            setLoading(false);
        }
    }

    useEffect(() => {
        void loadBatches().catch((error: unknown) => message.error(error instanceof Error ? error.message : String(error)));
    }, []);

    useEffect(() => {
        void loadCodes();
    }, [batchId, status, page]);

    async function generate() {
        setGenerating(true);
        try {
            const result = await apiRequest<{ generated: number }>("/api/admin/redeem/batches", {
                method: "POST",
                json: { name, credits, count, expiresAt: expiresAt ? expiresAt.toISOString() : null },
            });
            message.success(`已生成 ${result.generated} 个兑换码`);
            setGenerateOpen(false);
            setName("");
            setActiveTab("codes");
            await loadBatches();
            await loadCodes();
        } catch (error) {
            message.error(error instanceof Error ? error.message : String(error));
        } finally {
            setGenerating(false);
        }
    }

    function voidCode(code: Code) {
        modal.confirm({
            title: "作废该兑换码？",
            content: code.code,
            okButtonProps: { danger: true },
            onOk: async () => {
                await apiRequest(`/api/admin/redeem/codes/${code.id}/void`, { method: "POST" });
                await loadCodes();
            },
        });
    }

    const batchColumns: TableColumnsType<Batch> = [
        { title: "批次", dataIndex: "name", render: (value: string) => <span className="font-medium text-[var(--admin-ink)]">{value || "未命名批次"}</span> },
        { title: "面额", dataIndex: "credits", align: "right", render: (value: number) => <span className="tabular-nums">{value.toLocaleString("zh-CN")}</span> },

        { title: "已用", dataIndex: "used", align: "right", render: (value: number) => <span className="tabular-nums">{value}</span> },
        { title: "过期", dataIndex: "expiresAt", render: (value: string | null) => <span className="text-[var(--admin-muted)]">{value ? dayjs(value).format("YYYY-MM-DD") : "永久"}</span> },
        {
            title: "操作",
            key: "actions",
            align: "right",
            render: (_value, record) => (
                <div className="flex justify-end gap-1.5">
                    <Button
                        size="small"
                        onClick={() => {
                            setBatchId(record.id);
                            setPage(1);
                            setActiveTab("codes");
                        }}
                    >
                        查看
                    </Button>
                    {permissions.includes("redeem.export") && <Button size="small" onClick={() => window.open(`/api/admin/redeem/batches/${record.id}/export`, "_blank")}>
                        导出
                    </Button>}
                </div>
            ),
        },
    ];

    const codeColumns: TableColumnsType<Code> = [
        { title: "兑换码", dataIndex: "code", render: (value: string) => <span className="font-medium tabular-nums text-[var(--admin-ink)]">{value}</span> },
        { title: "面额", dataIndex: "credits", align: "right", render: (value: number) => <span className="tabular-nums">{value.toLocaleString("zh-CN")}</span> },
        { title: "状态", dataIndex: "status", render: (value: string) => <AdminBadge tone={STATUS[value]?.tone ?? "neutral"}>{STATUS[value]?.label ?? value}</AdminBadge> },
        { title: "使用者", dataIndex: "usedBy", render: (value: string | null) => <span className="text-[var(--admin-muted)]">{value ?? "-"}</span> },
        { title: "使用时间", dataIndex: "usedAt", render: (value: string | null) => <span className="tabular-nums text-[var(--admin-muted)]">{value ? dayjs(value).format("MM-DD HH:mm") : "-"}</span> },
        {
            title: "操作",
            key: "actions",
            align: "right",
            render: (_value, record) => (permissions.includes("redeem.manage") && record.status === "unused" ? <Button size="small" danger onClick={() => voidCode(record)}>作废</Button> : null),
        },
    ];

    return (
        <div>
            <Tabs
                activeKey={activeTab}
                onChange={setActiveTab}
                tabBarExtraContent={
                    permissions.includes("redeem.manage") && <Button type="primary" icon={<Plus className="size-4" />} onClick={() => setGenerateOpen(true)}>
                        生成兑换码
                    </Button>
                }
                items={[
                    {
                        key: "codes",
                        label: `兑换码 ${total}`,
                        children: (
                            <AdminPanel flush>
                                <AdminToolbar>
                                    <Select
                                        className="w-36"
                                        value={status}
                                        onChange={(value) => {
                                            setStatus(value);
                                            setPage(1);
                                        }}
                                        options={[
                                            { value: "", label: "全部状态" },

                                            { value: "void", label: "已作废" },
                                        ]}
                                    />
                                    {batchId ? (
                                        <AdminBadge tone="accent" dot={false}>
                                            已筛选批次
                                            <button type="button" className="ml-1 underline underline-offset-2" onClick={() => setBatchId("")}>
                                                清除
                                            </button>

                                    <span className="ml-auto text-xs text-[var(--admin-muted)]">共 {total.toLocaleString("zh-CN")} 个兑换码</span>
                                </AdminToolbar>

                                />
                            </AdminPanel>

                    },

                                <AdminToolbar>
                                    <span className="text-xs text-[var(--admin-muted)]">批次用于追踪一次生成的一批兑换码，可按批次导出。</span>

                                <Table<Batch> rowKey="id" dataSource={batches} columns={batchColumns} pagination={false} scroll={{ x: 760 }} />
                            </AdminPanel>
                        ),
                    },
                ]}
            />

            <Modal
                title="生成兑换码"
                open={generateOpen}
                onCancel={() => setGenerateOpen(false)}
                onOk={() => void generate()}
                okText="生成"
                confirmLoading={generating}

type SettingsState = Record<string, Record<string, unknown>>;

export function SettingsTab() {
    const apiRequest = useAdminRequest();
    const { message } = AntApp.useApp();
    const [settings, setSettings] = useState<SettingsState>({});
    const [saved, setSaved] = useState<SettingsState>({});
    const [activeTab, setActiveTab] = useState("site");
    const [saving, setSaving] = useState("");
    const [loading, setLoading] = useState(true);
    const [error, setError] = useState("");

    const load = useCallback(async () => {
        setLoading(true);
        setError("");
        try {
            const data = await apiRequest<{ settings: SettingsState }>("/api/admin/settings");
            setSettings(data.settings);
            setSaved(data.settings);
        } catch (reason) {
            const text = reason instanceof Error ? reason.message : String(reason);
            setError(text);
            message.error(text);
        } finally {
            setLoading(false);
        }
    }, [message]);

    useEffect(() => {
        void load();
    }, [load]);

    function update(section: SectionKey, field: string, value: unknown) {
        setSettings((previous) => ({ ...previous, [section]: { ...(previous[section] ?? {}), [field]: value } }));
    }

    function isDirty(tab: SettingsTab) {
        return tab.sections.some((section) => JSON.stringify(settings[section] ?? {}) !== JSON.stringify(saved[section] ?? {}));
    }

    async function save(tab: SettingsTab) {
        setSaving(tab.key);
        try {
            for (const section of tab.sections) {
                await apiRequest(`/api/admin/settings/${section}`, { method: "PUT", json: { value: settings[section] ?? {} } });
            }
            setSaved((previous) => {
                const next = { ...previous };
                for (const section of tab.sections) next[section] = settings[section] ?? {};
                return next;
            });
            if (tab.sections.includes("site")) clearSiteInfoCache();
            message.success("设置已保存，下一次请求生效");
        } catch (reason) {
            message.error(reason instanceof Error ? reason.message : String(reason));
        } finally {
            setSaving("");
        }
    }

    if (loading) return <AdminLoading label="正在读取系统设置" />;
    if (error && !Object.keys(settings).length) {
        return (
            <AdminPanel title="设置暂时不可用" description={error} actions={<Button icon={<RefreshCw className="size-4" />} onClick={() => void load()}>重试</Button>}>
                <p className="text-sm text-[var(--admin-muted)]">检查 API 服务或管理员权限后重试。</p>
            </AdminPanel>
        );
    }

    return (
        <div>
            <div className="mb-5 flex flex-wrap items-center justify-between gap-3">
                <p className="text-xs leading-5 text-[var(--admin-muted)]">修改保存后立即写入数据库，支付与邮件参数会在下一次请求使用。</p>
                <AdminBadge tone="neutral" dot={false}>按页签分别保存</AdminBadge>
            </div>
            <Tabs
                activeKey={activeTab}
                onChange={setActiveTab}
                items={TABS.map((tab) => ({
                    key: tab.key,
                    label: (
                        <span className="inline-flex items-center gap-2">
                            <span className="hidden sm:inline-flex">
                                <tab.icon className="size-3.5" />
                            </span>
                            {tab.label}
                            {isDirty(tab) ? <span className="size-1.5 rounded-full bg-[var(--admin-warning)]" /> : null}
                        </span>
                    ),
                    children: (
                        <SettingsPane
                            tab={tab}
                            settings={settings}
                            update={update}
                            dirty={isDirty(tab)}
                            saving={saving === tab.key}
                            disabled={Boolean(saving)}
                            onSave={() => void save(tab)}
                        />
                    ),
                }))}
            />
        </div>
    );
}

function SettingsPane({
    tab,
    settings,
    update,
    dirty,
    saving,
    disabled,
    onSave,
}: {
    tab: SettingsTab;
    settings: SettingsState;
    update: (section: SectionKey, field: string, value: unknown) => void;
    dirty: boolean;
    saving: boolean;
    disabled: boolean;
    onSave: () => void;
}) {
    const canManage = useAdminPermissions().includes("settings.manage");
    return (
        <AdminPanel
            title={tab.label}
            description={tab.description}
            actions={dirty ? <AdminBadge tone="warning" dot={false}>有未保存修改</AdminBadge> : <AdminBadge tone="positive" dot={false}>已同步</AdminBadge>}
        >
            <form autoComplete="off" onSubmit={(event) => event.preventDefault()}>
                <fieldset disabled={!canManage}>
                {tab.key === "payments" ? (
                    <div className="grid gap-4 xl:grid-cols-2">
                        {tab.sections.map((section) => (
                            <PaymentSection key={section} section={SECTIONS[section]} settings={settings} update={update} />
                        ))}
                    </div>
                ) : (
                    <SectionFields section={SECTIONS[tab.sections[0]]} settings={settings} update={update} />
                )}
                <div className="mt-6 flex flex-wrap items-center justify-between gap-3 border-t border-[var(--admin-line)] pt-4">
                    <span className="text-[11px] text-[var(--admin-muted)]">保存后立即生效；密钥类字段不会回显明文。</span>
                    {canManage && <Button type="primary" loading={saving} disabled={!dirty || disabled} onClick={onSave}>
                        保存更改
                    </Button>}
                </div>
                </fieldset>
            </form>
        </AdminPanel>
    );
}

function PaymentSection({
    section,
    settings,
    update,
}: {
    section: Section;
    settings: SettingsState;
    update: (section: SectionKey, field: string, value: unknown) => void;
}) {
    const enabled = Boolean(settings[section.key]?.enabled);
    return (
        <section className="rounded-[12px] border border-[var(--admin-line)] bg-[var(--admin-panel-soft)] p-4">
            <div className="mb-4 flex items-center justify-between gap-3">
                <h3 className="text-sm font-semibold text-[var(--admin-ink)]">{section.title}</h3>
                <AdminBadge tone={enabled ? "positive" : "neutral"}>{enabled ? "已启用" : "未启用"}</AdminBadge>
            </div>
            <div className="grid gap-4">
                {section.fields.filter((field) => field.type !== "bool").map((field) => (
                    <SettingControl key={field.name} section={section} field={field} settings={settings} update={update} />
                ))}
            </div>
            <div className="mt-4 border-t border-[var(--admin-line)] pt-4">
                {section.fields.filter((field) => field.type === "bool").map((field) => (
                    <SettingControl key={field.name} section={section} field={field} settings={settings} update={update} />
                ))}
            </div>
        </section>
    );
}

function SectionFields({ section, settings, update }: { section: Section; settings: SettingsState; update: (section: SectionKey, field: string, value: unknown) => void }) {
    return (

            {section.fields.map((field) => (
                <SettingControl key={field.name} section={section} field={field} settings={settings} update={update} className={field.span === 2 ? "md:col-span-2" : undefined} />
            ))}
        </div>
    );
}

function SettingControl({
    section,
    field,
    settings,
    update,
    className,
}: {
    section: Section;
    field: SettingField;
    settings: SettingsState;
    update: (section: SectionKey, field: string, value: unknown) => void;
    className?: string;
}) {
    const value = settings[section.key]?.[field.name];
    const name = `sundas-${section.key}-${field.name}`;
    const apiRequest = useAdminRequest();
    const [promptSources, setPromptSources] = useState<{ id: string; name: string }[]>([]);
    const [models, setModels] = useState<{ id: string; displayName: string; capability: string; enabled: boolean; routes: { unit: string; protocol: string; textMode: string; enabled: boolean }[] }[]>([]);

    useEffect(() => {
        if (field.type !== "source") return;
        let alive = true;
        fetchPromptSources().then((list) => { if (alive) setPromptSources(list); }).catch(() => { if (alive) setPromptSources([]); });
        return () => { alive = false; };
    }, [field.type]);

    useEffect(() => {
        if (field.type !== "model") return;
        let alive = true;
        apiRequest<{ models: { id: string; displayName: string; capability: string; enabled: boolean; routes: { unit: string; protocol: string; textMode: string; enabled: boolean }[] }[] }>("/api/admin/models")
            .then((data) => { if (alive) setModels(data.models); })
            .catch(() => { if (alive) setModels([]); });
        return () => { alive = false; };
    }, [apiRequest, field.type]);

    if (field.type === "source") {
        return (
            <AdminField label={field.label} hint={field.hint} className={className}>
                <Select
                    className="w-full"
                    allowClear
                    value={String(value ?? "") || undefined}
                    placeholder={field.placeholder}
                    options={promptSources.map((source) => ({ label: `${source.name}（${source.id}）`, value: source.id }))}
                    onChange={(next) => update(section.key, field.name, next ?? "")}
                />
            </AdminField>
        );
    }

    if (field.type === "model") {
        const options = models.filter((model) => model.enabled && model.capability === field.modelCapability
            && model.routes.some((route) => route.enabled && route.unit === "request" && route.protocol === "openai"))
            .map((model) => ({ value: model.id, label: model.displayName }));
        return (
            <AdminField label={field.label} hint={field.hint} className={className}>
                <Select
                    className="w-full"
                    showSearch
                    allowClear
                    optionFilterProp="label"
                    value={String(value ?? "") || undefined}
                    placeholder={field.placeholder}
                    options={options}
                    notFoundContent="暂无已启用的 OpenAI 文本模型"
                    onChange={(next) => update(section.key, field.name, next ?? "")}
                />
            </AdminField>
        );
    }

    if (field.type === "bool") {
        return (
            <AdminField label={field.label} hint={field.hint} className={className}>
                <div className="flex h-[38px] items-center justify-between rounded-[9px] border border-[var(--admin-line)] bg-[var(--admin-panel)] px-3">
                    <span className="text-xs text-[var(--admin-muted)]">{value ? "已启用" : "未启用"}</span>
                    <Switch checked={Boolean(value)} onChange={(checked) => update(section.key, field.name, checked)} />
                </div>
            </AdminField>
        );
    }

    if (field.type === "payTypes") {
        return (
            <AdminField label={field.label} hint={field.hint} className={className}>
                <Checkbox.Group name={name} value={Array.isArray(value) ? value : []}
                    options={[{ label: "微信支付", value: "wxpay" }, { label: "支付宝", value: "alipay" }]}
                    onChange={(next) => update(section.key, field.name, next)} />
            </AdminField>
        );
    }

    if (field.type === "textarea") {
        return (
            <AdminField label={field.label} hint={field.hint} className={className}>
                <Input.TextArea {...ADMIN_AUTOFILL_OFF} name={name} rows={3} value={String(value ?? "")} placeholder={field.placeholder} onChange={(event) => update(section.key, field.name, event.target.value)} />
            </AdminField>
        );
    }

    if (field.type === "number") {
        return (
            <AdminField label={field.label} hint={field.hint} className={className}>
                <InputNumber {...ADMIN_AUTOFILL_OFF} name={name} className="w-full" value={Number(value ?? 0)} step={field.step} onChange={(next) => update(section.key, field.name, Number(next) || 0)} />
            </AdminField>
        );
    }

    if (field.type === "list") {
        return (
            <AdminField label={field.label} hint={field.hint} className={className}>
                <Input
                    {...ADMIN_AUTOFILL_OFF}
                    name={name}
                    value={Array.isArray(value) ? value.join(", ") : String(value ?? "")}
                    placeholder={field.placeholder}
                    onChange={(event) => update(section.key, field.name, event.target.value.split(",").map((item) => item.trim()).filter(Boolean))}
                />
            </AdminField>
        );
    }

    return (
        <AdminField label={field.label} hint={field.hint} className={className}>
            {field.type === "password" ? (
                <Input.Password {...ADMIN_AUTOFILL_OFF} autoComplete="new-password" name={name} value={String(value ?? "")} placeholder={field.placeholder} onChange={(event) => update(section.key, field.name, event.target.value)} />
            ) : (
                <Input {...ADMIN_AUTOFILL_OFF} name={name} value={String(value ?? "")} placeholder={field.placeholder} onChange={(event) => update(section.key, field.name, event.target.value)} />
            )}
        </AdminField>
    );
}

};

export function AdminBadge({ tone = "neutral", children, dot = true }: { tone?: AdminBadgeTone; children: ReactNode; dot?: boolean }) {

}

import { useEffect, useState } from "react";
import type { Key } from "react";
import { App as AntApp, Button, Input, InputNumber, Modal, Select, Table } from "antd";
import type { TableColumnsType } from "antd";
import dayjs from "dayjs";
import { RefreshCw, Search } from "lucide-react";
import { useSearchParams } from "react-router-dom";

import { useAdminRequest } from "../use-admin-request";
import { ADMIN_AUTOFILL_OFF, AdminBadge, AdminField, AdminPanel, AdminToolbar } from "./ui";
import { UserDetail } from "../users/user-detail";
import { useAdminPermissions } from "../permissions";

type AdminUser = {
    id: string;
    email: string;
    displayName: string;
    role: string;
    status: string;
    balance: number;
    held: number;
    member: boolean;
    createdAt: string;
    lastLoginAt: string | null;
};

export function UsersTab() {
    const permissions = useAdminPermissions();
    const [urlParams] = useSearchParams();
    const apiRequest = useAdminRequest();
    const { message, modal } = AntApp.useApp();
    const [search, setSearch] = useState(urlParams.get("search") ?? "");
    const [query, setQuery] = useState(urlParams.get("search") ?? "");
    const [statusFilter, setStatusFilter] = useState("");
    const [page, setPage] = useState(1);
    const [data, setData] = useState<{ items: AdminUser[]; total: number }>({ items: [], total: 0 });
    const [loading, setLoading] = useState(false);
    const [target, setTarget] = useState<AdminUser | null>(null);
    const [detailUserId, setDetailUserId] = useState<string | null>(null);
    const [amount, setAmount] = useState(0);
    const [remark, setRemark] = useState("");
    const [adjusting, setAdjusting] = useState(false);
    const [selected, setSelected] = useState<Key[]>([]);
    const [batchStatus, setBatchStatus] = useState<"active" | "banned" | null>(null);
    const [batchNote, setBatchNote] = useState("");
    const [batchResult, setBatchResult] = useState<{ id: string; success: boolean; error?: string }[] | null>(null);

    async function load() {
        setLoading(true);
        try {
            const params = new URLSearchParams({ page: String(page), pageSize: "20", search: query, status: statusFilter });
            const result = await apiRequest<{ items: AdminUser[]; total: number }>(`/api/admin/users?${params}`);
            setData(result);
        } catch (error) {
            message.error(error instanceof Error ? error.message : String(error));
        } finally {
            setLoading(false);
        }
    }

    useEffect(() => {
        void load();
    }, [page, query, statusFilter]);

    async function adjust() {
        if (!target) return;
        setAdjusting(true);
        try {
            await apiRequest(`/api/admin/users/${target.id}/credits`, { method: "POST", json: { amount, remark } });
            message.success("积分已调整");
            setTarget(null);
            setAmount(0);
            setRemark("");
            await load();
        } catch (error) {
            message.error(error instanceof Error ? error.message : String(error));
        } finally {
            setAdjusting(false);
        }
    }

    function toggleStatus(user: AdminUser) {
        modal.confirm({
            title: user.status === "active" ? "禁用该用户？" : "恢复该用户？",
            content: user.email,
            okText: user.status === "active" ? "禁用" : "恢复",
            okButtonProps: { danger: user.status === "active" },
            onOk: async () => {
                await apiRequest(`/api/admin/users/${user.id}`, { method: "PATCH", json: { status: user.status === "active" ? "banned" : "active" } });
                await load();
            },
        });
    }

    async function batchUpdate() {
        if (!batchStatus || !batchNote.trim()) { message.warning("请填写操作原因"); return; }
        setAdjusting(true);
        try {
            const result = await apiRequest<{ results: { id: string; success: boolean; error?: string }[] }>("/api/admin/users/batch-status", { method: "POST", json: { userIds: selected, status: batchStatus, note: batchNote } });
            setBatchStatus(null); setBatchNote(""); setSelected([]); setBatchResult(result.results); await load();
        } catch (error) { message.error(error instanceof Error ? error.message : String(error)); }
        finally { setAdjusting(false); }
    }

    const columns: TableColumnsType<AdminUser> = [
        {
            title: "用户",
            dataIndex: "email",
            render: (value: string, record) => (
                <div className="min-w-0">
                    <div className="flex items-center gap-2">
                        <span className="truncate font-medium text-[var(--admin-ink)]">{value}</span>
                        {record.role === "admin" ? <AdminBadge tone="accent">管理员</AdminBadge> : null}
                    </div>
                    <div className="mt-0.5 truncate text-xs text-[var(--admin-muted)]">{record.displayName || "未设置昵称"}</div>
                </div>
            ),
        },
        {
            title: "积分",
            dataIndex: "balance",
            render: (value: number, record) => (
                <div>
                    <div className="font-medium tabular-nums text-[var(--admin-ink)]">{value.toLocaleString("zh-CN")}</div>
                    {record.held ? <div className="mt-0.5 text-[11px] text-[var(--admin-muted)]">冻结 {record.held}</div> : null}
                </div>
            ),
        },
        { title: "会员", dataIndex: "member", render: (value: boolean) => <AdminBadge tone={value ? "positive" : "neutral"}>{value ? "会员" : "普通"}</AdminBadge> },
        { title: "状态", dataIndex: "status", render: (value: string) => <AdminBadge tone={value === "active" ? "positive" : "danger"}>{value === "active" ? "正常" : "禁用"}</AdminBadge> },
        { title: "注册时间", dataIndex: "createdAt", render: (value: string) => <span className="tabular-nums text-[var(--admin-muted)]">{dayjs(value).format("YYYY-MM-DD HH:mm")}</span> },
        {
            title: "操作",
            key: "actions",
            align: "right",
            render: (_value, record) => (
                <div className="flex justify-end gap-1.5">
                    <Button size="small" onClick={() => setDetailUserId(record.id)}>详情</Button>
                    {permissions.includes("credits.adjust") && <Button size="small" onClick={() => setTarget(record)}>
                        调积分
                    </Button>}
                    {permissions.includes("users.manage") && <Button size="small" danger={record.status === "active"} onClick={() => toggleStatus(record)}>
                        {record.status === "active" ? "禁用" : "恢复"}
                    </Button>}
                </div>
            ),
        },
    ];

    return (
        <div className="space-y-5">
            <UserDetail userId={detailUserId} onClose={() => setDetailUserId(null)} onChanged={() => void load()} />
            <AdminPanel title="用户列表" description="按邮箱或昵称搜索，调整积分或控制账户访问。" actions={<Button size="small" icon={<RefreshCw className="size-3.5" />} loading={loading} onClick={() => void load()}>刷新</Button>} flush>
                <AdminToolbar>
                    <div className="w-full sm:w-80">
                        <Input.Search
                            {...ADMIN_AUTOFILL_OFF}
                            name="sundas-admin-user-search"
                            placeholder="搜索邮箱或昵称"
                            value={search}
                            onChange={(event) => setSearch(event.target.value)}
                            onSearch={(value) => {
                                setPage(1);
                                setQuery(value.trim());
                            }}
                            className="w-full"
                            allowClear
                            prefix={<Search className="size-4 text-[var(--admin-muted)]" />}
                        />
                    </div>
                    <Select value={statusFilter} onChange={value => { setPage(1); setStatusFilter(value); }} options={[{ value: "", label: "全部状态" }, { value: "active", label: "正常" }, { value: "banned", label: "已禁用" }]} />
                    <span className="ml-auto text-xs text-[var(--admin-muted)]">共 {data.total.toLocaleString("zh-CN")} 位用户</span>
                    {permissions.includes("users.manage") && <><Button disabled={!selected.length} onClick={() => setBatchStatus("active")}>批量恢复</Button><Button danger disabled={!selected.length} onClick={() => setBatchStatus("banned")}>批量禁用</Button></>}
                </AdminToolbar>
                <Table<AdminUser>
                    rowKey="id"
                    loading={loading}
                    dataSource={data.items}
                    columns={columns}
                    rowSelection={permissions.includes("users.manage") ? { selectedRowKeys: selected, onChange: setSelected } : undefined}
                    scroll={{ x: 900 }}
                    pagination={{ current: page, pageSize: 20, total: data.total, onChange: setPage, showSizeChanger: false }}
                />
            </AdminPanel>

            <Modal title={`确认${batchStatus === "banned" ? "禁用" : "恢复"} ${selected.length} 位用户`} open={Boolean(batchStatus)} onCancel={() => { setBatchStatus(null); setBatchNote(""); }} onOk={() => void batchUpdate()} confirmLoading={adjusting}>
                <p>选中用户：{selected.join("、")}</p><AdminField label="操作原因"><Input value={batchNote} onChange={event => setBatchNote(event.target.value)} /></AdminField>
            </Modal>
            <Modal title="批量操作结果" open={Boolean(batchResult)} onCancel={() => setBatchResult(null)} footer={<Button onClick={() => setBatchResult(null)}>关闭</Button>}>
                {batchResult?.map(item => <p key={item.id}>{item.id}：{item.success ? "成功" : item.error}</p>)}
            </Modal>

            <Modal
                title="调整积分"
                open={Boolean(target)}
                onCancel={() => setTarget(null)}
                onOk={() => void adjust()}
                okText="确认调整"
                confirmLoading={adjusting}
                okButtonProps={{ disabled: amount === 0 }}
                width={460}
            >
                <div className="space-y-4 pt-1">
                    <div className="rounded-[10px] border border-[var(--admin-line)] bg-[var(--admin-panel-soft)] px-4 py-3">
                        <div className="truncate text-sm font-medium text-[var(--admin-ink)]">{target?.email}</div>
                        <div className="mt-1 text-xs text-[var(--admin-muted)]">
                            当前余额 <span className="font-semibold tabular-nums text-[var(--admin-ink)]">{target?.balance ?? 0}</span>
                            {target?.held ? ` · 冻结 ${target.held}` : ""}
                        </div>
                    </div>
                    <AdminField label="调整数量" hint="正数增加，负数扣减；扣减时最多扣到余额为 0。">
                        <InputNumber {...ADMIN_AUTOFILL_OFF} name="sundas-user-credit-amount" className="w-full" value={amount} onChange={(value) => setAmount(Number(value) || 0)} placeholder="例如 100 或 -50" />
                    </AdminField>
                    <AdminField label="备注">
                        <Input {...ADMIN_AUTOFILL_OFF} name="sundas-user-credit-remark" value={remark} onChange={(event) => setRemark(event.target.value)} placeholder="说明这次调整的原因" />
                    </AdminField>
                </div>
            </Modal>
        </div>
    );
}
