import { FileText, Group, Image as ImageIcon, Music2, Settings2, Video } from "lucide-react";

import i18n from "@/i18n";

import { NODE_SPECS } from "@/constant/canvas";
import { registerNodeDefinitions } from "@/lib/canvas/node-registry";
import { CanvasNodeType, type CanvasNodeData } from "@/types/canvas";
import type { CanvasNodeDefinition, CanvasNodeResource } from "@/types/canvas-plugin";

// Extensible metadata for built-in nodes, reusing NODE_SPECS for size and initial metadata.
// Rendering remains in canvas-node's internal renderer, so no Content component is provided.
function builtinResource(node: CanvasNodeData): CanvasNodeResource | null {
    if (node.type === CanvasNodeType.Image && node.metadata?.content) return { kind: "image", url: node.metadata.content };
    if (node.type === CanvasNodeType.Video && node.metadata?.content) return { kind: "video", url: node.metadata.content };
    if (node.type === CanvasNodeType.Audio && node.metadata?.content) return { kind: "audio", url: node.metadata.content };
    if (node.type === CanvasNodeType.Text && (node.metadata?.content || node.metadata?.prompt)) return { kind: "text", text: node.metadata.content || node.metadata.prompt };
    return null;
}

const iconClass = "size-4";

const BUILTIN_DEFINITIONS: CanvasNodeDefinition[] = [
    { type: CanvasNodeType.Text, title: i18n.t("assets.kinds.text"), icon: <FileText className={iconClass} />, minimapColor: undefined, resource: builtinResource },
    { type: CanvasNodeType.Image, title: i18n.t("assets.kinds.image"), icon: <ImageIcon className={iconClass} />, minimapColor: "#10b981", keepAspectRatio: (node: CanvasNodeData) => !node.metadata?.freeResize, resource: builtinResource },
    { type: CanvasNodeType.Video, title: i18n.t("assets.kinds.video"), icon: <Video className={iconClass} />, minimapColor: "#f97316", keepAspectRatio: () => true, resource: builtinResource },
    { type: CanvasNodeType.Audio, title: i18n.t("assets.kinds.audio"), icon: <Music2 className={iconClass} />, minimapColor: "#a855f7", resource: builtinResource },
    { type: CanvasNodeType.Config, title: i18n.t("canvas.configNode.title"), icon: <Settings2 className={iconClass} />, minimapColor: "#60a5fa", hasSourceHandle: false },
    { type: CanvasNodeType.Group, title: i18n.t("canvas.node.group"), icon: <Group className={iconClass} />, minimapColor: "#94a3b8" },
].map((def) => {
    const spec = NODE_SPECS[def.type];
    return { ...def, title: spec.title, defaultSize: { width: spec.width, height: spec.height }, defaultMetadata: spec.metadata };
});

let registered = false;
export function registerBuiltinNodes() {
    if (registered) return;
    registered = true;
    registerNodeDefinitions(BUILTIN_DEFINITIONS, "builtin");
}

        return () => observer.disconnect();
    }, [open, viewportElement]);

    const baseSize = fitImage(image, viewportSize);
    const stageSize = { width: baseSize.width * zoom, height: baseSize.height * zoom };
    const contentSize = {
        width: Math.max(viewportSize.width, stageSize.width),
        height: Math.max(viewportSize.height, stageSize.height),
    };
    const stageOffset = {
        left: Math.max(0, Math.round((contentSize.width - stageSize.width) / 2)),
        top: Math.max(0, Math.round((contentSize.height - stageSize.height) / 2)),
    };

    useLayoutEffect(() => {
        const viewport = viewportNodeRef.current;
        const anchor = zoomAnchorRef.current;
        if (!viewport || !anchor || Math.abs(anchor.zoom - zoom) > 0.001) return;
        const nextWidth = baseSize.width * zoom;
        const nextHeight = baseSize.height * zoom;
        const nextLeft = Math.max(0, (Math.max(viewport.clientWidth, nextWidth) - nextWidth) / 2);
        const nextTop = Math.max(0, (Math.max(viewport.clientHeight, nextHeight) - nextHeight) / 2);
        viewport.scrollLeft = nextLeft + anchor.ratioX * nextWidth - anchor.viewportX;
        viewport.scrollTop = nextTop + anchor.ratioY * nextHeight - anchor.viewportY;
        zoomAnchorRef.current = null;
    }, [baseSize.height, baseSize.width, zoom]);

    const setZoomAround = useCallback(
        (nextZoom: number, clientX?: number, clientY?: number) => {
            const viewport = viewportNodeRef.current;
            const stage = stageRef.current;
            if (!viewport || !stage || !baseSize.width || !baseSize.height) return;

            const boundedZoom = clamp(nextZoom, minZoom, maxZoom);
            if (Math.abs(boundedZoom - zoom) < 0.001) return;

            const viewportRect = viewport.getBoundingClientRect();
            const stageRect = stage.getBoundingClientRect();
            const pointerX = clientX ?? viewportRect.left + viewportRect.width / 2;
            const pointerY = clientY ?? viewportRect.top + viewportRect.height / 2;
            const ratioX = clamp((pointerX - stageRect.left) / Math.max(1, stageRect.width), 0, 1);
            const ratioY = clamp((pointerY - stageRect.top) / Math.max(1, stageRect.height), 0, 1);
            const viewportX = pointerX - viewportRect.left;
            const viewportY = pointerY - viewportRect.top;

            zoomAnchorRef.current = { zoom: boundedZoom, ratioX, ratioY, viewportX, viewportY };
            setZoom(boundedZoom);
        },
        [baseSize.height, baseSize.width, zoom],
    );

    useEffect(() => {
        if (!open || !viewportElement) return;
        const handleWheel = (event: WheelEvent) => {
            event.preventDefault();
            event.stopPropagation();
            setZoomAround(event.deltaY < 0 ? zoom * zoomStep : zoom / zoomStep, event.clientX, event.clientY);
        };
        viewportElement.addEventListener("wheel", handleWheel, { passive: false });
        return () => viewportElement.removeEventListener("wheel", handleWheel);
    }, [open, setZoomAround, viewportElement, zoom]);

    const startPan = useCallback((event: ReactPointerEvent<HTMLDivElement>) => {
        if (event.button !== 1 && !(event.button === 0 && spacePressedRef.current)) return;
        event.preventDefault();
        event.stopPropagation();
        const viewport = event.currentTarget;
        panRef.current = { pointerId: event.pointerId, x: event.clientX, y: event.clientY, scrollLeft: viewport.scrollLeft, scrollTop: viewport.scrollTop };
        viewport.setPointerCapture(event.pointerId);
        setIsPanning(true);
    }, []);
    const movePan = useCallback((event: ReactPointerEvent<HTMLDivElement>) => {
        const pan = panRef.current;
        if (!pan || event.pointerId !== pan.pointerId) return;
        event.preventDefault();
        event.stopPropagation();
        event.currentTarget.scrollLeft = pan.scrollLeft - (event.clientX - pan.x);
        event.currentTarget.scrollTop = pan.scrollTop - (event.clientY - pan.y);
    }, []);
    const stopPan = useCallback((event: ReactPointerEvent<HTMLDivElement>) => {
        const pan = panRef.current;
        if (!pan || event.pointerId !== pan.pointerId) return;
        event.preventDefault();
        event.stopPropagation();
        if (event.currentTarget.hasPointerCapture(event.pointerId)) event.currentTarget.releasePointerCapture(event.pointerId);
        panRef.current = null;
        setIsPanning(false);
    }, []);
    const preventAuxClick = useCallback((event: ReactMouseEvent<HTMLDivElement>) => {
        if (event.button !== 1) return;
        event.preventDefault();
        event.stopPropagation();
    }, []);

    return {
        viewportRef,
        stageRef,
        zoom,
        isPanning,
        spacePressed,
        scrollClassName: zoom > minZoom + 0.001 ? "overflow-scroll" : "overflow-hidden",
        panHandlers: {
            onPointerDownCapture: startPan,
            onPointerMoveCapture: movePan,
            onPointerUpCapture: stopPan,
            onPointerCancelCapture: stopPan,
            onAuxClick: preventAuxClick,
        },
        canZoomIn: zoom < maxZoom,
        canZoomOut: zoom > minZoom,
        imageScale: image ? stageSize.width / image.width : 0,
        zoomIn: () => setZoomAround(zoom * zoomStep),
        zoomOut: () => setZoomAround(zoom / zoomStep),
        resetZoom: () => setZoomAround(minZoom),
        contentStyle: { width: contentSize.width, height: contentSize.height } satisfies CSSProperties,
        stageStyle: {
            left: stageOffset.left,
            top: stageOffset.top,
            width: stageSize.width,
            height: stageSize.height,
        } satisfies CSSProperties,
        mediaStyle: {
            width: baseSize.width,
            height: baseSize.height,
            transform: `translateZ(0) scale(${zoom})`,
            transformOrigin: "top left",
        } satisfies CSSProperties,
    };
}

function fitImage(image: ImageSize | null, viewport: ImageSize): ImageSize {
    if (!image || !viewport.width || !viewport.height) return { width: 0, height: 0 };
    const availableWidth = Math.max(1, viewport.width - viewportPadding * 2);
    const availableHeight = Math.max(1, viewport.height - viewportPadding * 2);
    const scale = Math.min(availableWidth / image.width, availableHeight / image.height, 1);
    return { width: Math.max(1, Math.floor(image.width * scale)), height: Math.max(1, Math.floor(image.height * scale)) };
}

function clamp(value: number, min: number, max: number) {
    return Math.min(max, Math.max(min, value));
}

import type { ReferenceImage } from "@/types/image";
import type { ReferenceAudio, ReferenceVideo } from "@/types/media";

export type Continuation = { title: string; prompt: string };

type Draft = { continuation?: Continuation; prompt: string; references: ReferenceImage[]; videos: ReferenceVideo[]; audios: ReferenceAudio[] };
let session = "";
const drafts: Partial<Record<"image" | "video", Draft>> = {};

function forSession(owner: string, epoch: number) {
    const current = `${owner}:${epoch}`;
    if (session !== current) {
        session = current;
        delete drafts.image;
        delete drafts.video;
    }
}

export function readCreationDraft(mode: "image" | "video", owner: string, epoch: number) {
    forSession(owner, epoch);
    return drafts[mode];
}

export function saveCreationDraft(mode: "image" | "video", owner: string, epoch: number, draft: Draft) {
    forSession(owner, epoch);
    drafts[mode] = draft;
}

export function clearCreationDrafts() { delete drafts.image; delete drafts.video; }

/**
 * 图片/视频/对话各自是独立页面（/create?mode=…），切换会重新挂载整页。
 * 这里记住上一页的选中项与轨道横坐标，新页面挂载时先画在旧位置再滑到新位置，
 * 视觉上就是一个框里的滑块在移动，而不是整块控件跳一下。
 */
let lastSwitchMemory: { mode: string; left: number } | null = null;

/** 一整条描边轨道，选中项是轨道内的一块滑块；只暴露 value / onChange，交互与图标都在内部。 */
export function CreationModeTrack({ value, options, onChange, disabled = false, busy = false, handoff = false }: {
    value: string; options: readonly CreationModeOption[]; onChange: (value: string) => void;
    disabled?: boolean; busy?: boolean; handoff?: boolean;
}) {
    const trackRef = useRef<HTMLDivElement | null>(null);
    const [bar, setBar] = useState<{ left: number; width: number } | null>(null);
    // 跨页面接力动画还在等第一帧时，不要被 ResizeObserver 直接拉到终点
    const handoffPendingRef = useRef(false);

    useLayoutEffect(() => {
        const track = trackRef.current;
        if (!track) return;
        const current = track.querySelector<HTMLButtonElement>('[role="radio"][aria-checked="true"]');
        if (!current) return;
        const trackRect = track.getBoundingClientRect();
        const rect = current.getBoundingClientRect();
        const target = { left: rect.left - trackRect.left - track.clientLeft, width: rect.width };
        const previous = lastSwitchMemory;
        lastSwitchMemory = { mode: value, left: trackRect.left };
        const previousButton = handoff && previous && previous.mode !== value
            ? track.querySelector<HTMLButtonElement>(`[role="radio"][data-mode-value="${previous.mode}"]`)
            : null;
        // 只有轨道位置基本一致时才做跨页面接力，否则（例如对话页输入框更窄）直接从目标位置开始。
        if (previousButton && Math.abs(previous!.left - trackRect.left) < 24) {
            const from = previousButton.getBoundingClientRect();
            setBar({ left: from.left - trackRect.left - track.clientLeft, width: from.width });
            handoffPendingRef.current = true;
            const timer = window.setTimeout(() => {
                handoffPendingRef.current = false;
                setBar(target);
            }, 30);
            return () => {
                window.clearTimeout(timer);
                handoffPendingRef.current = false;
            };
        }
        setBar(target);
    }, [handoff, value]);

    // 容器尺寸变化时重新量一次，避免滑块和文字错位
    useEffect(() => {
        const track = trackRef.current;
        if (!track) return;
        const observer = new ResizeObserver(() => {
            if (handoffPendingRef.current) return;
            const current = track.querySelector<HTMLButtonElement>('[role="radio"][aria-checked="true"]');
            if (!current) return;
            const trackRect = track.getBoundingClientRect();
            const rect = current.getBoundingClientRect();
            setBar({ left: rect.left - trackRect.left - track.clientLeft, width: rect.width });
        });
        observer.observe(track);
        return () => observer.disconnect();
    }, []);

    function handleKeyDown(event: KeyboardEvent<HTMLDivElement>) {
        const index = options.findIndex(item => item.value === value);
        if (index < 0) return;
        let next = index;
        if (event.key === "ArrowLeft" || event.key === "ArrowUp") next = (index - 1 + options.length) % options.length;
        else if (event.key === "ArrowRight" || event.key === "ArrowDown") next = (index + 1) % options.length;
        else if (event.key === "Home") next = 0;
        else if (event.key === "End") next = options.length - 1;
        else return;
        event.preventDefault();
        if (next === index) return;
        trackRef.current?.querySelector<HTMLButtonElement>(`[data-mode-value="${options[next].value}"]`)?.focus();
        onChange(options[next].value);
    }

    return (
        <div
            ref={trackRef}
            className="creation-mode-switch"
            role="radiogroup"
            aria-label="生成类型"
            aria-busy={busy || disabled || undefined}
            data-busy={busy || undefined}
            data-count={options.length}
            onKeyDown={handleKeyDown}
        >
            <span className="creation-mode-switch-bar" aria-hidden="true" style={bar ? { left: bar.left, width: bar.width } : { opacity: 0 }} />
            {options.map(item => {
                const active = item.value === value;
                const Icon = item.icon;
                return (
                    <button
                        key={item.value}
                        type="button"
                        role="radio"
                        aria-checked={active}
                        data-mode-value={item.value}
                        tabIndex={active ? 0 : -1}
                        disabled={disabled}
                        title={disabled && !busy ? "处理中，请稍候" : undefined}
                        className="creation-mode-switch-item"
                        onClick={() => { if (!active) onChange(item.value); }}
                    >
                        {Icon ? <Icon className="size-3.5" aria-hidden="true" /> : null}
                        <span>{item.label}</span>
                    </button>
                );
            })}
        </div>
    );
}

export function CreationModeSwitch({ mode, busy = false, onChat }: { mode: CreationMode; busy?: boolean; onChat?: () => Promise<string | undefined> }) {
    const navigate = useNavigate();
    const { message } = App.useApp();
    const [switching, setSwitching] = useState(false);

    const switchMode = useCallback(async (value: string) => {
        if (value === mode || switching) return;
        setSwitching(true);
        try {
            const id = value === "chat" ? await onChat?.() : undefined;
            const store = useCreationChat.getState();
            const threadId = id || (store.owner === useAuthStore.getState().user?.id ? store.activeId : "");
            navigate(`/create?mode=${value}${threadId ? `&thread=${encodeURIComponent(threadId)}` : ""}`);
        } catch (error) { message.error(error instanceof Error ? error.message : String(error)); }
        finally { setSwitching(false); }
    }, [message, mode, navigate, onChat, switching]);

    return <CreationModeTrack value={mode} options={CREATION_MODES} onChange={value => void switchMode(value)} disabled={busy || switching} busy={busy} handoff />;
}

async function mergedHistory(owner: string, mode: "image" | "video", cloud: ReturnType<GenerationHistoryPager["snapshot"]>) {
    const store = mode === "image" ? imageLogStorage : videoLogStorage;
    const local = await store.list<RestoredLog>(owner).catch(() => [] as RestoredLog[]);
    const resolved = await Promise.all(local.map(async log => ({
        ...log,
        images: await Promise.all((log.images || []).map(async image => ({ ...image, dataUrl: await resolveImageUrl(image.storageKey, image.dataUrl) }))),
        video: log.video ? { ...log.video, url: await resolveMediaUrl(log.video.storageKey, log.video.url) } : undefined,
    })));
    return mergeGenerationHistory(resolved, cloud.logs, cloud.hiddenTaskIds);
}

export function useOtherCreationHistory(owner: string | undefined, epoch: number, mode: "image" | "video") {
    const [state, setState] = useState<{ owner?: string; epoch: number; logs: RestoredLog[]; more: boolean; error: string }>({ epoch, logs: [], more: false, error: "" });
    const [pager, setPager] = useState<GenerationHistoryPager>();
    const currentPager = useRef<GenerationHistoryPager | undefined>(undefined);
    useEffect(() => {
        if (!owner) return;
        const next = new GenerationHistoryPager(owner, mode);
        currentPager.current = next;
        setPager(next);
        setState({ owner, epoch, logs: [], more: false, error: "" });
        let active = true;
        let timer: ReturnType<typeof setTimeout> | undefined;
        const refresh = () => {
            let error = "";
            void next.refresh().catch(() => { error = "云端记录暂未加载，显示本地记录"; return next.snapshot(); }).then(async result => {
            const logs = await mergedHistory(owner, mode, result);
            if (active) {
                setState({ owner, epoch, logs, more: result.hasMore, error });
                clearTimeout(timer);
                if (result.logs.some(log => log.status === "pending")) timer = setTimeout(refresh, 5000);
            }
        }).catch(() => { if (active) setState(current => ({ ...current, owner, epoch, error: "另一种创作记录暂未加载" })); }); };
        refresh();
        window.addEventListener("focus", refresh);
        return () => { active = false; currentPager.current = undefined; clearTimeout(timer); window.removeEventListener("focus", refresh); };
    }, [owner, epoch, mode]);
    const current = state.owner === owner && state.epoch === epoch;
    const loadMore = async () => {
        if (!pager || !owner) return;
        try {
            const result = await pager.loadMore();
            const logs = await mergedHistory(owner, mode, result);
            if (currentPager.current !== pager) return;
            setState({ owner, epoch, logs, more: result.hasMore, error: "" });
        } catch (error) {
            if (currentPager.current === pager) setState(value => ({ ...value, error: "加载更早记录失败，请重试" }));
            throw error;
        }
    };
    return { logs: current ? state.logs : [], more: current && state.more, error: current ? state.error : "", loadMore };
}

        setPending((value) => value + 1);
        try { return await work(); }
        finally { setPending((value) => value - 1); }
    };
    return { run, isTransferring: pending > 0 };
}

import { useEffect, useState } from "react";

const PAGE_SIZE = 30;

// Render recent records first so long histories do not mount every media item.
export function useVisibleRecords<T extends { id: string }>(logs: T[], owner: string | undefined, recordId: string | null) {
    const [limit, setLimit] = useState(PAGE_SIZE);
    useEffect(() => { setLimit(PAGE_SIZE); }, [owner]);
    useEffect(() => {
        if (!recordId) return;
        const index = logs.findIndex((item) => item.id === recordId);
        if (index >= 0) setLimit((current) => Math.max(current, index + 1));
    }, [logs, recordId]);

    const showEarlier = () => {
        const anchor = logs[Math.min(limit, logs.length) - 1]?.id;
        const element = anchor ? document.getElementById(`creation-record-${anchor}`) : null;
        const top = element?.getBoundingClientRect().top;
        let scroller = element?.parentElement ?? null;
        while (scroller && !/(auto|scroll)/.test(getComputedStyle(scroller).overflowY)) scroller = scroller.parentElement;
        setLimit((current) => current + PAGE_SIZE);
        if (element && scroller && top != null) requestAnimationFrame(() => requestAnimationFrame(() => {
            if (element.isConnected) scroller.scrollTop += element.getBoundingClientRect().top - top;
        }));
    };
    return { records: logs.slice(0, limit).reverse(), hasEarlier: logs.length > limit, showEarlier };
}

import { useEffect } from "react";
import type { ReactNode } from "react";
import { App } from "antd";

import { useCloudSync } from "@/hooks/use-cloud-sync";
import { applyServerModels, type ServerModel } from "@/lib/server-models";
import { apiRequest } from "@/services/api/client";
import { useAuthStore } from "@/stores/use-auth-store";
import { subscribeSessionChanges } from "@/lib/session-broadcast";

export function ClientRootInit({ children }: { children: ReactNode }) {
    const { message } = App.useApp();
    useCloudSync();

    useEffect(() => {
        void useAuthStore.getState().hydrate().catch((error) => message.error(error instanceof Error ? error.message : "无法确认登录状态"));
        return subscribeSessionChanges(() => { void useAuthStore.getState().externalSessionChanged().catch((error) => message.error(error instanceof Error ? error.message : "无法确认登录状态")); });
    }, []);

    useEffect(() => {
        const refresh = () => { if (document.visibilityState === "visible") void useAuthStore.getState().refreshWallet().catch(() => undefined); };
        document.addEventListener("visibilitychange", refresh);
        window.addEventListener("online", refresh);
        return () => { document.removeEventListener("visibilitychange", refresh); window.removeEventListener("online", refresh); };
    }, []);

    useEffect(() => {
        let cancelled = false;
        apiRequest<{ models: ServerModel[] }>("/api/models")
            .then((data) => {
                if (!cancelled && data.models.length) applyServerModels(data.models);
            })
            .catch(() => {
                // 模型列表拉取失败时保留本地缓存，登录后还会再拉一次。
            });
        return () => {
            cancelled = true;
        };
    }, []);

    return <>{children}</>;
}

import { Suspense, lazy, useEffect, useState } from "react";

import { useAuthDialogStore } from "@/stores/use-auth-dialog-store";
import { useConfigStore } from "@/stores/use-config-store";

/**
 * 全局浮层（登录弹窗 / 偏好设置弹窗）按需加载：
 * 这两块只在用户主动触发时才出现，但会静态带进 antd 的 Form / Modal / Select / Radio / Switch，
 * 是首屏包最大的一块业务开销。默认不挂载 → chunk 不下载；
 * 首次打开后保持挂载，关闭动画与内部状态与原来一致。
 */
const AuthDialog = lazy(() => import("@/components/auth/auth-dialog").then((module) => ({ default: module.AuthDialog })));
const AppConfigModal = lazy(() => import("@/components/layout/app-config-modal").then((module) => ({ default: module.AppConfigModal })));

function useKeepMountedWhen(open: boolean) {
    const [mounted, setMounted] = useState(open);
    useEffect(() => {
        if (open) setMounted(true);
    }, [open]);
    return mounted;
}

export function AuthDialogHost() {
    const open = useAuthDialogStore((state) => state.mode !== null);
    const mounted = useKeepMountedWhen(open);
    if (!mounted) return null;
    return (
        <Suspense fallback={null}>
            <AuthDialog />
        </Suspense>
    );
}

export function AppConfigModalHost() {
    const open = useConfigStore((state) => state.isConfigOpen);
    const mounted = useKeepMountedWhen(open);
    if (!mounted) return null;
    return (
        <Suspense fallback={null}>
            <AppConfigModal />
        </Suspense>
    );
}
