    function addCreativeMedia(runId: string, media: CharacterMedia[], sourceId?: string) {
        if (!isLoadedCanvasCurrent()) return;
        const source = sourceId ? nodesRef.current.find(node => node.id === sourceId) : undefined;
        const added: CanvasNodeData[] = [];
        for (const result of media) {
            if (!result.key || nodesRef.current.some(node => node.metadata?.creativeRunId === runId && node.metadata?.creativeSourceKey === result.key)) continue;
            const type = result.kind === "video" ? CanvasNodeType.Video : CanvasNodeType.Image;
            const size = fitNodeSize(result.width || 512, result.height || 512);
            added.push({ id: nanoid(), type, title: `${source?.title || "创作"} · 处理结果`, width: size.width, height: size.height,
                position: { x: (source?.position.x || 0) + (source?.width || 0) + 80, y: (source?.position.y || 0) + added.length * (size.height + 32) },
                metadata: { content: result.url || `/media/${result.key.split("/").map(encodeURIComponent).join("/")}`, status: "success", naturalWidth: result.width, naturalHeight: result.height, bytes: result.bytes, mimeType: result.mime, durationMs: result.duration === undefined ? undefined : result.duration * 1000, creativeRunId: runId, creativeSourceKey: result.key } });
        }
        if (!added.length) { message.info("结果已在当前画布中"); return; }
        nodesRef.current = [...nodesRef.current, ...added]; setNodes(nodesRef.current);
        if (source) { connectionsRef.current = [...connectionsRef.current, ...added.map(node => ({ id: nanoid(), fromNodeId: source.id, toNodeId: node.id }))]; setConnections(connectionsRef.current); }
        setSelectedNodeIds(new Set(added.map(node => node.id)));
        message.success(`已添加 ${added.length} 个处理结果到画布`);
    }

    async function upscaleProductionCandidate(runId: string, media: ProductionMedia) {
        if (!isLoadedCanvasCurrent() || !['image', 'video'].includes(media.kind)) return;

        (payload: StoryboardApplyPayload) => {

            const imageSpec = getNodeSpec(CanvasNodeType.Image);
            const videoSpec = getNodeSpec(CanvasNodeType.Video);
            const columns = payload.shots.length > 4 ? 3 : 2;
            const rows = Math.ceil(payload.shots.length / columns);
            const gapX = 56;
            const gapY = 168;
            const scriptGap = 120;
            // 整块（脚本 + 分镜网格）居中落在当前视口中心，避免镜头铺到视野外。
            const gridWidth = columns * imageSpec.width + (columns - 1) * gapX;
            const blockWidth = scriptNode.width + scriptGap + gridWidth;
            const blockHeight = Math.max(scriptNode.height, rows * imageSpec.height + (rows - 1) * gapY);
            const startX = center.x - blockWidth / 2 + scriptNode.width + scriptGap;
            const startY = center.y - blockHeight / 2;
            scriptNode.position = { x: center.x - blockWidth / 2, y: startY };

            const created: CanvasNodeData[] = [scriptNode];
            const nextConnections: CanvasConnection[] = [];

            payload.shots.forEach((shot, index) => {
                const column = index % columns;
                const row = Math.floor(index / columns);
                const shotTitle = uniqueNodeTitle([...nodesRef.current, ...created], t("canvas.storyboard.shotTitle", { index: shot.index, duration: shot.duration }));

                const imageNode = createCanvasNode(CanvasNodeType.Image, center, { prompt: shot.prompt, status: "idle", workflowKind: "shot", shotIndex: shot.index });
                imageNode.title = shotTitle;
                imageNode.position = { x: startX + column * (imageSpec.width + gapX), y: startY + row * (imageSpec.height + gapY) };
                created.push(imageNode);
                nextConnections.push({ id: nanoid(), fromNodeId: scriptNode.id, toNodeId: imageNode.id });

                if (!payload.createVideoNodes) return;
                const videoNode = createCanvasNode(CanvasNodeType.Video, center, { prompt: shot.videoPrompt, status: "idle", workflowKind: "shot_video", shotIndex: shot.index });
                videoNode.title = `${shotTitle}${t("canvas.storyboard.videoSuffix")}`;
                videoNode.position = {
                    x: imageNode.position.x + (imageSpec.width - videoSpec.width) / 2,
                    y: imageNode.position.y + imageSpec.height + 44,
                };
                created.push(videoNode);
                nextConnections.push({ id: nanoid(), fromNodeId: imageNode.id, toNodeId: videoNode.id });

            });

            message.success(t("canvas.storyboard.applied", { count: payload.shots.length }));

type Saved = { threads: CreationThread[]; activeId: string };
const cache = lazyStore("creation_conversations");
export const useCreationChat = create<Saved & { owner: string; ready: boolean; error: string }>(() => ({ owner: "", ready: false, error: "", threads: [], activeId: "" }));

let writes = Promise.resolve();
function persist() {
    const { owner, threads, activeId } = useCreationChat.getState();
    if (!owner) return;
    const snapshot = structuredClone({ threads, activeId });
    writes = writes.then(() => cache.setItem(owner, snapshot)).then(() => undefined).catch(() => {
        if (useCreationChat.getState().owner === owner) useCreationChat.setState({ error: "浏览器保存失败，请检查可用存储空间；当前内容仍保留在页面中。" });

}
export async function loadCreationChats(owner: string) {
    const state = useCreationChat.getState();
    if (state.owner === owner) { if (!state.ready) await loading; return; }
    useCreationChat.setState({ owner, ready: false, error: "", threads: [], activeId: "" });
    loading = (async () => {
        try {
            await writes;
            const saved = await cache.getItem<Saved>(owner);
            if (useCreationChat.getState().owner !== owner) return;
            const threads = (saved?.threads ?? []).map(thread => ({ ...thread, messages: thread.messages.map(item => item.role === "assistant" && !item.complete && !item.error ? { ...item, error: "上次回复已中断，可继续提问。" } : item) }));
            useCreationChat.setState({ threads, activeId: saved?.activeId ?? "", ready: true });
        } catch { if (useCreationChat.getState().owner === owner) useCreationChat.setState({ ready: true, error: "浏览器聊天记录读取失败，请刷新重试。" }); }
    })();
    await loading;
}
export function newCreationThread(owner: string, clearContext = true) {
    if (useCreationChat.getState().owner !== owner || !useCreationChat.getState().ready) return "";
    if (clearContext) clearCreationDrafts();
    const id = crypto.randomUUID(), now = Date.now();
    useCreationChat.setState(state => ({ activeId: id, threads: [...state.threads, { id, contextCleared: clearContext, title: "新对话", createdAt: now, updatedAt: now, messages: [], draft: "", records: [] }] }));

}

    if (useCreationChat.getState().owner !== owner) return;

export function saveCreationRecord(owner: string, record: CreationContextRecord, threadId?: string) {
    const state = useCreationChat.getState();
    if (state.owner !== owner) return;
    const thread = state.threads.find(t => t.records.some(r => r.id === record.id)) ?? state.threads.find(t => t.id === threadId);
    if (!thread) return;
    const imageRef = (image: NonNullable<CreationContextRecord["images"]>[number]) => image.storageKey ? { ...image, dataUrl: "", url: "" } : image;
    const storedRecord = { ...record, images: record.images?.map(imageRef), references: record.references?.map(imageRef) };
    const records = [...thread.records.filter(r => r.id !== record.id), storedRecord].sort((a,b) => a.createdAt-b.createdAt);
    patchCreationThread(owner, thread.id, { records, contextCleared: false, ...(thread.title === "新对话" ? { title: record.prompt.slice(0, 40) || "创作对话" } : {}) });
}

export async function handoffCreationToChat(owner: string, record?: CreationContextRecord) {
    await loadCreationChats(owner);
    const state = useCreationChat.getState();
    if (state.owner !== owner) throw new Error("账户已变化");
    const existing = record && state.threads.find(t => t.records.some(r => r.id === record.id));
    const id = existing?.id ?? await ensureCreationThread(owner);
    if (record) saveCreationRecord(owner, record, id);
    openCreationThread(owner, id);
    return id;

const FILM_CLIPS = [

    { src: "/hero-media/hero-ocean.mp4", poster: "/hero-media/hero-ocean.jpg" },
] as const;

const COLUMNS = 6;
const ROWS = 4;

export function HeroFilmWall() {
    const frames = useRef<HTMLDivElement>(null);
    // 尊重系统「减少动态效果」：仅显示海报，不自动播放
    const [reducedMotion, setReducedMotion] = useState(false);
    const [motionReady, setMotionReady] = useState(false);

    // The first paint needs only posters. Limit animated panels after idle instead
    // of starting every decoder while the application is still becoming interactive.
    useEffect(() => {
        const connection = (navigator as Navigator & { connection?: { saveData?: boolean; effectiveType?: string } }).connection;
        const mobile = window.matchMedia("(max-width: 767px)");
        let timer: number | undefined;
        const sync = () => {
            window.clearTimeout(timer);
            if (reducedMotion || connection?.saveData || /(^|-)2g$/.test(connection?.effectiveType || "") || mobile.matches) {
                setMotionReady(false);
                return;
            }
            timer = window.setTimeout(() => setMotionReady(true), 1200);
        };
        sync();
        mobile.addEventListener("change", sync);
        return () => {
            window.clearTimeout(timer);
            mobile.removeEventListener("change", sync);
        };
    }, [reducedMotion]);

    useEffect(() => {
        const query = window.matchMedia("(prefers-reduced-motion: reduce)");
        const sync = () => setReducedMotion(query.matches);
        sync();
        query.addEventListener("change", sync);
        return () => query.removeEventListener("change", sync);
    }, []);

    // 滚出视口即暂停，避免 24 路解码在后台持续占用资源
    useEffect(() => {
        const node = frames.current;
        if (!node || typeof IntersectionObserver === "undefined") return;
        const observer = new IntersectionObserver(
            ([entry]) => {
                for (const video of Array.from(node.querySelectorAll("video"))) {
                    if (entry.isIntersecting && !reducedMotion && video.hasAttribute("src")) void video.play().catch(() => undefined);
                    else video.pause();
                }
            },
            { rootMargin: "160px" },
        );
        observer.observe(node);
        return () => observer.disconnect();
    }, [reducedMotion, motionReady]);

    return (
        <div className="home-film-wall" ref={frames} aria-hidden="true">
            {Array.from({ length: COLUMNS }, (_, column) => (
                <div key={column} className="home-film-column">
                    {Array.from({ length: ROWS }, (_, row) => {
                        const index = column * ROWS + row;
                        const clip = FILM_CLIPS[(column + row * 2) % FILM_CLIPS.length];
                        // Keep one live panel in each visible column, rather than placing the
                        // entire playback budget in the cropped left-hand column.
                        const animated = motionReady && row === 0;
                        return (
                            <span key={row} className={`home-film-slot home-film-tone-${index % 16}`}>
                                <video
                                    src={animated ? clip.src : undefined}
                                    poster={clip.poster}
                                    muted
                                    loop
                                    playsInline
                                    autoPlay={animated && !reducedMotion}
                                    preload="none"
                                    disablePictureInPicture
                                    tabIndex={-1}
                                />
                            </span>
                        );
                    })}
                </div>
            ))}
        </div>
    );
}
