import { Euler, Matrix4, Vector3 } from "three";

import { interpolateDirectorTransform, resolveDirectorKeyframeProgress } from "@/lib/canvas/director/director-scene";
import { directorStageDirection, directorStageMatrix, directorStagePoint, directorStageTransform } from "@/lib/canvas/director/director-stage-transform";
import type { DirectorCamera, DirectorKeyframe, DirectorScene, DirectorShot, DirectorTransform, DirectorVec3 } from "@/types/director";

/**
 * 视口取景模式（3D / CAM / 五个正交轴向：俯视、正视、背视、左视、右视）。
 *
 * 与一级模式（director-modes.ts 的摆场/姿态/动画/摄影机）是两个正交的量：
 * 一级模式决定「能改什么」，取景模式只决定「从哪只眼睛看」。
 *
 * viewMode 是纯视口状态，绝不写入 DirectorScene，也绝不产生 undo/history 记录 ——
 * 换一只眼睛看场景不是对内容的修改。本模块只做纯函数，不持有任何 three 对象。
 *
 * 五个正交轴向和 CAM 一样锁环绕，但取景来自场景内容的包围盒而不是某台摄影机：
 * resolveDirectorOrthographicFraming 解出带 margin 的投影跨度，
 * resolveDirectorOrthographicFrustum 再按视口宽高比收成安全的正交半范围。
 */

export type DirectorViewMode = "free" | "camera" | "top" | "front" | "back" | "left" | "right";

export type DirectorViewModeCapabilities = {
    /** 是否允许 OrbitControls 自由环绕。只有 free 允许；CAM 与五个正交轴向都要锁定构图，拖拽会破坏取景。 */
    orbit: boolean;
    /** 是否由 shot 的摄影机接管取景（视口相机 == 当前镜头）。只有 CAM。 */
    framed: boolean;
    /** 投影方式。free/camera 是透视；五个正交轴向必须是正交投影，否则物体比例会随远近畸变。 */
    projection: "perspective" | "orthographic";
};

const CAPABILITIES: Record<DirectorViewMode, DirectorViewModeCapabilities> = {
    // 3D：编辑用的自由视角，可环绕、可缩放，摄影机只被关键帧驱动时才回写。
    free: { orbit: true, framed: false, projection: "perspective" },
    // CAM：所见即成片取景，视口 == 当前 shot 的摄影机，禁止环绕。
    camera: { orbit: false, framed: true, projection: "perspective" },
    // 五个正交轴向：取景来自场景包围盒而非某台摄影机，同样锁环绕，投影必须正交。
    top: { orbit: false, framed: false, projection: "orthographic" },
    front: { orbit: false, framed: false, projection: "orthographic" },
    back: { orbit: false, framed: false, projection: "orthographic" },
    left: { orbit: false, framed: false, projection: "orthographic" },
    right: { orbit: false, framed: false, projection: "orthographic" },
};

    { mode: "back", label: "BACK", hint: "背视：沿 +Z 方向看向场景，正交投影，环绕已锁定" },

export const DIRECTOR_DEFAULT_VIEW_MODE: DirectorViewMode = "free";

/** 未知模式回落默认能力，绝不返回 undefined —— 调用方直接读 .orbit / .framed。 */
export function directorViewModeCapabilities(mode: DirectorViewMode): DirectorViewModeCapabilities {
    return CAPABILITIES[mode] ?? CAPABILITIES[DIRECTOR_DEFAULT_VIEW_MODE];
}

/**
 * 当前 shot / 摄影机的唯一解析入口。
 *
 * activeShotId 指向不存在的 shot 时回落到第一个 shot；shot 的 cameraId 指向不存在的
 * 摄影机时回落到第一台。空场景返回 null 而不是抛异常 —— 视口在任何场景下都必须能渲染。
 */
export function resolveDirectorActiveShot(scene: DirectorScene): DirectorShot | null {
    return scene.shots.find((item) => item.id === scene.activeShotId) || scene.shots[0] || null;
}

export function resolveDirectorActiveCamera(scene: DirectorScene): DirectorCamera | null {
    const shot = resolveDirectorActiveShot(scene);
    return scene.cameras.find((item) => item.id === shot?.cameraId) || scene.cameras[0] || null;
}

/** 取景数据：CAM 模式下视口相机需要被写入的完整状态。 */
export type DirectorViewFraming = {
    cameraId: string;
    position: DirectorVec3;
    target: DirectorVec3;
    /** 世界系 up。保留 roll，且保证与视线不共线。 */
    up: DirectorVec3;
    fov: number;
    near: number;
    far: number;
};

// 叉积长度低于此值即视为共线：lookAt 会解出非法基向量（NaN 矩阵），视口直接黑屏。
const DIRECTOR_VIEW_COLLINEAR_EPSILON = 1e-6;

// 退化时的备选 up：世界 up 优先，其次两个水平轴 —— 正俯视/正仰视镜头落在这里。
const DIRECTOR_VIEW_UP_FALLBACKS: DirectorVec3[] = [
    [0, 1, 0],
    [0, 0, -1],
    [1, 0, 0],
];

/**
 * 解出带 roll 的 up 向量。
 *
 * 摄影机 rotation 的 z 分量就是荷兰角。裸 lookAt 会丢掉它 —— three 用固定的 camera.up
 * 重算基向量，画面永远水平。这里把世界 up 按摄影机自身欧拉角旋转，roll 才留得住。
 *
 * 再处理退化：正俯视/正仰视时旋转后的 up 与视线共线，叉积为 0，lookAt 解出 NaN 矩阵。
 * 此时按 世界up -> -Z -> +X 取第一个不共线的备选，绝不返回让相机失效的值。
 */
export function resolveDirectorViewUp(rotation: DirectorVec3, viewDirection: DirectorVec3): DirectorVec3 {
    const view = new Vector3(...viewDirection);
    if (view.lengthSq() <= DIRECTOR_VIEW_COLLINEAR_EPSILON) return [0, 1, 0];
    const rolled = new Vector3(0, 1, 0).applyEuler(new Euler(...rotation));
    const candidates = rolled.lengthSq() > DIRECTOR_VIEW_COLLINEAR_EPSILON ? [rolled.toArray() as DirectorVec3, ...DIRECTOR_VIEW_UP_FALLBACKS] : DIRECTOR_VIEW_UP_FALLBACKS;
    return candidates.find((candidate) => new Vector3(...candidate).cross(view).length() > DIRECTOR_VIEW_COLLINEAR_EPSILON) ?? [0, 1, 0];
}

/**
 * Three 透视投影是否可写进 PerspectiveCamera。
 *
 * 只检查有限还不够：fov=0 / 180、near<=0、far<=near 都能通过 isFinite，
 * 但 makePerspective 会得到 Inf/NaN 矩阵，视口黑屏。fov 必须落在 (0, 180)。
 */
export function directorUsablePerspectiveProjection(input: { fov: number; near: number; far: number }): boolean {
    const { fov, near, far } = input;
    if (![fov, near, far].every(Number.isFinite)) return false;
    if (fov <= 0 || fov >= 180) return false;
    if (near <= 0) return false;
    return far > near;
}

export type DirectorEffectiveViewport = {
    /** 视口应激活的专用相机。CAM 取景失败时必须是 free，不能停在从未写入的 camCamera。 */
    camera: "free" | "camera" | "orthographic";
    /** 是否允许环绕 free 相机。与 camera 同步：只有 free 为 true。 */
    orbit: boolean;
};

/**
 * 视口活动相机与交互的唯一决策。
 *
 * CAM 取景为 null（空场景、非法光学参数）时回落到已有的 free 相机和自由环绕，
 * 绝不改 DirectorScene。取景重新合法后确定性回到 CAM（orbit 锁死）。
 * 正交轴向不走这条回落：它们始终有安全默认包围盒。
 */
export function resolveDirectorEffectiveViewport(input: { mode: DirectorViewMode; framing: DirectorViewFraming | null }): DirectorEffectiveViewport {
    if (directorViewModeCapabilities(input.mode).projection === "orthographic") return { camera: "orthographic", orbit: false };
    if (input.mode === "camera" && input.framing) return { camera: "camera", orbit: false };
    return { camera: "free", orbit: true };
}

/**
 * 解出 CAM 模式下视口相机应有的取景。
 *
 * 返回 null 的情形都必须走 free 回落：3D 模式、空场景、位置/旋转非法、
 * 透视投影不可用（fov/near/far）。绝不把非法值写进 three 相机。
 *
 * position 取 playhead 上插值后的值，所以 CAM 模式下拖时间轴就是在预览成片运镜。
 */
export function resolveDirectorViewFraming(input: { scene: DirectorScene; mode: DirectorViewMode; playhead: number }): DirectorViewFraming | null {
    if (!directorViewModeCapabilities(input.mode).framed) return null;
    const camera = resolveDirectorActiveCamera(input.scene);
    if (!camera) return null;
    const local = resolveDirectorCameraLocalFraming(input.scene, camera, input.playhead);
    if (!local) return null;
    const stage = directorStageTransform(input.scene);
    return { ...local, position: directorStagePoint(stage, local.position), target: directorStagePoint(stage, local.target), up: directorStageDirection(stage, local.up), near: local.near * stage.scale, far: local.far * stage.scale };
}

/** CAM 画面与自由视角机位辅助图形共用的场景局部取景；也支持非活动机位。 */
export function resolveDirectorCameraLocalFraming(scene: DirectorScene, camera: DirectorCamera, playhead: number): DirectorViewFraming | null {
    const time = Number.isFinite(playhead) ? playhead : 0;
    const transform: DirectorTransform = resolveDirectorCameraTransform(camera, time);
    if (![...transform.position, ...transform.rotation].every(Number.isFinite)) return null;
    const optical = resolveDirectorCameraTrackValues(camera, time);
    const fov = optical.fov;
    if (!directorUsablePerspectiveProjection({ fov, near: camera.near, far: camera.far })) return null;
    const followed = scene.objects.find((item) => item.id === camera.followObjectId);
    const followedPosition = followed ? interpolateDirectorTransform(followed.transform, followed.keyframes, time).position : null;
    const followAnchor = camera.followAnchor;
    const followDelta = followedPosition && followAnchor && [...followedPosition, ...followAnchor].every(Number.isFinite)
        ? followedPosition.map((value, index) => value - followAnchor[index]) as DirectorVec3 : null;
    const position: DirectorVec3 = followDelta ? transform.position.map((value, index) => value + followDelta[index]) as DirectorVec3 : transform.position;
    let requestedTarget = optical.target;
    if (camera.lookAtMode === "rotation") {
        const forward = new Vector3(0, 0, -1).applyEuler(new Euler(...transform.rotation));
        requestedTarget = [position[0] + forward.x, position[1] + forward.y, position[2] + forward.z];
    } else if (camera.lookAtMode !== "coordinates" && camera.lookAtObjectId) {
        const object = scene.objects.find((item) => item.id === camera.lookAtObjectId);
        if (object) {
            const objectPosition = interpolateDirectorTransform(object.transform, object.keyframes, time).position;
            if (objectPosition.every(Number.isFinite)) requestedTarget = [objectPosition[0], objectPosition[1] + (object.kind === "actor" || object.primitive === "character" ? 1.2 : 0), objectPosition[2]];
        }
    }
    if (![...position, ...requestedTarget].every(Number.isFinite)) return null;
    const raw = new Vector3(requestedTarget[0] - position[0], requestedTarget[1] - position[1], requestedTarget[2] - position[2]);
    const degenerate = raw.lengthSq() <= DIRECTOR_VIEW_COLLINEAR_EPSILON;
    // 位置与焦点重合时视线为零向量，lookAt 无解：沿摄影机自身 -Z 造一个 1m 外的焦点。
    const view = degenerate ? new Vector3(0, 0, -1).applyEuler(new Euler(...transform.rotation)) : raw;
    const target: DirectorVec3 = degenerate ? [position[0] + view.x, position[1] + view.y, position[2] + view.z] : requestedTarget;
    return { cameraId: camera.id, position, target, up: resolveDirectorViewUp(transform.rotation, view.toArray() as DirectorVec3), fov, near: camera.near, far: camera.far };
}

export function resolveDirectorCameraTrackValues(camera: DirectorCamera, time: number): { target: DirectorVec3; fov: number } {
    return {
        target: interpolateDirectorCameraTrack(camera.keyframes, time, "target", camera.target),
        fov: interpolateDirectorCameraTrack(camera.keyframes, time, "fov", camera.fov),
    };
}

export function resolveDirectorCameraTransform(camera: DirectorCamera, time: number): DirectorTransform {
    const position = interpolateDirectorTransform(camera.transform, camera.keyframes.filter((key) => key.positionKeyed !== false), time);
    const rotation = interpolateDirectorTransform(camera.transform, camera.keyframes.filter((key) => key.rotationKeyed !== false), time);
    return { ...position, rotation: rotation.rotation };
}

/** Convert the visible camera aim into a rotation that preserves the frame when changing look-at modes. */
export function resolveDirectorCameraAimRotation(scene: DirectorScene, camera: DirectorCamera, time: number): DirectorVec3 | null {
    const framing = resolveDirectorCameraLocalFraming(scene, camera, time);
    if (!framing) return null;
    const euler = new Euler().setFromRotationMatrix(new Matrix4().lookAt(
        new Vector3(...framing.position), new Vector3(...framing.target), new Vector3(...framing.up),
    ));
    return [euler.x, euler.y, euler.z];
}

/** LibTV's inspector shows the effective aim even while coordinates/object tracking owns the camera view. */
export function resolveDirectorCameraInspectorRotation(scene: DirectorScene, camera: DirectorCamera, time: number): DirectorVec3 {
    const rotation = resolveDirectorCameraAimRotation(scene, camera, time) ?? resolveDirectorCameraTransform(camera, time).rotation;
    const degrees = 180 / Math.PI;
    return [-rotation[0] * degrees, ((rotation[1] * degrees + 180) % 360 + 360) % 360, rotation[2] * degrees];
}

function interpolateDirectorCameraTrack<K extends "target" | "fov">(keyframes: DirectorKeyframe[], time: number, track: K, fallback: NonNullable<DirectorKeyframe[K]>): NonNullable<DirectorKeyframe[K]> {
    const keys = keyframes.filter((key) => key[track] !== undefined).sort((a, b) => a.time - b.time);
    if (!keys.length) return fallback;
    const previous = [...keys].reverse().find((key) => key.time <= time) || keys[0];
    const next = keys.find((key) => key.time >= time) || keys.at(-1)!;
    if (previous === next || next.time <= previous.time) return previous[track]! as NonNullable<DirectorKeyframe[K]>;
    const progress = resolveDirectorKeyframeProgress((time - previous.time) / (next.time - previous.time), previous.easing);
    if (track === "fov") return (Number(previous.fov) + (Number(next.fov) - Number(previous.fov)) * progress) as NonNullable<DirectorKeyframe[K]>;
    return previous.target!.map((value, axis) => value + (next.target![axis] - value) * progress) as NonNullable<DirectorKeyframe[K]>;
}

/**
 * 取景同步键。
 *
 * 相机回写的 effect 只依赖这个字符串：草稿对象每次 render 都是新身份，但取景数值没变
 * 就不该重写相机。数值截到 4 位小数，避免浮点噪声引起无意义的重同步。
 */
export function directorViewFramingKey(framing: DirectorViewFraming | null): string {
    if (!framing) return "";
    return [...framing.position, ...framing.target, ...framing.up, framing.fov, framing.near, framing.far].map((value) => value.toFixed(4)).join("|") + `|${framing.cameraId}`;
}

/**
 * 五个正交轴向的视线方向与 up。这五个轴上视线与 up 恒定正交（都是不同的主轴向量），
 * 不会出现 CAM 取景那种「roll 导致 up 与视线共线」的退化情形，因此不需要
 * resolveDirectorViewUp 那套回落链。
 *
 * 方向约定（世界系，Y-up）：
 * - top：看向 -Y（从上往下看），up 取 -Z。
 * - front：看向 -Z，up 取 +Y。
 * - back：看向 +Z，up 取 +Y —— front 的正对面。
 * - left：看向 +X（从场景左侧看向右侧），up 取 +Y。
 * - right：看向 -X，up 取 +Y —— left 的正对面。
 */
type DirectorOrthographicAxis = "top" | "front" | "back" | "left" | "right";

/**
 * 正交画面上的世界轴。horizontal/vertical 是投影后的屏幕轴，不是视线轴：
 * - top：水平 X / 竖直 Z
 * - front / back：水平 X / 竖直 Y
 * - left / right：水平 Z / 竖直 Y
 */
const DIRECTOR_ORTHOGRAPHIC_AXES: Record<DirectorOrthographicAxis, { view: DirectorVec3; up: DirectorVec3; horizontal: 0 | 1 | 2; vertical: 0 | 1 | 2 }> = {
    top: { view: [0, -1, 0], up: [0, 0, -1], horizontal: 0, vertical: 2 },
    front: { view: [0, 0, -1], up: [0, 1, 0], horizontal: 0, vertical: 1 },
    back: { view: [0, 0, 1], up: [0, 1, 0], horizontal: 0, vertical: 1 },
    left: { view: [1, 0, 0], up: [0, 1, 0], horizontal: 2, vertical: 1 },
    right: { view: [-1, 0, 0], up: [0, 1, 0], horizontal: 2, vertical: 1 },
};

/** 穷举类型守卫：合法的正交轴向集合在类型层面就是穷举的，调用方无需再查能力表判断。 */
function isDirectorOrthographicAxis(mode: DirectorViewMode): mode is DirectorOrthographicAxis {
    return mode === "top" || mode === "front" || mode === "back" || mode === "left" || mode === "right";
}

function directorFinitePoint(point: DirectorVec3): boolean {
    return point.every(Number.isFinite);
}

/**
 * 取景点云：可见对象的旋转后缩放包围盒角点 + 全部摄影机/灯光的位置。
 *
 * 只收对象中心会把大比例背景板、长道具等裁掉。资源尚未加载时无法读取真实 geometry，
 * transform.scale 是当前场景合同里稳定可用的保守代理；旋转后的八个角点还能覆盖斜放对象。
 */
function collectDirectorFramingPoints(scene: DirectorScene): DirectorVec3[] {
    const points: DirectorVec3[] = [];
    for (const object of scene.objects) {
        const { position, rotation, scale } = object.transform;
        if (!object.visible || !directorFinitePoint(position)) continue;
        points.push(position);
        if (![...rotation, ...scale].every(Number.isFinite)) continue;
        const half = scale.map((value) => Math.abs(value) / 2) as DirectorVec3;
        const euler = new Euler(...rotation);
        for (const x of [-half[0], half[0]]) {
            for (const y of [-half[1], half[1]]) {
                for (const z of [-half[2], half[2]]) {
                    const corner = new Vector3(x, y, z).applyEuler(euler).add(new Vector3(...position));
                    points.push(corner.toArray() as DirectorVec3);
                }
            }
        }
    }
    for (const camera of scene.cameras) if (directorFinitePoint(camera.transform.position)) points.push(camera.transform.position);
    for (const light of scene.lights) if (directorFinitePoint(light.transform.position)) points.push(light.transform.position);
    return points;
}

type DirectorFramingBounds = { center: DirectorVec3; extent: DirectorVec3 };

// 空场景 / 全部点位非法时的安全默认包围盒，贴近场景模板的默认布景尺度（见 director-templates.ts 三点布光半径）。
const DIRECTOR_ORTHOGRAPHIC_DEFAULT_BOUNDS: DirectorFramingBounds = { center: [0, 0, 0], extent: [6, 6, 6] };
// 单轴范围下限：单点或共面内容不能把该轴压成 0，否则正交相机等于无限缩放。
const DIRECTOR_ORTHOGRAPHIC_MIN_EXTENT = 2;
// 构图留白：内容贴着画面边缘看不清楚，按包围盒放大这个比例再取景。
const DIRECTOR_ORTHOGRAPHIC_MARGIN = 1.2;
// 相机与目标的距离相对包围盒半径的倍数：太近会让正交相机的机位穿进内容内部。
const DIRECTOR_ORTHOGRAPHIC_DISTANCE_FACTOR = 2;
const DIRECTOR_ORTHOGRAPHIC_NEAR = 0.1;
// far 相对距离额外预留的深度倍数，保证视线方向上的内容不被裁切。
const DIRECTOR_ORTHOGRAPHIC_FAR_MARGIN = 4;
// 视锥换算的跨度下限：0 / 负 / 非有限值都不能当除数或半范围。
const DIRECTOR_ORTHOGRAPHIC_MIN_SPAN = 1e-3;
// 非法宽高比（0 / 负 / NaN / Infinity）回落正方形，避免半范围变成 NaN 或 Infinity。
const DIRECTOR_ORTHOGRAPHIC_DEFAULT_ASPECT = 1;

/** 点云的轴对齐包围盒中心与三轴范围。点云为空时回落安全默认值，每轴范围不低于最小下限。 */
function resolveDirectorFramingBounds(points: DirectorVec3[]): DirectorFramingBounds {
    if (points.length === 0) return DIRECTOR_ORTHOGRAPHIC_DEFAULT_BOUNDS;
    const min: DirectorVec3 = [Infinity, Infinity, Infinity];
    const max: DirectorVec3 = [-Infinity, -Infinity, -Infinity];
    for (const point of points) {
        for (let axis = 0; axis < 3; axis += 1) {
            if (point[axis] < min[axis]) min[axis] = point[axis];
            if (point[axis] > max[axis]) max[axis] = point[axis];
        }
    }
    const center = [0, 1, 2].map((axis) => (min[axis] + max[axis]) / 2) as DirectorVec3;
    const extent = [0, 1, 2].map((axis) => Math.max(max[axis] - min[axis], DIRECTOR_ORTHOGRAPHIC_MIN_EXTENT)) as DirectorVec3;
    return { center, extent };
}

/** 正交轴向取景数据：视口相机需要被写入的完整状态。 */
export type DirectorOrthographicFraming = {
    position: DirectorVec3;
    target: DirectorVec3;
    /** 世界系 up，恒定为主轴向量，绝不与视线共线。 */
    up: DirectorVec3;
    /** 投影后的水平跨度（已含构图留白）。不得单独按高度×宽高比展开，否则宽内容会被裁切。 */
    horizontalSpan: number;
    /** 投影后的竖直跨度（已含构图留白）。 */
    verticalSpan: number;
    near: number;
    far: number;
};

/** 正交相机的左右上下半范围。始终对称，且 width/height 等于安全后的宽高比。 */
export type DirectorOrthographicFrustum = {
    halfWidth: number;
    halfHeight: number;
    left: number;
    right: number;
    top: number;
    bottom: number;
};

function directorSafePositive(value: number, fallback: number): number {
    return Number.isFinite(value) && value > 0 ? value : fallback;
}

/**
 * 把投影跨度与视口宽高比收成正交半范围。
 *
 * 必须同时装下水平与竖直跨度：只按高度×aspect 算宽度时，宽于画面的内容会被裁掉。
 * aspect 为 0 / 负 / NaN / ±Infinity 时回落 1；跨度非法时回落最小正跨度。
 * 这是纯函数，不读 scene，也不写 three 相机。
 */
export function resolveDirectorOrthographicFrustum(input: { horizontalSpan: number; verticalSpan: number; aspect: number }): DirectorOrthographicFrustum {
    const horizontalSpan = directorSafePositive(input.horizontalSpan, DIRECTOR_ORTHOGRAPHIC_MIN_SPAN);
    const verticalSpan = directorSafePositive(input.verticalSpan, DIRECTOR_ORTHOGRAPHIC_MIN_SPAN);
    const aspect = directorSafePositive(input.aspect, DIRECTOR_ORTHOGRAPHIC_DEFAULT_ASPECT);
    const neededHalfWidth = horizontalSpan / 2;
    const neededHalfHeight = verticalSpan / 2;
    const halfWidth = Math.max(neededHalfWidth, neededHalfHeight * aspect);
    const halfHeight = Math.max(neededHalfHeight, neededHalfWidth / aspect);
    return {
        halfWidth,
        halfHeight,
        left: -halfWidth,
        right: halfWidth,
        top: halfHeight,
        bottom: -halfHeight,
    };
}

/**
 * 解出五个正交轴向的取景。free/camera 不是正交轴向，返回 null。
 *
 * target 与两条投影跨度来自可见对象的变换包围盒 + 全部摄影机/灯光的位置包围盒；空场景或全部
 * 位置非法（NaN/Infinity）时回落安全默认值，绝不返回会让正交相机退化成无限缩放的 0 值。
 * position 由 target 沿该轴向的视线方向反推，因此永远落在包围盒之外，不会穿模。
 */
export function resolveDirectorOrthographicFraming(input: { scene: DirectorScene; mode: DirectorViewMode }): DirectorOrthographicFraming | null {
    if (!isDirectorOrthographicAxis(input.mode)) return null;
    const axis = DIRECTOR_ORTHOGRAPHIC_AXES[input.mode];
    const stage = directorStageTransform(input.scene);
    const stageMatrix = directorStageMatrix(stage);
    const bounds = resolveDirectorFramingBounds(collectDirectorFramingPoints(input.scene).map((point) => new Vector3(...point).applyMatrix4(stageMatrix).toArray() as DirectorVec3));
    const radius = Math.max(...bounds.extent) / 2;
    const distance = Math.max(radius * DIRECTOR_ORTHOGRAPHIC_DISTANCE_FACTOR, DIRECTOR_ORTHOGRAPHIC_MIN_EXTENT);
    const target = bounds.center;
    const position: DirectorVec3 = [target[0] - axis.view[0] * distance, target[1] - axis.view[1] * distance, target[2] - axis.view[2] * distance];
    return {
        position,
        target,
        up: axis.up,
        horizontalSpan: bounds.extent[axis.horizontal] * DIRECTOR_ORTHOGRAPHIC_MARGIN,
        verticalSpan: bounds.extent[axis.vertical] * DIRECTOR_ORTHOGRAPHIC_MARGIN,
        near: DIRECTOR_ORTHOGRAPHIC_NEAR,
        far: distance + radius * DIRECTOR_ORTHOGRAPHIC_FAR_MARGIN,
    };
}

export function syncProductionBindings(next: ProductionDocument, previous: ProductionDocument | undefined, nodes: CanvasNodeData[], createId: () => string): { production: ProductionDocument; nodes: CanvasNodeData[] } {
    let production = next;
    const result = [...nodes], oldShots = new Map(previous?.shots.map(shot => [shot.shotId, shot]) || []);
    const claimed = new Set<string>();
    const right = nodes.reduce((value, node) => Math.max(value, node.position.x + node.width), 0);
    let created = 0;
    for (const original of next.shots) {
        let shot = original;
        let nodeIndex = shot.nodeId ? result.findIndex(node => node.id === shot.nodeId) : -1;
        if (nodeIndex < 0 && !oldShots.has(shot.shotId)) {
            const nodeId = createId();
            if (result.some(node => node.id === nodeId)) throw new Error('新镜头节点身份冲突，请重试');
            shot = { ...shot, nodeId };
            if (production === next) production = { ...next, shots: [...next.shots] };
            production.shots = production.shots.map(item => item.shotId === shot.shotId ? shot : item);
            result.push({ id: nodeId, type: CanvasNodeType.Image, title: shot.title, position: { x: right + (nodes.length ? 80 : 0) + created++ * 400, y: 0 }, width: 360, height: 240, metadata: { status: 'idle' } });
            nodeIndex = result.length - 1;
        }
        if (!shot.nodeId || nodeIndex < 0) continue;
        if (claimed.has(shot.nodeId)) throw new Error('一个画布节点只能绑定一个镜头');
        claimed.add(shot.nodeId);
        const node = result[nodeIndex], old = oldShots.get(shot.shotId);
        const changedMedia = shot.adoptedMedia && (JSON.stringify(shot.adoptedMedia) !== JSON.stringify(old?.adoptedMedia) || old?.nodeId !== shot.nodeId);
        const metadata = { ...node.metadata, prompt: shot.prompt, productionShotId: shot.shotId };
        if (changedMedia) {
            metadata.generationMode = shot.adoptedMedia!.kind;
            metadata.content = shot.adoptedMedia!.url || '';
            metadata.storageKey = shot.adoptedMedia!.storageKey;
            metadata.durationMs = shot.adoptedMedia!.durationMs;
            metadata.images = undefined; metadata.primaryImageId = undefined;
            metadata.naturalWidth = undefined; metadata.naturalHeight = undefined;
            metadata.bytes = undefined; metadata.mimeType = undefined;
            metadata.videoTaskId = undefined; metadata.videoStatusMessage = undefined; metadata.videoTaskProvider = undefined;
            metadata.status = 'success'; metadata.errorDetails = undefined;
        }
        result[nodeIndex] = { ...node, title: shot.title, ...(changedMedia ? { type: shot.adoptedMedia!.kind as CanvasNodeType } : {}), metadata };
    }
    for (let index = 0; index < result.length; index++) {
        const node = result[index];
        if (node.metadata?.productionShotId && !claimed.has(node.id)) result[index] = { ...node, metadata: { ...node.metadata, productionShotId: undefined } };
    }
    return { production, nodes: result };
}

/** Node layout/media generation never rewrites adopted assets; only shared text edits map back. */
export function productionFromCanvasNodes(production: ProductionDocument, previousNodes: CanvasNodeData[], nextNodes: CanvasNodeData[]): ProductionDocument {
    const before = new Map(previousNodes.map(node => [node.id, node])), after = new Map(nextNodes.map(node => [node.id, node]));
    let result = production;
    for (const shot of production.shots) {
        if (!shot.nodeId) continue;
        const node = after.get(shot.nodeId), old = before.get(shot.nodeId);
        if (!node) { result = applyProductionOperation(result, { type: 'shot.upsert', shot: { ...shot, nodeId: undefined } }); continue; }
        if (!old) continue;
        const title = node.title !== old.title ? node.title : shot.title;
        const prompt = node.metadata?.prompt !== old.metadata?.prompt ? node.metadata?.prompt || '' : shot.prompt;
        if (title !== shot.title || prompt !== shot.prompt) result = applyProductionOperation(result, { type: 'shot.upsert', shot: { ...shot, title, prompt } });
    }
    return result;
}

export function decideProjectMerge(localUpdatedAt: string, remote: { version: number; updatedAt: string } | null, base?: ProjectSyncBaseline, contentsEqual = false) {
    if (!remote) {
        if (!base) return "local";
        return localUpdatedAt === base.updatedAt ? "deleted" : "conflict";
    }
    if (!base) return contentsEqual ? "same" : "conflict";
    const localChanged = localUpdatedAt !== base.updatedAt;
    const remoteChanged = remote.version !== base.version;
    if (localChanged && remoteChanged) return contentsEqual ? "same" : "conflict";
    if (remoteChanged) return "remote";
    if (localChanged) return "local";
    return "same";
}

const RELOAD_KEY = "sundas:chunk-reload";

function isChunkLoadError(value: unknown) {
    const message = value instanceof Error ? `${value.name}: ${value.message}` : String(value ?? "");
    return /Failed to fetch dynamically imported module|Importing a module script failed|error loading dynamically imported module|ChunkLoadError/i.test(message);
}

function reloadOnce() {
    try {
        if (sessionStorage.getItem(RELOAD_KEY)) return;
        sessionStorage.setItem(RELOAD_KEY, "1");
    } catch {
        // Storage can be unavailable in private mode; a reload is still safer than the raw error page.
    }
    window.location.reload();
}

/** Reload once when a deploy replaced a lazy-loaded chunk under an open tab. */
export function initChunkReloadGuard() {
    if (typeof window === "undefined") return;
    const onPreloadError = (event: Event) => {
        event.preventDefault();
        reloadOnce();
    };
    const onRejection = (event: PromiseRejectionEvent) => {
        if (!isChunkLoadError(event.reason)) return;
        event.preventDefault();
        reloadOnce();
    };
    const onError = (event: ErrorEvent) => {
        if (isChunkLoadError(event.error ?? event.message)) reloadOnce();
    };
    window.addEventListener("vite:preloadError", onPreloadError);
    window.addEventListener("unhandledrejection", onRejection);
    window.addEventListener("error", onError);
    // Allow a later deployment in the same tab to trigger another one-shot reload.
    window.setTimeout(() => {
        try { sessionStorage.removeItem(RELOAD_KEY); } catch { /* ignore */ }
    }, 15_000);
}

export function hasLocalMedia(value: unknown): boolean {
    if (typeof value === 'string') return /^(blob:|data:(image|audio|video)\/)/.test(value) || /^[^:]+:(user|unowned):/.test(value);
    if (!value || typeof value !== 'object') return false;
    const record = value as Record<string, unknown>;
    if (record.pendingRemoteUpload === true) return true;
    const hosted = typeof record.storageKey === 'string' && /^(private|public)\//.test(record.storageKey)
        || [record.url,record.dataUrl,record.content].some(item => typeof item === 'string' && item.startsWith('/media/'));

        if (key === 'thumbnail' || key === 'thumbnailUrl') return false;
        if (hosted && ['url','dataUrl','content','coverUrl'].includes(key) && typeof item === 'string') return false;
        return hasLocalMedia(item);
    });
}

export function formatBytes(bytes: number) {
    if (!Number.isFinite(bytes) || bytes <= 0) {
        return "";
    }
    const units = ["B", "KB", "MB", "GB"];
    let value = bytes;
    let unitIndex = 0;
    while (value >= 1024 && unitIndex < units.length - 1) {
        value /= 1024;
        unitIndex += 1;
    }
    return `${value >= 10 || unitIndex === 0 ? value.toFixed(0) : value.toFixed(1)} ${units[unitIndex]}`;
}

export function formatDuration(ms: number) {
    const value = Math.max(0, Math.floor(ms / 1000));
    const minutes = Math.floor(value / 60);
    const seconds = value % 60;
    return minutes ? i18n.t("common.durationMinutes", { minutes, seconds: String(seconds).padStart(2, "0") }) : i18n.t("common.durationSeconds", { seconds });
}

export function getDataUrlByteSize(dataUrl: string) {
    const base64 = dataUrl.split(",", 2)[1];
    if (!base64) {
        return 0;
    }
    const padding = base64.endsWith("==") ? 2 : base64.endsWith("=") ? 1 : 0;

}

export function dataUrlToFile(image: ReferenceImage) {
    const [header, content] = image.dataUrl.split(",", 2);
    const mimeType = header.match(/data:(.*?);base64/)?.[1] || image.type || "image/png";
    const binary = atob(content || "");
    const bytes = new Uint8Array(binary.length);
    for (let index = 0; index < binary.length; index += 1) {
        bytes[index] = binary.charCodeAt(index);
    }
    return new File([bytes], image.name || "reference.png", { type: mimeType });
}

    if (!owner || currentMediaOwner() !== owner) throw new Error("登录账户已变化，请重新导入");
    const sources = new Map<string, { file: MediaFile; blob: Blob }>();
    for (const file of files) {
        const blob = zip.get(file.path);
        if (!blob) throw new Error("导入文件缺少媒体内容");
        const existing = sources.get(file.storageKey);
        if (existing) {
            const [a, b] = await Promise.all([existing.blob.arrayBuffer(), blob.arrayBuffer()]);
            const first = new Uint8Array(a), second = new Uint8Array(b);
            if (first.length !== second.length || first.some((value, index) => value !== second[index])) throw new Error("导入文件含有冲突的媒体键");
        } else sources.set(file.storageKey, { file, blob });
    }
    const result: ImportedMedia = new Map();
    for (const [oldKey, { file, blob }] of sources) {
        checkSession();
        if (currentMediaOwner() !== owner) throw new Error("登录账户已变化，请重新导入");
        const prefix = oldKey.startsWith('/media/') ? file.mimeType.startsWith('image/') ? 'image' : file.mimeType.startsWith('audio/') ? 'audio' : 'video' : oldKey.startsWith("image:") ? "image" : oldKey.split(":")[0];
        const typed = blob.type ? blob : blob.slice(0, blob.size, file.mimeType);
        const uploaded = await uploadBlobToServer(typed, prefix, owner).catch(() => null);
        checkSession();
        if (currentMediaOwner() !== owner) throw new Error("登录账户已变化，请重新导入");
        if (uploaded) {
            result.set(oldKey, { key: "", url: uploaded.url });
            continue;
        }
        const key = localMediaKey(prefix, owner, nanoid());
        const url = prefix === 'image' ? await setImageBlob(key, typed) : await setMediaBlob(key, typed);
        checkSession();
        notifyLocalMediaOnly(owner);
        result.set(oldKey, { key, url });
    }
    checkSession();
    if (currentMediaOwner() !== owner) throw new Error("登录账户已变化，请重新导入");
    return result;
}

export function remapImportedMedia<T>(value: T, media: ImportedMedia): T {
    if (Array.isArray(value)) return value.map((item) => remapImportedMedia(item, media)) as T;
    if (!value || typeof value !== "object") return value;
    const source = value as Record<string, unknown>;
    const data = source.data && typeof source.data === "object" ? source.data as Record<string, unknown> : null;
    const storageKey = typeof source.storageKey === "string" ? source.storageKey : typeof data?.storageKey === "string" ? data.storageKey : null;
    const mapped = storageKey ? media.get(storageKey) : undefined;
    if (storageKey && !mapped) throw new Error("导入文件缺少媒体内容");
    const output: Record<string, unknown> = {};
    for (const [key, item] of Object.entries(source)) {
        if (key === "storageKey" && mapped) output[key] = mapped.key;
        else if (mapped && ["content", "url", "dataUrl", "coverUrl", "previewUrl"].includes(key) && typeof item === "string" && (item.startsWith("blob:") || item.startsWith("data:"))) output[key] = mapped.url;
        else output[key] = remapImportedMedia(item, media);
    }
    return output as T;
}

import type localforageType from "localforage";

export type LocalForageInstance = ReturnType<typeof localforageType.createInstance>;

/**
 * localforage 驱动按需加载：只有真正读写本地缓存时才下载它，
 * 首屏不为这 30KB+ 的 IndexedDB/WebSQL 适配层买单。
 * 加载失败时清掉缓存的 Promise，下一次调用可以重试。
 */
let driver: Promise<typeof localforageType> | null = null;

export function loadLocalforage(): Promise<typeof localforageType> {
    if (!driver) {
        driver = import("localforage").then((module) => module.default);
        driver.catch(() => {
            driver = null;
        });
    }
    return driver;
}

const instances = new Map<string, Promise<LocalForageInstance>>();

/** 取（并缓存）某个 name/storeName 组合的实例。 */
export function getLocalForageStore(storeName: string, name = "sundas"): Promise<LocalForageInstance> {
    const key = `${name}::${storeName}`;
    const cached = instances.get(key);
    if (cached) return cached;
    const created = loadLocalforage().then((localforage) => localforage.createInstance({ name, storeName }));
    instances.set(key, created);
    created.catch(() => instances.delete(key));
    return created;
}

export type LazyStore = Pick<LocalForageInstance, "getItem" | "setItem" | "removeItem" | "iterate">;

/**
 * 与 `localforage.createInstance()` 行为一致的懒加载代理：
 * 调用点不需要改成 `await`，每次方法调用的返回值仍然是原来的 Promise。
 */
export function lazyStore(storeName: string, name = "sundas"): LazyStore {
    const withStore = <T>(run: (store: LocalForageInstance) => T | Promise<T>): Promise<T> => getLocalForageStore(storeName, name).then(run);
    const api = {
        getItem: <T>(key: string) => withStore((store) => store.getItem<T>(key)),
        setItem: (key: string, value: unknown) => withStore((store) => store.setItem(key, value)),
        removeItem: (key: string) =>
            withStore(async (store) => {
                await store.removeItem(key);
            }),
        iterate: <T, U>(callback: (value: T, key: string, iterationNumber: number) => U) => withStore((store) => store.iterate<T, U>(callback)),
    };
    return api as LazyStore;
}

export class MediaObjectUrls {
    private urls = new Map<string, string>();
    private pending = new Map<string, Promise<string | undefined>>();
    private changed: () => void;

    constructor(changed: () => void = () => {}) { this.changed = changed; }

    get(key: string) { return this.urls.get(key); }

    set(key: string, url: string) {
        const old = this.urls.get(key);
        this.pending.delete(key);
        if (old && old !== url) URL.revokeObjectURL(old);
        this.urls.set(key, url);
        if (old !== url) this.changed();
    }

    resolve(key: string, load: (isCurrent: () => boolean) => Promise<Blob | null | undefined>): Promise<string | undefined> {
        const cached = this.urls.get(key);
        if (cached) return Promise.resolve(cached);
        const existing = this.pending.get(key);
        if (existing) return existing;
        const current = () => this.pending.get(key) === request;
        const request = Promise.resolve().then(() => load(current)).then((blob) => {
            if (!blob || !current()) return undefined;
            const url = URL.createObjectURL(blob);
            this.urls.set(key, url);
            this.changed();
            return url;
        }).finally(() => { if (current()) this.pending.delete(key); });
        this.pending.set(key, request);
        return request;
    }

    delete(key: string) {
        this.pending.delete(key);
        const url = this.urls.get(key);
        if (url) { URL.revokeObjectURL(url); this.urls.delete(key); this.changed(); }
    }

    clear() {
        this.pending.clear();
        if (!this.urls.size) return;
        this.urls.forEach((url) => URL.revokeObjectURL(url));
        this.urls.clear();
        this.changed();
    }
}

type Snapshot = { url?: string; error?: string };
type Entry = { snapshot: Snapshot; listeners: Set<(value: Snapshot) => void>; controller: AbortController; pending?: Promise<void> };

// Visible consumers share one request and object URL; the last release aborts/revokes it.
export class MediaPreviewCache {
    private entries = new Map<string, Entry>();
    acquire(key: string, load: (signal: AbortSignal, manual: boolean) => Promise<Blob>, changed: (value: Snapshot) => void) {
        let entry = this.entries.get(key);
        if (!entry) { entry = { snapshot: {}, listeners: new Set(), controller: new AbortController() }; this.entries.set(key, entry); }
        const current = entry;
        current.listeners.add(changed);
        const publish = () => current.listeners.forEach(listener => listener(current.snapshot));
        const retry = (manual = true) => {
            if (current.pending || current.snapshot.url || current.controller.signal.aborted) return;
            current.snapshot = {}; publish();
            current.pending = load(current.controller.signal, manual).then(blob => {
                if (this.entries.get(key) === current && !current.controller.signal.aborted) current.snapshot = { url: URL.createObjectURL(blob) };
            }).catch(error => {
                if (!current.controller.signal.aborted) current.snapshot = { error: error instanceof Error ? error.message : '预览加载失败' };
            }).finally(() => { current.pending = undefined; publish(); });
        };
        changed(current.snapshot);
        retry(false);
        return { retry, release: () => {
            current.listeners.delete(changed);
            if (!current.listeners.size) {
                current.controller.abort();
                if (current.snapshot.url) URL.revokeObjectURL(current.snapshot.url);
                if (this.entries.get(key) === current) this.entries.delete(key);
            }
        } };
    }
}

export function inferMediaScale(size: string, storedScale?: string) {

export function inferMediaRatio(size: string, fallback = "1:1") {
    if (!size || size === "auto") return "auto";
    if (mediaRatioOptions.some((item) => item.value === size)) return size;
    const pixels = parsePixelSize(size) || parseAspectRatio(size);
    if (!pixels) return fallback;
    const target = pixels.width / pixels.height;
    return mediaRatioOptions
        .filter((item) => item.value !== "auto")
        .reduce((best, item) => {

            const bestOption = mediaRatioOptions.find((option) => option.value === best);
            const bestRatio = (bestOption?.width || 1) / Math.max(1, bestOption?.height || 1);
            return Math.abs(current - target) < Math.abs(bestRatio - target) ? item.value : best;

}

export type PendingPromptMode = "image" | "video";
export type PendingMaterialKind = "image" | "video" | "audio";

export type PendingReference = {
    id?: string;
    kind?: PendingMaterialKind;
    name: string;
    type: string;
    dataUrl: string;
    storageKey?: string;
};

export type PendingPromptPayload = {
    prompt: string;
    references: PendingReference[];
    appendReferences?: boolean;
};

const PENDING_PROMPT_KEY = "sundas:pending-workbench-prompt";
export const GUEST_DRAFT_KEY = "sundas:guest-creation-draft";
// SPA 跳转先用内存交接；sessionStorage 负责刷新恢复，禁用存储时也不能丢素材。
const pendingTransfers = new Map<string, { mode: PendingPromptMode } & PendingPromptPayload>();

export function removePromptParameter(params: URLSearchParams) {
    const remaining = new URLSearchParams(params);
    remaining.delete("prompt");
    return remaining;
}

export function storePendingPrompt(mode: PendingPromptMode, prompt: string, references: PendingReference[], owner: string, options?: { appendReferences?: boolean }) {
    const key = `${PENDING_PROMPT_KEY}:${encodeURIComponent(owner)}`;
    const value = { mode, prompt, references: references.map(reference => ({ ...reference })), appendReferences: options?.appendReferences };
    pendingTransfers.set(key, value);
    try {
        sessionStorage.setItem(key, JSON.stringify(value));
    } catch {
        // Storage can be unavailable in private or restricted contexts.
    }
}

export function takePendingPrompt(mode: PendingPromptMode, owner: string): PendingPromptPayload | null {
    const key = `${PENDING_PROMPT_KEY}:${encodeURIComponent(owner)}`;
    const transfer = pendingTransfers.get(key);
    if (transfer) {
        if (transfer.mode !== mode) return null;
        pendingTransfers.delete(key);
        try { sessionStorage.removeItem(key); } catch { /* 内存交接不依赖浏览器存储权限。 */ }
        return { prompt: transfer.prompt, references: transfer.references, appendReferences: transfer.appendReferences };
    }
    try {
        const raw = sessionStorage.getItem(key);
        if (!raw) return null;
        const value = JSON.parse(raw) as { mode?: PendingPromptMode; prompt?: string; references?: PendingReference[]; appendReferences?: boolean };
        if (value.mode !== mode || typeof value.prompt !== "string") return null;
        sessionStorage.removeItem(key);
        return { prompt: value.prompt, references: Array.isArray(value.references) ? value.references : [], appendReferences: value.appendReferences };
    } catch {
        return null;
    }
}

export type ReleaseInfo = {
    version: string;
    date: string;
    items: { type: string; content: string }[];
};

export function parseChangelog(content: string): ReleaseInfo[] {
    return content
        .split(/^## /m)
        .slice(1)
        .map((block) => {
            const [title = "", ...lines] = block.trim().split("\n");
            const [, version = title.trim(), date = ""] = title.match(/^(.+?)(?:\s+-\s+(.+))?$/) || [];
            return {
                version: version.trim(),
                date: date.trim(),
                items: lines
                    .map((line) => line.trim().match(/^\+\s+\[(.+?)\]\s+(.+)$/))
                    .filter((match): match is RegExpMatchArray => Boolean(match))
                    .map((match) => ({ type: match[1], content: match[2] })),
            };
        })
        .filter((release) => release.items.length);
}

/**
 * 路由预取：鼠标悬停/键盘聚焦导航时，提前把目标页面的 chunk 拉下来。
 * 只在浏览器空闲时下载，避免和首屏资源抢带宽；同一个路由只会预取一次。
 */

type Loader = () => Promise<unknown>;

const ROUTE_LOADERS: Record<string, Loader> = {
    "/": () => import("@/pages/home"),
    "/explore": () => import("@/pages/home"),
    "/create": () => import("@/pages/image"),
    "/assets": () => import("@/pages/assets"),
    "/prompts": () => import("@/pages/prompts"),
    "/canvas": () => import("@/pages/canvas"),
    "/account": () => import("@/pages/account"),
    "/admin": () => import("@/pages/admin"),
};

const started = new Set<string>();

export function prefetchRoute(to: string) {
    const path = to.split(/[?#]/)[0] || "/";
    const loader = ROUTE_LOADERS[path];
    if (!loader || started.has(path)) return;
    started.add(path);

    const run = () => {
        void loader().catch(() => {
            // 预取失败不打扰用户，下次悬停允许重试。
            started.delete(path);
        });
    };

    const idle = (window as Window & { requestIdleCallback?: (cb: () => void, options?: { timeout: number }) => number }).requestIdleCallback;
    if (idle) idle(run, { timeout: 1200 });
    else window.setTimeout(run, 200);
}

export function createSerializedWriter() {
    let tail: Promise<unknown> = Promise.resolve();
    return {
        write<T>(operation: () => Promise<T>): Promise<T> {
            const result = tail.then(operation);
            tail = result.catch(() => undefined);
            return result;
        },
        async flush() { await tail; },
    };
}

const KEY = "sundas:session-changed";
const SOURCE = `${Date.now()}:${Math.random()}`;

export function broadcastSessionChange() {
    if (typeof window === "undefined") return;
    const eventId = `${SOURCE}:${Date.now()}:${Math.random()}`;
    try { localStorage.setItem(KEY, eventId); } catch { /* 继续尝试广播通道。 */ }
    try {
        if (typeof BroadcastChannel !== "undefined") {
            const channel = new BroadcastChannel(KEY);
            channel.postMessage(eventId);
            channel.close();
        }
    } catch { /* 两种通知均不可用时依赖服务端账户校验。 */ }
}

export function subscribeSessionChanges(onChange: () => void) {
    if (typeof window === "undefined") return () => {};
    let lastEvent = "";
    const receive = (eventId: unknown) => {
        if (typeof eventId !== "string" || !eventId || eventId.startsWith(`${SOURCE}:`) || eventId === lastEvent) return;
        lastEvent = eventId;
        onChange();
    };
    const listener = (event: StorageEvent) => { if (event.key === KEY) receive(event.newValue); };
    window.addEventListener("storage", listener);
    let channel: BroadcastChannel | undefined;
    try {
        if (typeof BroadcastChannel !== "undefined") {
            channel = new BroadcastChannel(KEY);
            channel.onmessage = (event) => receive(event.data);
        }
    } catch { channel = undefined; }
    return () => { window.removeEventListener("storage", listener); channel?.close(); };
}

type ZipFile = {
    name: string;
    data: BlobPart;
};

// fflate 只在导入/导出压缩包时才用到，改成按需加载，首屏不必为它买单。
const loadFflate = () => import("fflate");

export async function createZip(files: ZipFile[]) {
    const { zipSync } = await loadFflate();
    const entries = await Promise.all(
        files.map(async (file) => {
            const data = new Uint8Array(await new Blob([file.data]).arrayBuffer());
            return [file.name, data] as const;
        }),
    );
    return new Blob([zipSync(Object.fromEntries(entries), { level: 0 })], { type: "application/zip" });
}

export async function readZip(file: Blob) {
    const { unzipSync } = await loadFflate();
    const entries = unzipSync(new Uint8Array(await file.arrayBuffer()));
    return new Map(Object.entries(entries).map(([name, data]) => [name, new Blob([data])]));
}

import { availablePaymentProviders, paymentClient, type PaymentClient } from '@/lib/payment-client';
import "./account.css";

type Plan = { id: string; name: string; days: number; priceCents: number; monthlyCredits: number; tierId: string | null; tierName: string | null; tierLevel: number | null; description: string; generationDiscount: number; grantCredits: number };
type RechargePackage = { id: string; name: string; credits: number; priceCents: number; originalPriceCents: number | null; description: string; sort: number };
type SiteInfo = { creditPriceCents: number; minCredits: number; payTypes: string[]; providers: string[]; alipayProducts?: PaymentClient[] };
type OrderRow = { orderNo: string; type: string; credits: number; amountCents: number; refundAmountCents: number | null; fulfillmentIssue: string | null; provider: string; payType: string; status: string; createdAt: string; paidAt: string | null };
type LedgerRow = { id: number; type: string; amount: number; balanceAfter: number; remark: string | null; createdAt: string };

const formatTime = (value?: string | null) => (value ? dayjs(value).format("YYYY-MM-DD HH:mm") : "-");

export default function AccountPage() {
    const userId = useAuthStore((state) => state.user?.id ?? null);
    const epoch = useAuthStore((state) => state.sessionEpoch);
    const ready = useAuthStore((state) => state.ready);
    if (!ready || !userId) return null;
    return <AccountContent key={`${epoch}:${userId}`} />;
}

function useAccountScope() {
    const owner = useAuthStore((state) => state.user?.id ?? null);
    const epoch = useAuthStore((state) => state.sessionEpoch);
    return { owner, current: () => Boolean(owner && matchesCanvasSession(owner, epoch, useAuthStore.getState())) };
}

/* 所有二级页共用同一个面板骨架：标题 15/600 + 说明 12 + 右上操作。 */
function Panel({ title, desc, extra, bodyClassName, className, children }: {
    title?: ReactNode;
    desc?: ReactNode;
    extra?: ReactNode;
    bodyClassName?: string;
    className?: string;
    children: ReactNode;
}) {
    return (
        <section className={className ? `acct-panel ${className}` : "acct-panel"}>
            {title || extra ? (
                <header className="acct-panel-head">
                    <div className="min-w-0">
                        {title ? <h2 className="acct-panel-title">{title}</h2> : null}
                        {desc ? <p className="acct-panel-desc">{desc}</p> : null}
                    </div>
                    {extra ? <div className="acct-panel-extra">{extra}</div> : null}
                </header>
            ) : null}
            <div className={bodyClassName ? `acct-panel-body ${bodyClassName}` : "acct-panel-body"}>{children}</div>
        </section>
    );
}

function Loading() {
    return <div className="acct-loading"><Spin /></div>;
}

function RefreshAction({ loading, onRefresh }: { loading: boolean; onRefresh: () => void }) {
    const { t } = useTranslation();
    return (
        <Button type="text" size="small" icon={<RefreshCw className="size-3.5" />} loading={loading} onClick={onRefresh}>
            {t("account.refresh")}
        </Button>
    );
}

function AccountContent() {
    const { t } = useTranslation();
    const user = useAuthStore((state) => state.user);
    const refreshWallet = useAuthStore((state) => state.refreshWallet);
    const [searchParams, setSearchParams] = useSearchParams();
    const requestedTab = searchParams.get("tab");
    const tab = ["overview", "membership", "recharge", "orders", "ledger", "redeem", "invite", "settings"].includes(requestedTab ?? "") ? requestedTab! : "overview";
    const setTab = (next: string) => setSearchParams({ tab: next });

    useEffect(() => {
        void refreshWallet();
    }, [refreshWallet]);

    return (
        // 外层必须是定高 flex 列：.jm-page 靠 flex:1 + overflow-y 自己滚。
        // 少了这层，.jm-page 的 flex:1 没有 flex 父级，高度退化成内容高，
        // 会被 layout 的 overflow-hidden 直接裁掉——表现就是二级页内容多了却没有滚动条。
        <div className="page-surface flex h-full flex-col overflow-hidden bg-background text-foreground">
            <main className="account-page jm-page">
                <div className="jm-page-inner">
                    <header className="jm-page-head">
                        <div className="min-w-0">
                            <h1 className="jm-page-title">{t("account.title")}</h1>
                            <p className="jm-page-desc truncate">{user?.email}</p>
                        </div>
                    </header>
                    <Tabs
                        className="account-tabs"
                        activeKey={tab}
                        onChange={setTab}
                        items={[
                            { key: "overview", label: t("account.overview"), children: <Overview onTab={setTab} /> },
                            { key: "membership", label: t("account.membership"), children: <Recharge membershipOnly /> },
                            { key: "recharge", label: t("account.recharge"), children: <Recharge /> },
                            { key: "orders", label: t("account.orders"), children: <Orders /> },
                            { key: "ledger", label: t("account.ledger"), children: <Ledger /> },
                            { key: "redeem", label: t("account.redeem"), children: <Redeem /> },
                            { key: "invite", label: t("account.invite"), children: <Invite /> },
                            { key: "settings", label: t("account.settings"), children: <Settings /> },
                        ]}
                    />
                </div>
            </main>
        </div>
    );
}

function Recharge({ membershipOnly = false }: { membershipOnly?: boolean }) {
    const { t } = useTranslation();
    const { message, modal } = AntApp.useApp();
    const scope = useAccountScope();
    const wallet = useAuthStore((state) => state.wallet);
    const member = wallet?.member;
    const [site, setSite] = useState<SiteInfo | null>(null);
    const [plans, setPlans] = useState<Plan[]>([]);
    const [packages, setPackages] = useState<RechargePackage[]>([]);
    const [packageId, setPackageId] = useState("");
    const [credits, setCredits] = useState(100);
    const [payType, setPayType] = useState("");
    const [provider, setProvider] = useState("");
    const [paying, setPaying] = useState(false);

    useEffect(() => {
        void Promise.all([
            apiRequest<SiteInfo>("/api/site"),
            membershipOnly ? apiRequest<{ plans: Plan[] }>("/api/plans") : Promise.resolve({ plans: [] }),
            membershipOnly ? Promise.resolve({ packages: [] }) : apiRequest<{ packages: RechargePackage[] }>("/api/recharge-packages"),
        ])
            .then(([siteData, planData, packageData]) => {
                if (!scope.current()) return;
                setSite(siteData);
                setPlans(planData.plans);
                setPackages(packageData.packages);
                const provider = availablePaymentProviders(siteData, paymentClient())[0] ?? '';
                setProvider(provider);
                setPayType(siteData.payTypes.find((value) => provider !== 'alipay' || value === 'alipay') ?? "");
            })
            .catch((error: unknown) => { if (scope.current()) message.error(error instanceof Error ? error.message : String(error)); });
    }, [message, membershipOnly]);

    if (!site) return <Loading />;

    const selected = packages.find((item) => item.id === packageId) ?? null;
    const price = selected ? selected.priceCents / 100 : ((credits || 0) * site.creditPriceCents) / 100;
    const payTypes = site.payTypes.filter((value) => provider !== 'alipay' || value === 'alipay').map((value) => ({ value, label: value === "wxpay" ? t("account.wxpay") : t("account.alipay") }));
    const providers = availablePaymentProviders(site, paymentClient()).map((value) => ({ value, label: value === 'alipay' ? t('account.alipayOfficial') : value === "zpay" ? t("account.zpay") : t("account.epay") }));
    const payDisabled = site.providers.length === 0 || payTypes.length === 0;
    function selectProvider(value: string) {
        setProvider(value);
        setPayType(site!.payTypes.find((type) => (value !== 'alipay' || type === 'alipay') && type === payType) ?? site!.payTypes.find((type) => value !== 'alipay' || type === 'alipay') ?? '');
    }

    async function submit(payload: { type: "credits" | "plan"; credits?: number; planId?: string; packageId?: string }) {
        if (!scope.current()) return;
        if (!provider || !providers.some(item => item.value === provider) || !payTypes.some((type) => type.value === payType)) {
            message.error(t("account.payNotConfigured"));
            return;
        }
        const auth = useAuthStore.getState();
        const owner = auth.ready ? auth.user?.id : undefined;
        const session = auth.sessionEpoch;
        if (!owner) { message.error(t("account.paymentAccountChanged")); return; }
        const sameAccount = () => useAuthStore.getState().ready && useAuthStore.getState().user?.id === owner && useAuthStore.getState().sessionEpoch === session;
        setPaying(true);
        try {
            const order = await apiRequest<{ orderNo: string; payUrl: string; amountCents: number; credits: number; planDays: number | null; planName: string | null; upgradeQuote: { previousEndsAt: string } | null }>("/api/orders", { method: "POST", headers: { "X-Sundas-User": owner }, json: { ...payload, provider, payType, paymentClient: paymentClient() } });
            if (!sameAccount()) throw new Error(t("account.paymentAccountChanged"));
            const amount = (order.amountCents / 100).toFixed(2);
            const content = order.upgradeQuote
                ? t("account.upgradeConfirm", { amount, end: formatTime(order.upgradeQuote.previousEndsAt) })
                : payload.type === "plan"
                    ? t("account.planOrderConfirm", { amount, name: order.planName, days: order.planDays, credits: order.credits })
                    : t("account.creditOrderConfirm", { amount, credits: order.credits });
            let cancelled = false;
            modal.confirm({ title: t(order.upgradeQuote ? "account.upgradeConfirmTitle" : "account.orderConfirmTitle"), content, okText: t("account.upgradePay"), cancelText: t("account.upgradeCancel"), onOk: async () => {
                try {
                    if (!sameAccount()) throw new Error(t("account.paymentAccountChanged"));
                    const status = await apiRequest<{ order: { payable: boolean } }>(`/api/orders/${encodeURIComponent(order.orderNo)}`, { headers: { "X-Sundas-User": owner } });
                    if (cancelled) return;
                    if (!sameAccount()) throw new Error(t("account.paymentAccountChanged"));
                    if (!status.order.payable) throw new Error(t("account.paymentOrderUnavailable"));
                    window.location.href = order.payUrl;
                } catch (error) { setPaying(false); if (!cancelled) message.error(error instanceof Error ? error.message : String(error)); }
            }, onCancel: () => { cancelled = true; setPaying(false); } });
        } catch (error) {
            message.error(error instanceof Error ? error.message : String(error));
            setPaying(false);
        }
    }

    const benefits = [
        { icon: Sparkles, title: t("account.benefitCredits"), text: t("account.benefitCreditsDesc") },
        { icon: ImagePlus, title: t("account.benefitModalities"), text: t("account.benefitModalitiesDesc") },
        { icon: Layers, title: t("account.benefitShared"), text: t("account.benefitSharedDesc") },
    ];

    const payTypeControl = (
        <div className="flex flex-wrap items-center gap-2">
            <span className="acct-label">{t("account.payType")}</span>
            {payTypes.length ? (
                <Segmented options={payTypes} value={payType} onChange={(value) => setPayType(String(value))} />
            ) : (
                <span className="acct-note">{t("account.payNotConfigured")}</span>
            )}
            {providers.length > 1 ? (
                <Segmented options={providers} value={provider} onChange={(value) => selectProvider(String(value))} />
            ) : null}
        </div>
    );

    return (
        <div className="acct-stack">
            {membershipOnly && (
                <section className="acct-panel acct-hero">
                    <div className="acct-hero-body">
                        <div className="min-w-0">
                            <div className="acct-eyebrow">
                                <Crown className="size-4" />
                                <BrandName text="SUNDAS MEMBERSHIP" imageClassName="opacity-70" />
                            </div>
                            <h2 className="acct-hero-title">{t("account.membershipHeroTitle")}</h2>
                            <p className="acct-hero-desc">{t("account.membershipHeroDesc")}</p>
                        </div>
                        <div className="acct-hero-aside">
                            <span className="acct-label">{t("account.currentMember")}</span>
                            <div className="acct-member-name">
                                <span className={`acct-dot ${member ? "is-on" : ""}`} />
                                {member ? wallet?.tierName || wallet?.planName || t("account.member") : t("account.nonMember")}
                            </div>
                            <div className="acct-kv">
                                <span>{t("account.discountLabel")}</span>
                                <span className="acct-kv-value">{formatGenerationDiscount(wallet?.generationDiscount)}</span>
                            </div>
                            <p className="acct-note mt-3.5">
                                {member ? t("account.memberUntil", { date: formatTime(wallet?.memberUntil) }) : t("account.memberActiveHint")}
                            </p>
                        </div>
                    </div>
                    <div className="acct-benefit-grid">
                        {benefits.map(({ icon: Icon, title, text }) => (
                            <div key={title}>
                                <Icon aria-hidden className="size-4 text-foreground" />
                                <h3 className="acct-benefit-title">{title}</h3>
                                <p className="acct-benefit-desc">{text}</p>
                            </div>
                        ))}
                    </div>
                </section>
            )}

            {payDisabled ? <Alert type="warning" showIcon message={t("account.payNotConfigured")} /> : null}

            {!membershipOnly && (
                <div className="acct-split">
                    <Panel title={t("account.recharge")} desc={t("account.priceHint", { price: (site.creditPriceCents / 100).toFixed(2), min: site.minCredits })}>
                        {packages.length ? (
                            <div className="acct-package-grid">
                                {packages.map((item) => (
                                    <button
                                        key={item.id}
                                        type="button"
                                        onClick={() => setPackageId((current) => (current === item.id ? "" : item.id))}
                                        className={`acct-package ${packageId === item.id ? "is-active" : ""}`}
                                        aria-pressed={packageId === item.id}
                                    >
                                        <span className="acct-package-name">{item.name}</span>
                                        <span className="acct-package-price">
                                            ¥{(item.priceCents / 100).toFixed(2)}
                                            {item.originalPriceCents && item.originalPriceCents > item.priceCents ? (
                                                <span className="acct-package-origin">¥{(item.originalPriceCents / 100).toFixed(2)}</span>
                                            ) : null}
                                        </span>
                                        <span className="acct-package-meta">{t("account.packageCredits", { credits: item.credits })}</span>
                                        {item.description ? <span className="acct-package-desc">{item.description}</span> : null}
                                    </button>
                                ))}
                            </div>
                        ) : null}

                        <div className={`acct-amount-row ${packages.length ? "" : "is-first"}`}>
                            <div>
                                <label className="acct-label mb-2" htmlFor="account-credits">
                                    {t("account.customAmount")}
                                    <span className="acct-hint">{t("account.customCreditsHint", { min: site.minCredits })}</span>
                                </label>
                                <InputNumber
                                    id="account-credits"
                                    size="large"
                                    min={site.minCredits}
                                    step={10}
                                    value={credits}
                                    onChange={(value) => { setCredits(Number(value) || site.minCredits); setPackageId(""); }}
                                />
                            </div>
                        </div>
                    </Panel>

                    <Panel title={t("account.payInfo")} desc={t("account.paySummaryHint")}>
                        <span className="acct-label">{t("account.payAmount")}</span>
                        <div className="acct-pay-amount">¥{price.toFixed(2)}</div>
                        <p className="acct-note mt-2">
                            {selected ? t("account.selectedPackage", { name: selected.name, credits: selected.credits }) : t("account.packageCredits", { credits })}
                        </p>
                        <div className="acct-pay-block">
                            <span className="acct-label">{t("account.payType")}</span>
                            {payTypes.length ? (
                                <Segmented block options={payTypes} value={payType} onChange={(value) => setPayType(String(value))} />
                            ) : (
                                <span className="acct-note">{t("account.payNotConfigured")}</span>
                            )}
                        </div>
                        {providers.length > 1 ? (
                            <div className="acct-pay-block">
                                <span className="acct-label">{t("account.provider")}</span>
                                <Segmented block options={providers} value={provider} onChange={(value) => selectProvider(String(value))} />
                            </div>
                        ) : null}
                        <div className="acct-pay-footer">
                            <Button
                                type="primary"
                                size="large"
                                block
                                loading={paying}
                                disabled={payDisabled}
                                onClick={() => void submit(selected ? { type: "credits", packageId: selected.id } : { type: "credits", credits })}
                            >
                                {t("account.payNow")}
                            </Button>
                        </div>
                    </Panel>
                </div>
            )}

            {membershipOnly && (plans.length ? (
                <>
                    <Panel title={t("account.membershipPlans")} desc={t("account.membershipPlansDesc")} extra={payTypeControl}>
                        {member && <p className="acct-note mb-4">{t("account.renewalCurrentTerms")}</p>}
                        <div className="acct-plan-grid">
                            {plans.map((plan) => {
                                const current = wallet?.planId === plan.id;
                                return (
                                    <div key={plan.id} className={`acct-plan ${current ? "is-current" : ""}`}>
                                        <div className="acct-plan-meta">
                                            <span>{plan.tierLevel ? `LV.${plan.tierLevel} · ${plan.tierName}` : t("account.plans")}</span>
                                            {current ? <span className="acct-plan-badge">{t("account.currentPlan")}</span> : null}
                                        </div>
                                        <h3 className="acct-plan-name">{plan.name}</h3>
                                        <p className="acct-plan-desc">{plan.description || t("account.membershipPlansDesc")}</p>
                                        <div className="acct-plan-price">
                                            ¥{(plan.priceCents / 100).toFixed(2)}
                                            <span>/ {t("account.planDays", { days: plan.days })}</span>
                                        </div>
                                        <div className="acct-plan-discount">
                                            <span className="acct-label">{t("account.discountLabel")}</span>
                                            <div className="acct-plan-discount-value">{formatGenerationDiscount(plan.generationDiscount)}</div>
                                            <p className="acct-note mt-1.5">基础报价 100 积分，折后 {Math.ceil(100 * plan.generationDiscount)} 积分</p>
                                            <p className="acct-note">仅演示会员折扣，实际按模型及活动报价计费。</p>
                                        </div>
                                        <ul className="acct-plan-features">
                                            <li><Check aria-hidden className="size-3.5" />{t("account.planMonthly", { credits: plan.monthlyCredits })}</li>
                                            {plan.grantCredits > 0 && <li><Check aria-hidden className="size-3.5" />开通赠送 {plan.grantCredits} 积分</li>}
                                            <li><Check aria-hidden className="size-3.5" />工作台与画布共享会员折扣</li>
                                            <li><Check aria-hidden className="size-3.5" />有效期 {t("account.planDays", { days: plan.days })}</li>
                                        </ul>
                                        <Button
                                            type={current ? "primary" : "default"}
                                            loading={paying}
                                            disabled={payDisabled}
                                            onClick={() => void submit({ type: "plan", planId: plan.id })}
                                        >
                                            {current ? "续费当前套餐" : member ? "选择此套餐" : "开通会员"}
                                            <ArrowUpRight className="size-4" />
                                        </Button>
                                    </div>
                                );
                            })}
                        </div>
                    </Panel>

                    <section className="acct-panel">
                        <div className="acct-panel-body grid gap-6 sm:grid-cols-2">
                            <div>
                                <h3 className="acct-benefit-title mt-0">折扣如何生效？</h3>
                                <p className="acct-benefit-desc">开通后自动用于生成报价与扣费；最终积分向上取整，单次最低 1 积分。当前会员按购买时的折扣执行，新购或续费以订单确认的权益为准。</p>
                            </div>
                            <div>
                                <h3 className="acct-benefit-title mt-0">升级与续费如何计算？</h3>
                                <p className="acct-benefit-desc">同套餐续费延长有效期；升档仅覆盖原会员剩余时间，按新旧套餐价差及剩余时间比例计算，积分按周期差额折算。具体以订单确认信息为准。</p>
                            </div>
                        </div>
                    </section>
                </>
            ) : <Panel title={t("account.membershipPlans")}><EmptyState text="暂无可购买的会员套餐，请稍后再来。" /></Panel>)}
        </div>
    );
}

function EmptyState({ text }: { text: string }) {
    return (
        <div className="jm-empty">
            <span className="jm-empty-icon"><Inbox className="size-6" strokeWidth={1.5} /></span>
            <p className="jm-empty-title">{text}</p>
        </div>
    );
}

function Orders() {
    const { t } = useTranslation();
    const { message } = AntApp.useApp();
    const scope = useAccountScope();
    const [rows, setRows] = useState<OrderRow[]>([]);
    const [loading, setLoading] = useState(true);

    async function load() {
        if (!scope.owner || !scope.current()) return;

function Settings() {
    const { t } = useTranslation();
    const { message } = AntApp.useApp();
    const scope = useAccountScope();
    const user = useAuthStore((state) => state.user);
    const wallet = useAuthStore((state) => state.wallet);
    const changePassword = useAuthStore((state) => state.changePassword);
    const [oldPassword, setOldPassword] = useState("");
    const [newPassword, setNewPassword] = useState("");
    const [confirm, setConfirm] = useState("");
    const [loading, setLoading] = useState(false);

    async function submit() {
        if (!scope.owner || !scope.current()) return;
        if (newPassword.length < 8) {
            message.error(t("auth.passwordTooShort"));
            return;
        }
        if (newPassword !== confirm) {
            message.error(t("auth.passwordMismatch"));
            return;
        }
        setLoading(true);
        try {
            await changePassword(oldPassword, newPassword, scope.owner);
            if (!scope.current()) return;
            message.success(t("account.passwordChanged"));
            setOldPassword("");
            setNewPassword("");
            setConfirm("");
        } catch (error) {
            if (scope.current()) message.error(error instanceof Error ? error.message : String(error));
        } finally {
            if (scope.current()) setLoading(false);
        }
    }

    return (
        <div className="acct-settings-grid">
            <Panel title={t("account.accountInfo")} desc={t("account.accountInfoHint")}>
                <div className="acct-info-row">
                    <span className="acct-label">{t("account.emailLabel")}</span>
                    <span className="acct-info-value">{user?.email}</span>
                </div>
                <div className="acct-info-row">
                    <span className="acct-label">{t("account.member")}</span>
                    <span className="acct-info-value">
                        {wallet?.member ? (wallet?.tierName || wallet?.planName || t("account.member")) : t("account.nonMember")}
                    </span>
                </div>
                <div className="acct-info-row">
                    <span className="acct-label">{t("account.balance")}</span>
                    <span className="acct-info-value tabular-nums">{wallet?.availableCredits ?? wallet?.balance ?? 0}</span>
                </div>
            </Panel>

            <Panel title={t("account.changePassword")} desc={t("account.passwordHint")}>
                <div className="acct-field-grid">
                    <label className="acct-field">
                        <span className="acct-label">{t("account.oldPassword")}</span>
                        <Input.Password size="large" value={oldPassword} onChange={(event) => setOldPassword(event.target.value)} autoComplete="current-password" />
                    </label>
                    <label className="acct-field">
                        <span className="acct-label">{t("account.newPassword")}</span>
                        <Input.Password size="large" value={newPassword} onChange={(event) => setNewPassword(event.target.value)} autoComplete="new-password" />
                    </label>
                    <label className="acct-field">
                        <span className="acct-label">{t("account.confirmNewPassword")}</span>
                        <Input.Password size="large" value={confirm} onChange={(event) => setConfirm(event.target.value)} autoComplete="new-password" />
                    </label>
                </div>
                <div className="acct-form-footer is-end">
                    <Button type="primary" size="large" loading={loading} onClick={() => void submit()}>
                        {t("account.savePassword")}
                    </Button>
                </div>
            </Panel>
        </div>
    );
}

function permissionLabel(value: string) {
    const [section, action] = value.split(".");
    return `${sections[section] ?? section}：${actions[action] ?? action}`;

export function AccessTab() {

    const [roles, setRoles] = useState<AdminRole[]>([]); const [all, setAll] = useState<string[]>([]); const [admins, setAdmins] = useState<Administrator[]>([]);

    const [role, setRole] = useState<RoleDraft | null>(null); const [assignment, setAssignment] = useState<{ id?: string; userId: string; roles: string[] } | null>(null);
    const report = (error: unknown) => message.error(error instanceof Error ? error.message : String(error));
    async function load() { setLoading(true); try { const [a, b] = await Promise.all([api.roles(), api.administrators()]); setRoles(a.roles); setAll(a.permissions); setAdmins(b.administrators); } catch (e) { report(e); } finally { setLoading(false); } }

    async function saveAdmin() { if (!assignment) return; setSaving(true); try { if (assignment.id) await api.assign(assignment.id, assignment.roles); else await api.addAdmin(assignment.userId, assignment.roles); setAssignment(null); message.success("管理员角色已保存"); await load(); } catch (e) { report(e); } finally { setSaving(false); } }
    function removeRole(item: AdminRole) { modal.confirm({ title: `删除角色 ${item.name}？`, onOk: async () => { try { await api.deleteRole(item.id); await load(); } catch (e) { report(e); } } }); }
