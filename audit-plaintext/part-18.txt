type SettingsState = Record<string, Record<string, unknown>>;

export function SettingsTab() {
    const apiRequest = useAdminRequest();
    const { message } = AntApp.useApp();
    const [settings, setSettings] = useState<SettingsState>({});
    const [saved, setSaved] = useState<SettingsState>({});
    const [activeTab, setActiveTab] = useState("site");
    const [saving, setSaving] = useState("");
    const [loading, setLoading] = useState(true);
    const [error, setError] = useState("");

    const load = useCallback(async () => {
        setLoading(true);
        setError("");
        try {
            const data = await apiRequest<{ settings: SettingsState }>("/api/admin/settings");
            setSettings(data.settings);
            setSaved(data.settings);
        } catch (reason) {
            const text = reason instanceof Error ? reason.message : String(reason);
            setError(text);
            message.error(text);
        } finally {
            setLoading(false);
        }
    }, [message]);

    useEffect(() => {
        void load();
    }, [load]);

    function update(section: SectionKey, field: string, value: unknown) {
        setSettings((previous) => ({ ...previous, [section]: { ...(previous[section] ?? {}), [field]: value } }));
    }

    function isDirty(tab: SettingsTab) {
        return tab.sections.some((section) => JSON.stringify(settings[section] ?? {}) !== JSON.stringify(saved[section] ?? {}));
    }

    async function save(tab: SettingsTab) {
        setSaving(tab.key);
        try {
            for (const section of tab.sections) {
                await apiRequest(`/api/admin/settings/${section}`, { method: "PUT", json: { value: settings[section] ?? {} } });
            }
            setSaved((previous) => {
                const next = { ...previous };
                for (const section of tab.sections) next[section] = settings[section] ?? {};
                return next;
            });
            if (tab.sections.includes("site")) clearSiteInfoCache();
            message.success("设置已保存，下一次请求生效");
        } catch (reason) {
            message.error(reason instanceof Error ? reason.message : String(reason));
        } finally {
            setSaving("");
        }
    }

    if (loading) return <AdminLoading label="正在读取系统设置" />;
    if (error && !Object.keys(settings).length) {
        return (
            <AdminPanel title="设置暂时不可用" description={error} actions={<Button icon={<RefreshCw className="size-4" />} onClick={() => void load()}>重试</Button>}>
                <p className="text-sm text-[var(--admin-muted)]">检查 API 服务或管理员权限后重试。</p>
            </AdminPanel>
        );
    }

    return (
        <div>
            <div className="mb-5 flex flex-wrap items-center justify-between gap-3">
                <p className="text-xs leading-5 text-[var(--admin-muted)]">修改保存后立即写入数据库，支付与邮件参数会在下一次请求使用。</p>
                <AdminBadge tone="neutral" dot={false}>按页签分别保存</AdminBadge>
            </div>
            <Tabs
                activeKey={activeTab}
                onChange={setActiveTab}
                items={TABS.map((tab) => ({
                    key: tab.key,
                    label: (
                        <span className="inline-flex items-center gap-2">
                            <span className="hidden sm:inline-flex">
                                <tab.icon className="size-3.5" />
                            </span>
                            {tab.label}
                            {isDirty(tab) ? <span className="size-1.5 rounded-full bg-[var(--admin-warning)]" /> : null}
                        </span>
                    ),
                    children: (
                        <SettingsPane
                            tab={tab}
                            settings={settings}
                            update={update}
                            dirty={isDirty(tab)}
                            saving={saving === tab.key}
                            disabled={Boolean(saving)}
                            onSave={() => void save(tab)}
                        />
                    ),
                }))}
            />
        </div>
    );
}

function SettingsPane({
    tab,
    settings,
    update,
    dirty,
    saving,
    disabled,
    onSave,
}: {
    tab: SettingsTab;
    settings: SettingsState;
    update: (section: SectionKey, field: string, value: unknown) => void;
    dirty: boolean;
    saving: boolean;
    disabled: boolean;
    onSave: () => void;
}) {
    const canManage = useAdminPermissions().includes("settings.manage");
    return (
        <AdminPanel
            title={tab.label}
            description={tab.description}
            actions={dirty ? <AdminBadge tone="warning" dot={false}>有未保存修改</AdminBadge> : <AdminBadge tone="positive" dot={false}>已同步</AdminBadge>}
        >
            <form autoComplete="off" onSubmit={(event) => event.preventDefault()}>
                <fieldset disabled={!canManage}>
                {tab.key === "payments" ? (
                    <div className="grid gap-4 xl:grid-cols-2">
                        {tab.sections.map((section) => (
                            <PaymentSection key={section} section={SECTIONS[section]} settings={settings} update={update} />
                        ))}
                    </div>
                ) : (
                    <SectionFields section={SECTIONS[tab.sections[0]]} settings={settings} update={update} />
                )}
                <div className="mt-6 flex flex-wrap items-center justify-between gap-3 border-t border-[var(--admin-line)] pt-4">
                    <span className="text-[11px] text-[var(--admin-muted)]">保存后立即生效；密钥类字段不会回显明文。</span>
                    {canManage && <Button type="primary" loading={saving} disabled={!dirty || disabled} onClick={onSave}>
                        保存更改
                    </Button>}
                </div>
                </fieldset>
            </form>
        </AdminPanel>
    );
}

function PaymentSection({
    section,
    settings,
    update,
}: {
    section: Section;
    settings: SettingsState;
    update: (section: SectionKey, field: string, value: unknown) => void;
}) {
    const enabled = Boolean(settings[section.key]?.enabled);
    return (
        <section className="rounded-[12px] border border-[var(--admin-line)] bg-[var(--admin-panel-soft)] p-4">
            <div className="mb-4 flex items-center justify-between gap-3">
                <h3 className="text-sm font-semibold text-[var(--admin-ink)]">{section.title}</h3>
                <AdminBadge tone={enabled ? "positive" : "neutral"}>{enabled ? "已启用" : "未启用"}</AdminBadge>
            </div>
            <div className="grid gap-4">
                {section.fields.filter((field) => field.type !== "bool").map((field) => (
                    <SettingControl key={field.name} section={section} field={field} settings={settings} update={update} />
                ))}
            </div>
            <div className="mt-4 border-t border-[var(--admin-line)] pt-4">
                {section.fields.filter((field) => field.type === "bool").map((field) => (
                    <SettingControl key={field.name} section={section} field={field} settings={settings} update={update} />
                ))}
            </div>
        </section>
    );
}

function SectionFields({ section, settings, update }: { section: Section; settings: SettingsState; update: (section: SectionKey, field: string, value: unknown) => void }) {
    return (
        <div className="grid gap-x-5 gap-y-5 md:grid-cols-2">
            {section.fields.map((field) => (
                <SettingControl key={field.name} section={section} field={field} settings={settings} update={update} className={field.span === 2 ? "md:col-span-2" : undefined} />
            ))}
        </div>
    );
}

function SettingControl({
    section,
    field,
    settings,
    update,
    className,
}: {
    section: Section;
    field: SettingField;
    settings: SettingsState;
    update: (section: SectionKey, field: string, value: unknown) => void;
    className?: string;
}) {
    const value = settings[section.key]?.[field.name];
    const name = `sundas-${section.key}-${field.name}`;
    const apiRequest = useAdminRequest();
    const [promptSources, setPromptSources] = useState<{ id: string; name: string }[]>([]);
    const [models, setModels] = useState<{ id: string; displayName: string; capability: string; enabled: boolean; routes: { unit: string; protocol: string; textMode: string; enabled: boolean }[] }[]>([]);

    useEffect(() => {
        if (field.type !== "source") return;
        let alive = true;
        fetchPromptSources().then((list) => { if (alive) setPromptSources(list); }).catch(() => { if (alive) setPromptSources([]); });
        return () => { alive = false; };
    }, [field.type]);

    useEffect(() => {
        if (field.type !== "model") return;
        let alive = true;
        apiRequest<{ models: { id: string; displayName: string; capability: string; enabled: boolean; routes: { unit: string; protocol: string; textMode: string; enabled: boolean }[] }[] }>("/api/admin/models")
            .then((data) => { if (alive) setModels(data.models); })
            .catch(() => { if (alive) setModels([]); });
        return () => { alive = false; };
    }, [apiRequest, field.type]);

    if (field.type === "source") {
        return (
            <AdminField label={field.label} hint={field.hint} className={className}>
                <Select
                    className="w-full"
                    allowClear
                    value={String(value ?? "") || undefined}

            </AdminField>
        );
    }

    if (field.type === "list") {
        return (
            <AdminField label={field.label} hint={field.hint} className={className}>
                <Input
                    {...ADMIN_AUTOFILL_OFF}
                    name={name}
                    value={Array.isArray(value) ? value.join(", ") : String(value ?? "")}
                    placeholder={field.placeholder}
                    onChange={(event) => update(section.key, field.name, event.target.value.split(",").map((item) => item.trim()).filter(Boolean))}
                />
            </AdminField>
        );
    }

    return (
        <AdminField label={field.label} hint={field.hint} className={className}>
            {field.type === "password" ? (

                <Input {...ADMIN_AUTOFILL_OFF} name={name} value={String(value ?? "")} placeholder={field.placeholder} onChange={(event) => update(section.key, field.name, event.target.value)} />
            )}
        </AdminField>
    );
}

    );

import { useMemo } from "react";
import { ApiError, apiRequest, type ApiRequestInit } from "@/services/api/client";
import { useAuthStore } from "@/stores/use-auth-store";
import { matchesCanvasSession } from "@/lib/canvas/canvas-session";

export function createAdminRequest(owner: string | null, epoch: number) {
    const check = () => {
        const auth = useAuthStore.getState();
        if (!owner || auth.user?.role !== "admin" || !matchesCanvasSession(owner, epoch, auth)) throw new Error("管理员会话已变化，请重新操作");
    };
    return async <T = unknown>(path: string, init: ApiRequestInit = {}): Promise<T> => {
        check();
        const headers = new Headers(init.headers);
        headers.set("X-Sundas-User", owner!);
        let result: T;
        try {
            result = await apiRequest<T>(path, { ...init, headers: Object.fromEntries(headers.entries()) });
        } catch (error) {
            if (error instanceof ApiError && error.status === 403 && path !== "/api/admin/me") window.dispatchEvent(new Event("sundas-admin-permissions-changed"));
            throw error;
        }
        check();
        return result;
    };
}

export function useAdminRequest() {
    const owner = useAuthStore((state) => state.user?.id ?? null);
    const epoch = useAuthStore((state) => state.sessionEpoch);
    return useMemo(() => createAdminRequest(owner, epoch), [owner, epoch]);
}

export function useAdminScope() {
    const owner = useAuthStore((state) => state.user?.id ?? null);
    const epoch = useAuthStore((state) => state.sessionEpoch);
    return useMemo(() => ({ owner, current: () => {
        const auth = useAuthStore.getState();
        return Boolean(owner && auth.user?.role === "admin" && matchesCanvasSession(owner, epoch, auth));
    } }), [owner, epoch]);
}

import { saveAs } from "file-saver";

import { createZip, readZip } from "@/lib/zip";
import { getMediaBlob } from "@/services/file-storage";
import { getImageBlob } from "@/services/image-storage";
import { captureAccountSession } from "@/lib/account-session";
import { importMediaFiles, remapImportedMedia } from "@/lib/imported-media";
import type { Asset } from "@/stores/use-asset-store";

type AssetExportFile = {
    app: "sundas";
    version: 1;
    exportedAt: string;
    assets: Asset[];
    files: AssetExportItem[];
};

type AssetExportItem = {
    storageKey: string;
    path: string;
    mimeType: string;
    bytes: number;
};

export async function exportAssets(assets: Asset[], filename: string) {
    const session = captureAccountSession();
    const files: AssetExportItem[] = [];
    const zipFiles: { name: string; data: BlobPart }[] = [];

    await Promise.all(
        assets.map(async (asset) => {
            if (asset.kind !== "image" && asset.kind !== "video" && asset.kind !== "audio") return;
            const storageKey = asset.data.storageKey;
            if (!storageKey) return;
            const blob = asset.kind === "image" ? await getImageBlob(storageKey) : await getMediaBlob(storageKey);
            if (!blob) throw new Error("素材引用的本地文件缺失，无法完整导出；请先恢复文件后重试。");
            const path = `files/${safeFileName(storageKey)}.${fileExtension(blob.type, asset.kind)}`;
            files.push({ storageKey, path, mimeType: blob.type || asset.data.mimeType, bytes: blob.size });
            zipFiles.push({ name: path, data: blob });
        }),
    );

    const data: AssetExportFile = { app: "sundas", version: 1, exportedAt: new Date().toISOString(), assets, files };
    session.check();
    const zip = await createZip([{ name: "assets.json", data: JSON.stringify(data, null, 2) }, ...zipFiles]);
    session.check();
    saveAs(zip, filename);
}

export async function readAssetPackage(file: File) {
    const session = captureAccountSession();
    const owner = session.owner;
    if (!owner) throw new Error("请先登录");
    const zip = await readZip(file);
    session.check();
    const assetFile = zip.get("assets.json");
    if (!assetFile) throw new Error("missing assets.json");
    const data = JSON.parse(await assetFile.text()) as AssetExportFile;
    session.check();
    if (data.app !== "sundas" || data.version !== 1 || !Array.isArray(data.assets) || !Array.isArray(data.files)) throw new Error("不支持的素材文件");
    const media = await importMediaFiles(zip, data.files, owner, session.check);
    session.check();
    return remapImportedMedia(data.assets, media);
}

function safeFileName(value: string) {
    return value.replace(/[\\/:*?"<>|]/g, "_");
}

function fileExtension(mimeType: string, kind: Asset["kind"]) {
    if (mimeType.includes("png")) return "png";
    if (mimeType.includes("jpeg")) return "jpg";
    if (mimeType.includes("webp")) return "webp";
    if (mimeType.includes("gif")) return "gif";
    if (mimeType.includes("mp4")) return "mp4";
    if (mimeType.includes("webm")) return "webm";
    return kind === "image" ? "png" : "bin";
}

function SourceDialog({item,close,changed}:{item:StorageItem;close:()=>void;changed:()=>void}) {

async function readAssetMediaBlob(asset: Extract<Asset, { kind: "image" | "video" | "audio" }>) {
    const storageKey = asset.data.storageKey;
    if (storageKey) {
        const stored = asset.kind === "image" ? await getImageBlob(storageKey) : await getMediaBlob(storageKey);
        if (stored) return stored;
    }
    const url = asset.kind !== "image" ? asset.data.url : asset.data.dataUrl || asset.coverUrl;
    if (!url) return null;
    const response = await fetch(url);
    return response.ok ? response.blob() : null;
}

function assetSummary(asset: Asset) {
    if (asset.kind === "text") return asset.data.content;
    if (asset.kind === "model") return `${asset.data.fileName || "3D 模型"} · ${formatBytes(asset.data.bytes)} · ${asset.data.mimeType}`;
    return `${asset.data.width}x${asset.data.height} · ${formatBytes(asset.data.bytes)} · ${asset.data.mimeType}`;
}

function assetSearchText(asset: Asset) {
    return [asset.title, asset.source || "", asset.note || "", (asset.tags || []).join(" "), asset.kind === "text" ? asset.data.content : asset.data.mimeType].join(" ").toLowerCase();
}

import { useEffect, useRef } from "react";

import HomePage from "@/pages/home";
import { useAuthStore } from "@/stores/use-auth-store";
import { useAuthDialogStore, type AuthMode } from "@/stores/use-auth-dialog-store";

/**
 * /login、/register、/forgot-password 三个地址的落地页。
 *
 * 登录已经改为弹窗形态（参照即梦），所以这些地址不再渲染独立页面：
 * 先渲染首页作为背景，再自动浮出对应模式的弹窗；关掉弹窗就回首页。
 * 这样旧链接、`?redirect=` 回跳、RequireAuth 的重定向都继续有效。
 */
export function AuthGate({ mode }: { mode: AuthMode }) {
    const [params] = useSearchParams();
    const redirect = params.get("redirect") || "/";
    const invite = params.get("invite") || "";
    const open = useAuthDialogStore((state) => state.open);
    const user = useAuthStore((state) => state.user);
    const navigate = useNavigate();
    const openedRef = useRef(false);

    useEffect(() => {
        if (openedRef.current) return;
        openedRef.current = true;
        open(mode, { redirect, invite });
    }, [mode, redirect, invite, open]);

    // 已经登录（含从别的标签页登录）直接落到目标页
    useEffect(() => {
        if (user) navigate(redirect, { replace: true });
    }, [user, redirect, navigate]);

    // 关闭后回首页由 AuthDialog 的 closeDialog 处理：它知道当前是否停在 /login 这类地址。
    // 这里不再监听 mode 变 null，否则会和「登录成功后跳 redirect」抢路由。

    return <HomePage landing suppressPromo />;
}

        setContextMenu(guardGenerationUpdate<ContextMenuState | null>(isCurrent, null));
        setAgentUndoSnapshot(null);
        return { ...agentUndoSnapshot, projectId, title: projectTitle };
    }, [agentUndoSnapshot, projectTitle, projectId, active, owner, sessionEpoch]);

    useEffect(() => { setAgentUndoSnapshot(null); }, [owner, sessionEpoch]);

    useEffect(() => {
        if (!isCurrent()) { setAgentCanvasContext(null); return; }
        setAgentCanvasContext({ snapshot: agentSnapshot, applyOps: applyAgentOps, undoOps: undoAgentOps, canUndo: Boolean(agentUndoSnapshot) });
        return () => setAgentCanvasContext(null);
    }, [agentSnapshot, applyAgentOps, agentUndoSnapshot, setAgentCanvasContext, undoAgentOps, active, owner, sessionEpoch]);

    return { applyAgentOps };
}

import { useEffect, useRef } from "react";
import { useNavigate, useSearchParams } from "react-router-dom";
import { App, Button } from "antd";
import { Download, FileUp, LayoutGrid, Plus } from "lucide-react";
import { useTranslation } from "react-i18next";

import { readCanvasPackage } from "@/lib/canvas/canvas-import";
import { captureAccountSession } from "@/lib/account-session";
import { matchesCanvasSession } from "@/lib/canvas/canvas-session";
import { currentMediaOwner } from "@/services/cloud-media";
import { CanvasDeleteProjectsDialog } from "@/components/canvas/canvas-delete-projects-dialog";
import { CanvasProjectCard } from "@/components/canvas/canvas-project-card";
import { isCanvasAccountActive, useCanvasStore } from "@/stores/canvas/use-canvas-store";
import { useAuthStore } from "@/stores/use-auth-store";
import { useCanvasUiStore } from "@/stores/canvas/use-canvas-ui-store";
import { exportCanvasProjects } from "@/lib/canvas/canvas-export";
import { hasAgentUrlBootstrap } from "@/lib/agent/agent-url-bootstrap";

export default function CanvasPage() {
    const { message } = App.useApp();
    const { t } = useTranslation();
    const navigate = useNavigate();
    const [searchParams] = useSearchParams();
    const inputRef = useRef<HTMLInputElement>(null);
    const autoOpenRef = useRef(false);
    const hydrated = useCanvasStore((state) => state.hydrated);
    const userId = useAuthStore((state) => state.user?.id ?? null);
    const authReady = useAuthStore((state) => state.ready);
    const sessionEpoch = useAuthStore((state) => state.sessionEpoch);
    const accountReady = authReady && hydrated && isCanvasAccountActive(userId);
    const isCurrent = () => matchesCanvasSession(userId, sessionEpoch, useAuthStore.getState()) && isCanvasAccountActive(userId);
    const projects = useCanvasStore((state) => state.projects);
    const createProject = useCanvasStore((state) => state.createProject);
    const importProject = useCanvasStore((state) => state.importProject);
    const selectedIds = useCanvasUiStore((state) => state.selectedProjectIds);
    const setDeleteIds = useCanvasUiStore((state) => state.setDeleteProjectIds);

    const mode = searchParams.get("mode");
    const agentMode = mode === "new" || mode === "recent" || mode === "choose";
    const agentQuery = agentMode ? `?${searchParams.toString()}` : "";
    const enterProject = (id: string) => {
        if (!isCurrent()) return;
        const agentHash = hasAgentUrlBootstrap(window.location.hash) ? window.location.hash : "";
        navigate(`/canvas/${id}${agentQuery}${agentHash}`, { replace: Boolean(agentHash) });
    };
    const createAndEnter = () => { if (isCurrent()) enterProject(createProject()); };
    const importCanvas = async (file?: File) => {
        if (!file || !isCurrent()) return;
        try {
            const session = captureAccountSession();
            const owner = session.owner;
            const projects = await readCanvasPackage(file);
            session.check();
            if (!owner || currentMediaOwner() !== owner) throw new Error("登录账户已变化，请重新导入");
            projects.forEach(importProject);
            message.success(t("canvas.imported", { count: projects.length }));
        } catch {
            message.error(t("canvas.importFailed"));
        } finally {
            if (inputRef.current) inputRef.current.value = "";
        }
    };

    useEffect(() => {
        if (!accountReady || !isCurrent() || autoOpenRef.current || (mode !== "new" && mode !== "recent")) return;
        autoOpenRef.current = true;
        enterProject(mode === "new" ? createProject() : projects[0]?.id || createProject());
    }, [accountReady, createProject, mode, projects, t]);

    if (!accountReady) return <main className="flex h-full items-center justify-center bg-background text-sm text-muted-foreground">{t("canvas.loading")}</main>;

    if (hydrated && (mode === "new" || mode === "recent")) return <main className="flex h-full items-center justify-center bg-background text-sm text-muted-foreground">{t("canvas.opening")}</main>;

    return (
        // 外层必须是定高 flex 列：jm-page 靠 flex:1 + overflow-y 自己滚；
        // 少了这层它会被内容撑到全高，再被 layout 的 overflow-hidden 裁掉（表现为没有滚动条、底部卡片点不到）
        <div className="page-surface flex h-full min-h-0 flex-col overflow-hidden bg-background text-foreground">
            <main className="jm-page">
                <div className="jm-page-inner flex flex-col gap-6">
                    <header className="jm-page-head">
                        <div className="min-w-0">
                            <p className="mb-2 text-xs leading-4 text-muted-foreground">{t("canvas.library")}</p>
                            <h1 className="jm-page-title">{t("canvas.title")}</h1>
                        </div>
                        <div className="flex flex-wrap items-center gap-2">
                            {selectedIds.length ? (
                                <>
                                    <Button disabled={!hydrated} icon={<Download className="size-4" />} onClick={() => { if (isCurrent()) void exportCanvasProjects(projects.filter((project) => selectedIds.includes(project.id)), `${t("canvas.title")}-${selectedIds.length}`).catch((error) => message.error(error instanceof Error ? error.message : t("canvas.sidePanel.exportFailed"))); }}>
                                        {t("canvas.exportSelected")}
                                    </Button>
                                    <Button disabled={!hydrated} onClick={() => { if (isCurrent()) setDeleteIds(selectedIds); }}>
                                        {t("canvas.deleteSelected")}
                                    </Button>
                                </>
                            ) : null}
                            {projects.length ? (
                                <Button disabled={!hydrated} onClick={() => { if (isCurrent()) setDeleteIds(projects.map((project) => project.id)); }}>
                                    {t("canvas.deleteAll")}
                                </Button>
                            ) : null}
                            <Button disabled={!hydrated} icon={<FileUp className="size-4" />} onClick={() => { if (isCurrent()) inputRef.current?.click(); }}>
                                {t("canvas.import")}
                            </Button>
                            <button type="button" disabled={!hydrated} className="jm-btn jm-btn-primary" onClick={createAndEnter}>
                                <Plus className="size-4" />
                                {t("canvas.create")}
                            </button>
                        </div>
                    </header>

                    {!hydrated ? (
                        <section className="flex min-h-[360px] items-center justify-center text-sm text-muted-foreground">{t("canvas.loading")}</section>
                    ) : projects.length ? (
                        <div className="jm-canvas-grid">
                            {projects.map((project) => (
                                <CanvasProjectCard key={project.id} project={project} />
                            ))}
                        </div>
                    ) : (
                        <section className="jm-card jm-empty min-h-[360px]">
                            <div className="jm-empty-icon"><LayoutGrid className="size-8" strokeWidth={1.5} /></div>
                            <h2 className="jm-empty-title">{t("canvas.empty")}</h2>
                            <p className="jm-empty-desc">{t("canvas.emptyDescription")}</p>
                            <button type="button" className="jm-btn jm-btn-primary mt-4" onClick={createAndEnter}>
                                <Plus className="size-4" />
                                {t("canvas.create")}
                            </button>
                        </section>
                    )}
                </div>

                <input ref={inputRef} type="file" accept="application/zip,.zip" className="hidden" onChange={(event) => void importCanvas(event.target.files?.[0])} />
                <CanvasDeleteProjectsDialog />
            </main>
        </div>
    );
}
