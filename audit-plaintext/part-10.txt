export function useApiIntegrationUrl() {
    const [url, setUrl] = useState("");
    useEffect(() => {
        let active = true;
        void loadSiteInfo().then(({ site }) => {
            const value = site.apiIntegrationUrl?.trim();
            if (active && value && /^https?:\/\//i.test(value)) setUrl(value);
        }).catch(() => undefined);
        return () => { active = false; };
    }, []);
    return url;
}

import { useEffect } from "react";
import type { ReactNode } from "react";
import { App } from "antd";

import { useCloudSync } from "@/hooks/use-cloud-sync";
import { applyServerModels, type ServerModel } from "@/lib/server-models";
import { apiRequest } from "@/services/api/client";
import { useAuthStore } from "@/stores/use-auth-store";
import { subscribeSessionChanges } from "@/lib/session-broadcast";

export function ClientRootInit({ children }: { children: ReactNode }) {
    const { message } = App.useApp();
    useCloudSync();

    useEffect(() => {
        void useAuthStore.getState().hydrate().catch((error) => message.error(error instanceof Error ? error.message : "无法确认登录状态"));
        return subscribeSessionChanges(() => { void useAuthStore.getState().externalSessionChanged().catch((error) => message.error(error instanceof Error ? error.message : "无法确认登录状态")); });
    }, []);

    useEffect(() => {
        const refresh = () => { if (document.visibilityState === "visible") void useAuthStore.getState().refreshWallet().catch(() => undefined); };
        document.addEventListener("visibilitychange", refresh);
        window.addEventListener("online", refresh);
        return () => { document.removeEventListener("visibilitychange", refresh); window.removeEventListener("online", refresh); };
    }, []);

    useEffect(() => {
        let cancelled = false;
        apiRequest<{ models: ServerModel[] }>("/api/models")
            .then((data) => {
                if (!cancelled && data.models.length) applyServerModels(data.models);
            })
            .catch(() => {
                // 模型列表拉取失败时保留本地缓存，登录后还会再拉一次。
            });
        return () => {
            cancelled = true;
        };
    }, []);

    return <>{children}</>;
}

        if (!ready || !user) return;

        const videoMode = pathname === "/video" || (pathname === "/create" && new URLSearchParams(search).get("mode") === "video");

import { Suspense, lazy, useEffect, useState } from "react";

import { useAuthDialogStore } from "@/stores/use-auth-dialog-store";
import { useConfigStore } from "@/stores/use-config-store";

/**
 * 全局浮层（登录弹窗 / 偏好设置弹窗）按需加载：
 * 这两块只在用户主动触发时才出现，但会静态带进 antd 的 Form / Modal / Select / Radio / Switch，
 * 是首屏包最大的一块业务开销。默认不挂载 → chunk 不下载；
 * 首次打开后保持挂载，关闭动画与内部状态与原来一致。
 */
const AuthDialog = lazy(() => import("@/components/auth/auth-dialog").then((module) => ({ default: module.AuthDialog })));
const AppConfigModal = lazy(() => import("@/components/layout/app-config-modal").then((module) => ({ default: module.AppConfigModal })));

function useKeepMountedWhen(open: boolean) {
    const [mounted, setMounted] = useState(open);
    useEffect(() => {
        if (open) setMounted(true);
    }, [open]);
    return mounted;
}

export function AuthDialogHost() {
    const open = useAuthDialogStore((state) => state.mode !== null);
    const mounted = useKeepMountedWhen(open);
    if (!mounted) return null;
    return (
        <Suspense fallback={null}>
            <AuthDialog />
        </Suspense>
    );
}

export function AppConfigModalHost() {
    const open = useConfigStore((state) => state.isConfigOpen);
    const mounted = useKeepMountedWhen(open);
    if (!mounted) return null;
    return (
        <Suspense fallback={null}>
            <AppConfigModal />
        </Suspense>
    );
}

import { App, Button, Checkbox, Input, Modal, Tabs } from "antd";
import { RefreshCw, Search } from "lucide-react";
import { useEffect, useMemo, useState } from "react";
import { useTranslation } from "react-i18next";

import { fetchChannelModels } from "@/services/api/image";
import type { ModelChannel } from "@/stores/use-config-store";

// Channel model selector: fetch upstream models or add them manually, then include checked models in the channel list.
export function ModelSelectModal({ open, channel, selectedNames, onConfirm, onClose }: { open: boolean; channel: ModelChannel | null; selectedNames: string[]; onConfirm: (names: string[]) => void; onClose: () => void }) {
    const { message } = App.useApp();
    const { t } = useTranslation();
    const [existing, setExisting] = useState<string[]>([]);
    const [fetched, setFetched] = useState<string[]>([]);
    const [selected, setSelected] = useState<Set<string>>(new Set());
    const [activeTab, setActiveTab] = useState("new");
    const [search, setSearch] = useState("");
    const [manual, setManual] = useState("");
    const [loading, setLoading] = useState(false);

    useEffect(() => {
        if (!open) return;
        setExisting(selectedNames);
        setFetched([]);
        setSelected(new Set(selectedNames));
        setActiveTab(selectedNames.length ? "existing" : "new");
        setSearch("");
        setManual("");
    }, [open, selectedNames]);

    const currentList = activeTab === "new" ? fetched : existing;
    const visibleList = useMemo(() => {
        const keyword = search.trim().toLowerCase();
        return keyword ? currentList.filter((name) => name.toLowerCase().includes(keyword)) : currentList;
    }, [currentList, search]);
    const visibleSelectedCount = visibleList.filter((name) => selected.has(name)).length;

    const toggle = (name: string, checked: boolean) =>
        setSelected((current) => {
            const next = new Set(current);
            if (checked) next.add(name);
            else next.delete(name);
            return next;
        });

    const selectVisible = (checked: boolean) =>
        setSelected((current) => {
            const next = new Set(current);
            visibleList.forEach((name) => (checked ? next.add(name) : next.delete(name)));
            return next;
        });

    const addManual = () => {
        const name = manual.trim();
        if (!name) return;
        if (!fetched.includes(name) && !existing.includes(name)) setFetched((current) => [name, ...current]);
        setSelected((current) => new Set(current).add(name));
        setManual("");
        setActiveTab("new");
    };

    const fetchModels = async () => {
        if (!channel) return;
        if (!channel.baseUrl.trim() || !channel.apiKey.trim()) {
            message.error(t("config.modelSelect.missingConfig"));
            return;
        }
        setLoading(true);
        try {
            const models = await fetchChannelModels(channel);
            setFetched(models);
            setActiveTab("new");
            message.success(t("config.modelSelect.fetched", { count: models.length }));
        } catch (error) {
            message.error(error instanceof Error ? error.message : t("config.modelSelect.fetchFailed"));
        } finally {
            setLoading(false);
        }
    };

    const confirm = () => {
        const ordered = [...existing, ...fetched].filter((name, index, list) => list.indexOf(name) === index).filter((name) => selected.has(name));
        onConfirm(ordered);
        onClose();
    };

    return (
        <Modal
            open={open}
            width={880}
            centered
            onCancel={onClose}
            title={
                <span>
                    {t("config.modelSelect.title")} <span className="ml-2 text-xs font-normal text-muted-foreground">{t("config.modelSelect.selected", { selected: selected.size, total: new Set([...existing, ...fetched]).size })}</span>
                </span>
            }
            styles={{ body: { maxHeight: "62vh", overflowY: "auto" } }}
            footer={[
                <Button key="cancel" onClick={onClose}>
                    {t("common.cancel")}
                </Button>,
                <Button key="confirm" type="primary" onClick={confirm}>
                    {t("config.modelSelect.confirm")}
                </Button>,
            ]}
        >
            <div className="flex flex-wrap items-center gap-3">
                <Input className="min-w-[200px] flex-1" value={search} onChange={(event) => setSearch(event.target.value)} placeholder={t("config.modelSelect.search")} prefix={<Search className="size-4 text-muted-foreground" />} allowClear />
                <Input className="min-w-[180px] flex-1" value={manual} onChange={(event) => setManual(event.target.value)} onPressEnter={addManual} placeholder={t("config.modelSelect.modelName")} />
                <Button onClick={addManual}>{t("config.modelSelect.add")}</Button>
                <Button icon={<RefreshCw className="size-4" />} loading={loading} onClick={() => void fetchModels()}>
                    {t("config.modelSelect.fetch")}
                </Button>
            </div>
            <div className="mt-2 text-xs text-muted-foreground">{t("config.modelSelect.description")}</div>

            <Tabs
                className="mt-3"
                activeKey={activeTab}
                onChange={setActiveTab}
                items={[
                    { key: "new", label: t("config.modelSelect.fetchedTab", { count: fetched.length }) },
                    { key: "existing", label: t("config.modelSelect.existingTab", { count: existing.length }) },
                ]}
            />

            <div className="mb-3 flex items-center justify-between gap-2">
                <span className="text-xs text-muted-foreground">{t("config.modelSelect.visibleSelected", { selected: visibleSelectedCount, total: visibleList.length })}</span>
                <div className="flex gap-2">
                    <Button size="small" disabled={!visibleList.length} onClick={() => selectVisible(true)}>
                        {t("config.modelSelect.selectVisible")}
                    </Button>
                    <Button size="small" disabled={!visibleSelectedCount} onClick={() => selectVisible(false)}>
                        {t("config.modelSelect.clearVisible")}
                    </Button>
                </div>
            </div>

            {visibleList.length ? (
                <div className="grid grid-cols-1 gap-x-8 gap-y-3 md:grid-cols-2">
                    {visibleList.map((name) => (
                        <Checkbox key={name} checked={selected.has(name)} onChange={(event) => toggle(name, event.target.checked)}>
                            <span className="truncate" title={name}>
                                {name}
                            </span>
                        </Checkbox>
                    ))}
                </div>
            ) : (
                <div className="py-8 text-center text-sm text-muted-foreground">{t(activeTab === "new" ? "config.modelSelect.fetchedEmpty" : "config.modelSelect.existingEmpty")}</div>
            )}
        </Modal>
    );
}

import { useEffect, useLayoutEffect, useRef, type ReactNode } from "react";
import { useLocation } from "react-router-dom";

export function PageTransition({ children, disabled = false, className = "h-full min-h-0" }: { children: ReactNode; disabled?: boolean; className?: string }) {
    const { pathname, search } = useLocation();
    const frame = useRef<HTMLDivElement>(null);

    useLayoutEffect(() => {
        const preference = window.matchMedia("(prefers-reduced-motion: reduce)");
        if (disabled || preference.matches || !frame.current?.animate) return;
        const animation = frame.current.animate(
            [{ opacity: 0.55, translate: "0 6px" }, { opacity: 1, translate: "0 0" }],
            { duration: 200, easing: "cubic-bezier(0.22, 1, 0.36, 1)" },
        );
        const stop = () => { if (preference.matches) animation.cancel(); };
        preference.addEventListener("change", stop);
        return () => { animation.cancel(); preference.removeEventListener("change", stop); };
    }, [disabled, pathname, search]);

    useEffect(() => {
        if (disabled) return;
        const preference = window.matchMedia("(prefers-reduced-motion: reduce)");
        let pressed: Animation | undefined;
        const stop = () => pressed?.cancel();
        const click = (event: MouseEvent) => {
            if (preference.matches || !(event.target instanceof Element)) return;
            const control = event.target.closest<HTMLElement>("button, a[href], [role='button'], [role='tab'], [role='menuitem'], .ant-segmented-item, .ant-tag-checkable");
            if (!control?.animate || control.closest(":disabled, [aria-disabled='true'], .ant-btn-loading, .ant-tabs-tab-disabled, .ant-segmented-item-disabled")) return;
            stop();
            pressed = control.animate([{ scale: "1" }, { scale: "0.97", offset: 0.35 }, { scale: "1" }], { duration: 160, easing: "ease-out" });
        };
        // 捕获阶段覆盖导航、普通按钮及弹窗内的按钮，不改变原有点击行为。
        document.addEventListener("click", click, true);
        preference.addEventListener("change", stop);
        return () => { stop(); document.removeEventListener("click", click, true); preference.removeEventListener("change", stop); };
    }, [disabled]);

    return <div ref={frame} className={className}>{children}</div>;
}

export function LanguageMenu({ buttonClassName, iconStyle }: { buttonClassName?: string; iconStyle?: CSSProperties }) {
    const { i18n, t } = useTranslation();
    const locale = i18n.resolvedLanguage as LocaleCode;
    const locales = listLocales();
    const currentLocale = locales.find((item) => item.code === locale);
    const languageLabel = t("topNav.switchLanguage", { language: currentLocale ? currentLocale.label : locale });

    return (
        <Dropdown
            trigger={["click"]}
            placement="bottom"
            menu={{
                items: locales.map((item) => ({ key: item.code, label: item.label })),
                selectable: true,
                selectedKeys: [locale],
                onClick: ({ key }) => {
                    if (key !== locale) void changeAppLocale(key as LocaleCode);
                },
            }}
        >
            <button type="button" className={buttonClassName} style={iconStyle} aria-label={languageLabel} aria-haspopup="menu">
                {currentLocale?.short ?? "中"}
                <ChevronDown className="size-3 opacity-60" />
            </button>
        </Dropdown>
    );
}

import { useEffect, useState, type ReactNode } from 'react';
import { useMediaPreview } from '@/hooks/use-media-preview';

// Small cards never fetch a cloud original to stand in for its derivative.
export function MediaThumbnail({source,kind='image',alt='',className='grid size-full place-items-center overflow-hidden',imageClassName='size-full object-cover',fallback}: {
    source?:string;kind?:'image'|'video';alt?:string;className?:string;imageClassName?:string;fallback?:ReactNode;
}) {
    const [element,setElement]=useState<HTMLElement|null>(null),[visible,setVisible]=useState(false);
    useEffect(()=>{
        if (!element) return;
        if (typeof IntersectionObserver==='undefined') {setVisible(true);return;}
        const observer=new IntersectionObserver(entries=>setVisible(entries.some(entry=>entry.isIntersecting)));
        observer.observe(element);return()=>observer.disconnect();
    },[element]);
    const preview=useMediaPreview(visible?source:undefined);
    const url=visible?(preview.cloud?preview.url:source):undefined;
    return <span ref={setElement} className={className} title={preview.error?`${preview.error}，点击可重试缩略图`:undefined} onClick={()=>{if(preview.error)preview.retry();}}>
        {url?(kind==='video'&&!preview.cloud?<video src={url} muted preload="metadata" className={imageClassName}/>:<img src={url} alt={alt} loading="lazy" decoding="async" className={imageClassName}/>):fallback}
    </span>;
}

                                            <span className="flex w-full items-center justify-between gap-2"><span className="font-medium">{name}</span><span className="shrink-0 text-muted-foreground">{paused ? '暂不可用' : unsupported ? '切换后调整清晰度' : `${formatCredits(route.baseCredits * (multiplier ?? 1), true)} 积分/${route.unit === 'second' ? '秒' : '次'}`}</span></span>
                                        </SelectItem>;
                                    })}
                                </div>;
                            })}
                        </SelectGroup>
                    ))
                ) : (
                    <SelectItem value="__empty__" disabled className="whitespace-normal py-2 text-xs leading-5">{emptyModelLabel(config, capability)}</SelectItem>
                )}
            </SelectContent>
        </Select>
    );
}

function groupModels(config: AiConfig, options: string[], infoOf: (model: string) => ServerModelInfo | undefined): ModelGroup[] {
    const groups = new Map<string, ModelGroup>();
    for (const model of options) {
        const info = infoOf(model);
        const channel = config.channels.find((item) => item.id === model.split("::")[0]);
        const key = info?.familyId || info?.familyName || channel?.id || "other";
        const existing = groups.get(key);
        if (existing) {
            existing.models.push(model);
            continue;
        }
        const label = info?.familyName || channel?.name || i18n.t("settingsPanels.model.select");
        const hint = info?.familyVendor || (info ? `${info.routeCount} 条线路` : channel?.baseUrl || "");
        groups.set(key, { key, label, hint, models: [model] });
    }
    return [...groups.values()];
}

function emptyModelLabel(config: AiConfig, capability?: ModelCapability) {
    if (capability === "video") return "暂无可用的视频模型，请联系管理员";
    const label = capability ? i18n.t(`settingsPanels.model.capabilities.${capability}`) : "";
    if (capability && config.models.length) return i18n.t("settingsPanels.model.assign", { capability: label });
    return config.models.length ? i18n.t("settingsPanels.model.noMatch", { capability: label }) : i18n.t("settingsPanels.model.addFirst");
}

function displayRouteName(name: string) {
    if (name.includes("稳定快速版")) return "稳定快速版";
    if (name.includes("按秒线路")) return "按秒线路";
    const backup = name.match(/备用线路\s*(\d+)/);
    return backup ? `标准线路 ${backup[1]}` : name;
}

function ModelLabel({ config, model, info, hideRoutes = false }: { config: AiConfig; model: string; info?: ServerModelInfo; hideRoutes?: boolean }) {
    if (!info) {
        return (
            <span className="flex min-w-0 items-center gap-2">
                <ModelIcon model={model} />
                <span className="truncate">{modelOptionLabel(config, model)}</span>
            </span>
        );
    }
    const min = formatCredits(info.minCredits, info.capability === "video");
    const max = formatCredits(info.maxCredits, info.capability === "video");
    const price = info.routeCount || info.minCredits > 0 ? `${min === max ? min : `${min}–${max}`} 积分${info.unit === "second" ? "/秒" : info.unit === "request" ? "/次" : ""}` : "价格待定";
    return (
        <span className="flex min-w-0 flex-1 items-start gap-2.5">
            <ModelIcon model={model} icon={info.familyIcon} />
            <span className="min-w-0 flex-1">
                <span className="flex items-center justify-between gap-2">
                    <span className="truncate text-sm font-medium text-foreground">{info.name}</span>
                    <span className="shrink-0 text-[10px] tabular-nums text-muted-foreground">{price}</span>
                </span>
                {!hideRoutes && (info.description || info.routes.length > 0) && <span className="mt-0.5 block truncate text-[11px] text-muted-foreground">{info.description || info.routes.map((route) => route.channelName).join(" / ")}</span>}
                {!hideRoutes && <span className="mt-1 flex items-center gap-2">
                    <span className="text-[10px] text-muted-foreground">{info.routeCount ? `${info.routeCount} 条线路` : "即将上线"}</span>
                    <span className="flex items-center gap-1">
                        {info.routes.slice(0, 7).map((route) => (
                            <span key={route.id} title={route.channelName} className="size-1.5 rounded-full bg-green-500" />
                        ))}
                    </span>
                </span>}
            </span>
        </span>
    );
}

function formatCredits(value: number, integer = false) {
    if (integer) return `${Number.isInteger(value) ? "" : "约"}${Math.round(value)}`;
    return Number.isInteger(value) ? String(value) : String(Number(value.toFixed(2)));
}

function ModelIcon({ model, icon }: { model: string; icon?: string | null }) {
    const src = icon || resolveModelIcon(modelOptionName(model));
    return src ? <img src={src} alt="" className="mt-0.5 size-4 shrink-0 dark:invert" /> : <Cpu className="mt-0.5 size-4 shrink-0 opacity-70" />;
}

function resolveModelIcon(model: string) {
    const name = model.toLowerCase();
    if (name.includes("claude") || name.includes("anthropic")) return "/icons/claude.svg";
    if (name.includes("gemini") || name.includes("google")) return "/icons/gemini.svg";
    if (name.includes("gpt") || name.includes("openai")) return "/icons/openai.svg";
    if (name.includes("grok")) return "/icons/grok.svg";
    if (name.includes("deepseek")) return "/icons/deepseek.svg";
    if (name.includes("glm")) return "/icons/glm.svg";
    return "";
}

const FALLBACK_RATIO = 2.28;

export function PromoCarousel({ items, variant = "default" }: { items: PromoItem[]; variant?: "default" | "hero" }) {
    const action = usePromoAction();
    const [index, setIndex] = useState(0);
    const [failed, setFailed] = useState<Set<string>>(() => new Set());
    const [paused, setPaused] = useState(false);
    const [ratio, setRatio] = useState(FALLBACK_RATIO);
    const visible = useMemo(() => items.filter((item) => item.enabled), [items]);

    useEffect(() => {
        if (index >= visible.length) setIndex(0);
    }, [index, visible.length]);

    useEffect(() => {
        if (visible.length < 2 || paused) return;
        const timer = window.setInterval(() => setIndex((value) => (value + 1) % visible.length), 5000);
        return () => window.clearInterval(timer);
    }, [paused, visible.length]);

    if (!visible.length) return null;
    const current = Math.min(index, visible.length - 1);
    const item = visible[current];
    const previous = visible.length > 1 ? visible[(current - 1 + visible.length) % visible.length] : null;
    const next = visible.length > 1 ? visible[(current + 1) % visible.length] : null;

    const run = () => {
        void trackPromoClick({ placement: "carousel", itemId: item.id });
        action(item);
    };

    // 预加载相邻两张，点击切换时立即出图，避免“点了没反应”的慢加载错觉
    useEffect(() => {
        if (visible.length < 2) return;
        const neighbours = [visible[(current + 1) % visible.length], visible[(current - 1 + visible.length) % visible.length]];
        for (const entry of neighbours) {
            if (entry && !failed.has(entry.id)) {
                const img = new window.Image();
                img.src = entry.imageUrl;
            }
        }
    }, [current, visible, failed]);

    const image = (entry: PromoItem, hero: boolean) => (
        <picture>
            {entry.mobileImageUrl ? <source media="(max-width: 760px)" srcSet={entry.mobileImageUrl} /> : null}
            {failed.has(entry.id) ? (
                // 加载失败的图不再从轮播中剔除（否则切换会回落到同一张，看起来“点不动”），
                // 改为显示兜底占位，保证切换始终有可见反馈
                <span className={hero ? "promo-carousel-fallback is-hero" : "promo-carousel-fallback"}>{entry.title || entry.alt}</span>
            ) : (
                <img
                    key={entry.id}
                    className={hero ? "promo-carousel-hero-media" : "promo-carousel-media"}
                    src={entry.imageUrl}
                    alt={entry.alt}
                    onLoad={(event) => {
                        const value = event.currentTarget.naturalWidth / event.currentTarget.naturalHeight;
                        if (value > 0.5 && value < 6) setRatio(value);
                    }}
                    onError={() => {
                        setFailed((value) => new Set(value).add(entry.id));
                    }}
                />
            )}
        </picture>
    );

    // 右下角分页器：仅保留「02 / 03」计数（短横线指示条已按需求移除）
    const pager =
        visible.length > 1 ? (
            <div className="promo-carousel-pager">
                <span className="promo-carousel-count">
                    <b>{String(current + 1).padStart(2, "0")}</b>
                    <span className="promo-carousel-count-sep">/</span>
                    {String(visible.length).padStart(2, "0")}
                </span>
            </div>
        ) : null;

        return (
            <section
                className="promo-carousel is-hero"
                aria-label="首页轮播"

                onMouseLeave={() => setPaused(false)}

                onBlurCapture={() => setPaused(false)}
            >
                <div className="promo-carousel-stage" style={{ aspectRatio: ratio }}>
                    {previous ? (
                        <div className="promo-carousel-peek is-prev" aria-hidden>
                            <img src={previous.imageUrl} alt="" onError={(event) => { event.currentTarget.style.display = "none"; }} />
                        </div>
                    ) : null}
                    {next ? (
                        <div className="promo-carousel-peek is-next" aria-hidden>
                            <img src={next.imageUrl} alt="" onError={(event) => { event.currentTarget.style.display = "none"; }} />
                        </div>
                    ) : null}

                    <button type="button" className="promo-carousel-main" onClick={run} aria-label={item.alt}>
                        {image(item, true)}
                    </button>

                    {visible.length > 1 ? (
                        <button type="button" className="promo-carousel-arrow is-prev" aria-label="上一张" onClick={() => setIndex((value) => (value - 1 + visible.length) % visible.length)}>
                            <ChevronLeft className="size-4" />

                    ) : null}

                            <ChevronRight className="size-4" />
                        </button>
                    ) : null}
                </div>

                {pager}
            </section>
        );
    }

    return (
        <section
            className="promo-carousel"
            aria-label="首页轮播"
            onMouseEnter={() => setPaused(true)}
            onMouseLeave={() => setPaused(false)}
            onFocusCapture={() => setPaused(true)}
            onBlurCapture={() => setPaused(false)}
        >
            <button type="button" className="promo-carousel-slide" onClick={run} aria-label={item.alt}>
                {image(item, false)}
            </button>

            {visible.length > 1 ? (
                <>
                    <button type="button" className="promo-carousel-arrow is-prev" aria-label="上一张" onClick={() => setIndex((value) => (value - 1 + visible.length) % visible.length)}>
                        <ChevronLeft className="size-4" />

/**
 * 按浏览器本地记录判断是否应该展示。
 * visit = 每次都展示；session = 每个会话一次；daily = 每天一次；24h = 距上次至少 24 小时。
 */
export function canShowPromo(storagePrefix: string, item: PromoItem) {
    if (item.frequency === "visit") return true;
    if (item.frequency === "session") return sessionStorage.getItem(`${storagePrefix}${item.id}:session`) !== "1";
    if (item.frequency === "daily") return localStorage.getItem(`${storagePrefix}${item.id}:day`) !== new Date().toDateString();
    const last = Number(localStorage.getItem(`${storagePrefix}${item.id}:time`) ?? 0);
    return !last || Date.now() - last >= 24 * 60 * 60 * 1000;
}

export function markPromoShown(storagePrefix: string, item: PromoItem) {
    if (item.frequency === "visit") return;
    if (item.frequency === "session") {
        sessionStorage.setItem(`${storagePrefix}${item.id}:session`, "1");
        return;
    }
    if (item.frequency === "daily") {
        localStorage.setItem(`${storagePrefix}${item.id}:day`, new Date().toDateString());
        return;
    }
    localStorage.setItem(`${storagePrefix}${item.id}:time`, String(Date.now()));
}

import { useEffect, useState } from "react";
import { Modal } from "antd";
import { ArrowRight, X } from "lucide-react";

import { trackPromoClick, type PromoItem } from "@/services/api/promo";
import { canShowPromo, markPromoShown } from "./promo-frequency";
import { useCountdown } from "./use-countdown";
import { usePromoAction } from "./use-promo-action";
import "./promo.css";

const STORAGE_PREFIX = "sundas:promo:popup:";

function pad(value: number) {
    return String(value).padStart(2, "0");
}

export function PromoPopup({ item }: { item: PromoItem | null }) {
    const action = usePromoAction();
    const [open, setOpen] = useState(false);
    const [imageFailed, setImageFailed] = useState(false);
    const countdown = useCountdown(open ? (item?.countdownAt ?? null) : null);
    const lines = (item?.description ?? "")
        .split(/\r?\n/)
        .map((line) => line.trim())
        .filter(Boolean);
    const clickable = Boolean(item && item.ctaLabel.trim() && item.actionType !== "none");
    const hasBody = lines.length > 0 || clickable;

    useEffect(() => {
        setOpen(false);
        setImageFailed(false);
        if (!item || !canShowPromo(STORAGE_PREFIX, item)) return;
        if (item.countdownAt && new Date(item.countdownAt).getTime() <= Date.now()) return;
        const timer = window.setTimeout(() => {
            markPromoShown(STORAGE_PREFIX, item);
            setOpen(true);
        }, 1200);
        return () => window.clearTimeout(timer);
    }, [item]);

    useEffect(() => {
        if (countdown?.expired) setOpen(false);
    }, [countdown?.expired]);

    if (!item) return null;

    const showImage = Boolean(item.imageUrl) && !imageFailed;
    const showCard = showImage || hasBody;

    const run = () => {
        void trackPromoClick({ placement: "popup", itemId: item.id });
        action(item);
        setOpen(false);
    };

    return (
        <Modal
            open={open && showCard}
            onCancel={() => setOpen(false)}
            footer={null}
            centered
            width={520}
            closable={false}
            className="promo-popup-modal"
            styles={{
                container: { padding: 0, background: "transparent", boxShadow: "none" },
                body: { padding: 0 },
                mask: { background: "rgba(8, 7, 6, 0.66)", backdropFilter: "blur(6px)" },
            }}
        >
            <div className={`promo-popup-card${showImage ? "" : " is-text-only"}`}>
                <button type="button" className="promo-popup-close" onClick={() => setOpen(false)} aria-label="关闭">
                    <X className="size-4" />
                </button>
                {showImage ? (
                    <div className={`promo-popup-hero${item.mobileImageUrl ? " has-mobile-art" : ""}`}>
                        <picture>
                            {item.mobileImageUrl ? <source media="(max-width: 760px)" srcSet={item.mobileImageUrl} /> : null}
                            <img
                                src={item.imageUrl}
                                alt={item.alt || item.title}
                                onError={() => {
                                    setImageFailed(true);
                                    if (!hasBody) setOpen(false);
                                }}
                            />
                        </picture>
                    </div>
                ) : null}
                {hasBody ? (
                    <div className="promo-popup-body">
                        {countdown && !countdown.expired ? (
                            <div className="promo-popup-timer" aria-label="活动倒计时">
                                <span>距结束</span>
                                <b>{pad(countdown.days)}</b>
                                <i>:</i>
                                <b>{pad(countdown.hours)}</b>
                                <i>:</i>
                                <b>{pad(countdown.minutes)}</b>
                                <i>:</i>
                                <b>{pad(countdown.seconds)}</b>
                            </div>
                        ) : null}
                        {item.title ? <h2 className="promo-popup-title">{item.title}</h2> : null}
                        {lines.length > 1 ? (
                            <ul className="promo-popup-points">
                                {lines.map((line) => (
                                    <li key={line}>
                                        <span className="promo-popup-point-mark" aria-hidden />
                                        {line}
                                    </li>
                                ))}
                            </ul>
                        ) : lines.length === 1 ? (
                            <p className="promo-popup-description">{lines[0]}</p>
                        ) : null}
                        {clickable ? (
                            <button type="button" className="promo-popup-cta" onClick={run}>
                                <span>{item.ctaLabel}</span>
                                <ArrowRight className="size-4" />
                            </button>
                        ) : null}
                    </div>
                ) : null}
            </div>
        </Modal>
    );
}

import { useEffect, useState } from "react";

export function PromoTopBar({ item }: { item: PromoItem | null }) {
    const action = usePromoAction();
    const [closed, setClosed] = useState(false);
    const countdown = useCountdown(item?.countdownAt ?? null);

    useEffect(() => {
        if (!item) return;
        setClosed(sessionStorage.getItem(`sundas:promo:topbar:${item.id}`) === "1");
    }, [item]);

    if (!item || closed || countdown?.expired) return null;

    const clickable = item.actionType !== "none";
    const run = () => {
        if (!clickable) return;
        void trackPromoClick({ placement: "topbar", itemId: item.id });
        action(item);
    };

    return (
        <div className={clickable ? "promo-topbar is-clickable" : "promo-topbar"}>
            <div
                className="promo-topbar-inner"
                role={clickable ? "link" : undefined}
                tabIndex={clickable ? 0 : undefined}
                onClick={run}
                onKeyDown={(event) => {
                    if (!clickable) return;
                    if (event.key === "Enter" || event.key === " ") {
                        event.preventDefault();
                        run();
                    }
                }}
            >
                {countdown ? (
                    <span className="promo-topbar-countdown" aria-label="活动倒计时">
                        <b key={`d-${countdown.days}`}>{String(countdown.days).padStart(2, "0")}</b><em>:</em>
                        <b key={`h-${countdown.hours}`}>{String(countdown.hours).padStart(2, "0")}</b><em>:</em>
                        <b key={`m-${countdown.minutes}`}>{String(countdown.minutes).padStart(2, "0")}</b><em>:</em>
                        <b key={`s-${countdown.seconds}`}>{String(countdown.seconds).padStart(2, "0")}</b>
                    </span>
                ) : null}
                <span className="promo-topbar-text">{item.title}</span>
                {item.ctaLabel && clickable ? <span className="promo-topbar-cta">{item.ctaLabel}</span> : null}
            </div>
            <button
                type="button"
                className="promo-topbar-close"
                aria-label="关闭活动条"
                onClick={(event) => {
                    event.stopPropagation();
                    sessionStorage.setItem(`sundas:promo:topbar:${item.id}`, "1");
                    setClosed(true);
                }}
            >
                <X className="size-3.5" />
            </button>
        </div>
    );
}

import { useEffect, useMemo, useState } from "react";

/** Live countdown for promo campaigns; returns null when no deadline is configured. */
export function useCountdown(value: string | null) {
    const end = useMemo(() => (value ? new Date(value).getTime() : null), [value]);
    const [now, setNow] = useState(() => Date.now());

    useEffect(() => {
        if (!end) return;
        setNow(Date.now());
        const timer = window.setInterval(() => setNow(Date.now()), 1000);
        return () => window.clearInterval(timer);
    }, [end]);

    if (!end) return null;
    const remaining = Math.max(0, end - now);
    return {
        expired: remaining <= 0,
        days: Math.floor(remaining / 86_400_000),
        hours: Math.floor((remaining % 86_400_000) / 3_600_000),
        minutes: Math.floor((remaining % 3_600_000) / 60_000),
        seconds: Math.floor((remaining % 60_000) / 1000),
    };
}

import copy from "copy-to-clipboard";
import { App } from "antd";
import { useNavigate } from "react-router-dom";
import { useTranslation } from "react-i18next";

import { usePromoContactStore } from "@/stores/use-promo-contact-store";
import type { PromoItem } from "@/services/api/promo";

export function usePromoAction() {
    const navigate = useNavigate();
    const { message } = App.useApp();
    const { t } = useTranslation();
    const openContact = usePromoContactStore((state) => state.openDialog);

    return (item: Pick<PromoItem, "actionType" | "actionValue">) => {
        if (item.actionType === "none") return;
        if (item.actionType === "internal") {
            navigate(item.actionValue);
            return;
        }
        if (item.actionType === "external") {
            window.open(item.actionValue, "_blank", "noopener,noreferrer");
            return;
        }
        if (item.actionType === "contact") {
            openContact();
            return;
        }
        copy(item.actionValue);
        message.success(t("common.copied"));
    };
}

import copy from "copy-to-clipboard";
import { App } from "antd";

import { trackPromoClick, type PromoContact } from "@/services/api/promo";

export function usePromoContactAction() {
    const { message } = App.useApp();

    return (contact: PromoContact) => {
        void trackPromoClick({ placement: "contact", contactId: contact.id });
        if (!contact.value) return;
        if (contact.channel === "email") {
            window.location.href = `mailto:${contact.value}`;
            return;
        }
        if (contact.channel === "phone") {
            window.location.href = `tel:${contact.value.replace(/[^\d+]/g, "")}`;
            return;
        }
        if (contact.channel === "link") {
            window.open(contact.value, "_blank", "noopener,noreferrer");
            return;
        }
        if (contact.channel === "whatsapp" && !contact.value.startsWith("http")) {
            window.open(`https://wa.me/${contact.value.replace(/\D/g, "")}`, "_blank", "noopener,noreferrer");
            return;
        }

import { useEffect, useRef, useState } from "react";

/**
 * 视频封面：默认显示视频首帧（#t=0.5 避免黑帧），hover 时静音自动播放，移开后暂停并回到封面帧。
 * 用于灵感大厅等卡片场景，无需额外点击即可预览。
 */
export function PromptVideoCover({ src, poster, alt, className = "" }: { src: string; poster?: string; alt: string; className?: string }) {
    const ref = useRef<HTMLVideoElement>(null);
    const hovered = useRef(false);
    const [loadedSrc, setLoadedSrc] = useState<string>();
    // #t=0.5：让 preload="metadata" 停在前半秒，避免部分视频首帧为黑屏
    const srcWithFrame = src.includes("#") ? src : `${src}#t=0.5`;

    useEffect(() => {
        const video = ref.current;
        if (!video) return;
        const observer = new IntersectionObserver(entries => {
            if (entries.some(entry => entry.isIntersecting)) {
                setLoadedSrc(src);
                observer.disconnect();
            }
        }, { rootMargin: "200px" });
        observer.observe(video);
        return () => {
            observer.disconnect();
            hovered.current = false;
            video.pause();
        };
    }, [src]);

    const start = () => {
        const video = ref.current;
        if (!video || !hovered.current) return;
        // 浏览器只允许静音自动播放；这里再强制一次，避免 muted 未生效导致 play() 被拒绝
        video.muted = true;
        void video.play()?.catch(() => undefined);
    };

    return (
        <video
            ref={ref}
            src={loadedSrc === src ? srcWithFrame : undefined}
            poster={poster || undefined}
            muted
            loop
            playsInline
            preload="metadata"
            aria-label={alt}
            className={className}
            onMouseEnter={() => { hovered.current = true; setLoadedSrc(src); start(); }}
            onLoadedData={start}
            onPlay={() => { if (!hovered.current) ref.current?.pause(); }}
            onMouseLeave={() => {
                hovered.current = false;
                const video = ref.current;
                if (!video) return;
                video.pause();
                try {
                    video.currentTime = 0.5;
                } catch {
                    /* metadata 未就绪时忽略 */
                }
            }}
        />
    );
}

import { useEffect, useMemo, useState } from "react";
import { useInfiniteQuery } from "@tanstack/react-query";

import { ALL_PROMPTS_OPTION, fetchPrompts } from "@/services/api/prompts";

export const PROMPT_PAGE_SIZE = 20;

export function usePromptList({ keyword, tags = [], category = ALL_PROMPTS_OPTION, sourceId, metadata = true, enabled = true }: { keyword: string; tags?: string[]; category?: string; sourceId?: string; metadata?: boolean; enabled?: boolean }) {
    const [debouncedKeyword, setDebouncedKeyword] = useState(keyword);
    useEffect(() => {
        const timer = setTimeout(() => setDebouncedKeyword(keyword), 300);
        return () => clearTimeout(timer);
    }, [keyword]);
    const query = useInfiniteQuery({
        queryKey: ["prompts", debouncedKeyword, [...tags].sort(), category, sourceId, metadata],
        queryFn: ({ pageParam, signal }) => fetchPrompts({ keyword: debouncedKeyword, tag: tags, category, sourceId, page: pageParam, pageSize: PROMPT_PAGE_SIZE, metadata: metadata && pageParam === 1, signal }),
        initialPageParam: 1,
        getNextPageParam: (lastPage) => (lastPage.hasMore && lastPage.items.length ? lastPage.page + 1 : undefined),
        enabled: enabled && keyword.trim() === debouncedKeyword.trim(),
    });
    const firstPage = query.data?.pages[0];
    return {
        query,
        items: useMemo(() => query.data?.pages.flatMap((page) => page.items) || [], [query.data?.pages]),
        tags: useMemo(() => [ALL_PROMPTS_OPTION, ...(firstPage?.tags || [])], [firstPage?.tags]),
        categories: useMemo(() => [ALL_PROMPTS_OPTION, ...(firstPage?.categories || [])], [firstPage?.categories]),
        total: firstPage?.total || 0,
    };
}
