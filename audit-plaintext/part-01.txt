import type { DirectorKeyframe, DirectorObject, DirectorVec3 } from "../web/src/types/director";
export type DirectorActorPathKind = "line" | "ring" | "rectangle";

export function buildDirectorActorPath(object: DirectorObject, kind: DirectorActorPathKind, duration: number, createId: () => string, startTime = 0): DirectorObject {
    const seconds = Number.isFinite(duration) && duration > 0 ? duration : 10;
    const [x, y, z] = object.transform.position;
    const positions: DirectorVec3[] = kind === "line"
        ? [[x, y, z], [x, y, z + 4]]
        : kind === "ring"
            ? Array.from({ length: 33 }, (_, index) => {
                const angle = index * Math.PI / 16;
                return [x - 2 + 2 * Math.cos(angle), y, z + 2 * Math.sin(angle)];
            })
            : [[x, y, z], [x + 4, y, z], [x + 4, y, z + 3], [x, y, z + 3], [x, y, z]];
    const center: DirectorVec3 = kind === "line" ? [x, y, z + 2] : kind === "ring" ? [x - 2, y, z] : [x + 2, y, z + 1.5];
    const keyframes: DirectorKeyframe[] = positions.map((position, index) => ({
        id: createId(),
        time: startTime + (index === positions.length - 1 ? seconds : seconds * index / (positions.length - 1)),
        transform: { ...object.transform, position },
        easing: kind === "ring" ? "smooth" : "linear",
    }));
    const withPath: DirectorObject = {
        ...object,
        keyframes,
        motionPath: { kind, duration: seconds, ...(startTime ? { startTime } : {}), facePath: true, transform: { position: center, rotation: [0, 0, 0], scale: [1, 1, 1] }, originalKeyframes: object.motionPath?.originalKeyframes || object.keyframes, ...(kind === "ring" ? { controlKeyframeIds: [keyframes[0].id, keyframes[16].id, keyframes[32].id] } : {}) },
    };
    return setDirectorActorPathFacing(withPath, true);
}


export function setDirectorActorPathFacing(object: DirectorObject, facing: boolean): DirectorObject {
    if (!object.motionPath) return object;
    const keyframes = object.keyframes.map((frame, index, all) => {
        const from = all[index === all.length - 1 && object.motionPath?.kind === "ring" ? 0 : index].transform.position;
        const to = all[index === all.length - 1 ? index === 0 ? 0 : index - 1 : index + 1].transform.position;
        const dx = index === all.length - 1 ? from[0] - to[0] : to[0] - from[0];
        const dz = index === all.length - 1 ? from[2] - to[2] : to[2] - from[2];
        const yaw = Math.hypot(dx, dz) > 1e-6 ? Math.atan2(dx, dz) : object.transform.rotation[1];
        return { ...frame, transform: { ...frame.transform, rotation: facing ? [object.transform.rotation[0], yaw, object.transform.rotation[2]] as DirectorVec3 : [...object.transform.rotation] as DirectorVec3 } };
    });
    return { ...object, motionPath: { ...object.motionPath, facePath: facing }, keyframes };
}

import type { DirectorScene, DirectorVec3, DirectorPose } from '../web/src/types/director';
import { clampDirectorSceneToGround } from './director-ground-clamp';
import { buildDirectorMotionPresetKeyframes, type DirectorMotionPresetKind } from './director-motion-presets';
import { buildDirectorActorPath, setDirectorActorPathFacing, type DirectorActorPathKind } from './director-actor-path';
import { editDirectorSequence, validateDirectorSceneSequence, type DirectorSequenceCommand, type DirectorSequence } from './director-sequence';

export type DirectorCommand =
    | { type: 'sequence.create'; sequence: DirectorSequence }
    | { type: 'sequence.edit'; command: DirectorSequenceCommand }
    | { type: 'camera.clone'; sourceId: string; id: string; name: string }
    | { type: 'camera.preset'; cameraId: string; id: string; preset: DirectorMotionPresetKind; startTime: number; endTime: number }
    | { type: 'actor.frame'; objectId: string; id: string; time: number; position: DirectorVec3; rotation: DirectorVec3 }
    | { type: 'actor.pose'; objectId: string; pose: DirectorPose }
    | { type: 'actor.path'; objectId: string; id: string; kind: DirectorActorPathKind; startTime: number; endTime: number; facePath: boolean }
    | { type: 'camera.frame'; cameraId: string; id: string; time: number; position: DirectorVec3; target: DirectorVec3; fov: number };

/** One validated batch on a clone; callers commit via project CAS. */
export function applyDirectorCommands(scene: DirectorScene, commands: DirectorCommand[]): DirectorScene {
    const next = structuredClone(scene);
    const objectIds = new Set<string>();
    const cameraIds = new Set<string>();
    validateDirectorSceneSequence(next);
    for (const command of commands) {
        if (command.type === 'sequence.create') {
            if (next.cameraSequence) throw new Error('场景已有序列，请编辑现有切点');
            next.cameraSequence = structuredClone(command.sequence);
        } else if (command.type === 'sequence.edit') {
            if (!next.cameraSequence) throw new Error('场景尚未建立序列');
            next.cameraSequence = editDirectorSequence(next.cameraSequence, command.command, next.cameras.map(camera => camera.id));
        } else if (command.type === 'camera.clone') {
            const source = next.cameras.find(camera => camera.id === command.sourceId);
            if (!source || source.locked) throw new Error('源机位不存在或已锁定');
            if (!command.id || !command.name || next.cameras.some(camera => camera.id === command.id)) throw new Error('新机位标识无效或重复');
            next.cameras.push({ ...structuredClone(source), id: command.id, name: command.name });
            cameraIds.add(command.id);
        } else if (command.type === 'camera.preset') {
            const camera = next.cameras.find(camera => camera.id === command.cameraId);
            if (!camera || camera.locked) throw new Error('机位不存在或已锁定');
            if (!command.id || !Number.isFinite(command.startTime) || !Number.isFinite(command.endTime) || command.startTime < 0 || command.endTime <= command.startTime) throw new Error('运镜必须指定有效的全局起止时间');
            if (camera.keyframes.some(frame => frame.time > command.startTime && frame.time <= command.endTime)) throw new Error('运镜范围内已有关键帧，请先在导演台处理，或复制机位后重新安排');
            const start = camera.keyframes.find(frame => frame.time === command.startTime);
            if (camera.keyframes.length && !start) throw new Error('请先明确起始时间的机位关键帧，再应用运镜预设');
            let index = 0;
            const generated = buildDirectorMotionPresetKeyframes(command.preset, 'replace', { ...camera, transform: start?.transform || camera.transform, target: start?.target || camera.target, fov: start?.fov ?? camera.fov }, command.endTime, () => `${command.id}:${index++}`).keyframes.map(frame => ({ ...frame, time: command.startTime + frame.time / 2 * (command.endTime - command.startTime), positionKeyed: true, rotationKeyed: false }));
            if (generated.some(frame => camera.keyframes.some(old => old.id === frame.id))) throw new Error('运镜关键帧标识重复，请重新核对方案');
            camera.keyframes = [...camera.keyframes, ...generated.filter((_, index) => !start || index !== 0)].sort((a,b) => a.time-b.time);
            camera.lookAtMode = 'coordinates';
            delete camera.lookAtObjectId;
            cameraIds.add(camera.id);
        } else if (command.type === 'actor.frame' || command.type === 'actor.pose' || command.type === 'actor.path') {
            const actor = next.objects.find(object => object.id === command.objectId);
            if (!actor || actor.locked || !(actor.kind === 'actor' || actor.primitive === 'character' || actor.kind === 'model')) throw new Error('人物不存在、已锁定或类型不支持');
            if (command.type === 'actor.path') {
                if (!command.id || !Number.isFinite(command.startTime) || !Number.isFinite(command.endTime) || command.startTime < 0 || command.endTime <= command.startTime) throw new Error('人物路径必须指定有效的全局起止时间');
                let index = 0;
                const path = setDirectorActorPathFacing(buildDirectorActorPath(actor, command.kind, command.endTime - command.startTime, () => `${command.id}:${index++}`, command.startTime), command.facePath);
                if (path.keyframes.some(frame => actor.keyframes.some(old => old.id === frame.id))) throw new Error('路径关键帧标识重复，请重新核对方案');
                Object.assign(actor, path, { animationTrackEnabled: true });
                objectIds.add(actor.id);
            } else if (command.type === 'actor.pose') actor.pose = command.pose;
            else {
                if (actor.motionPath) throw new Error('人物已有运动路径，请先在导演台确认移除路径后再安排关键帧');
                if (!Number.isFinite(command.time) || command.time < 0 || ![command.position, command.rotation].every(vector => vector.length === 3 && vector.every(Number.isFinite))) throw new Error('人物关键帧参数无效');
                if (actor.keyframes.some(frame => frame.id !== command.id && frame.time === command.time)) throw new Error('该时刻已有其他人物关键帧');
                const existing = actor.keyframes.find(frame => frame.id === command.id);
                actor.keyframes = [...actor.keyframes.filter(frame => frame.id !== command.id), { ...existing, id: command.id, time: command.time, transform: { ...structuredClone(existing?.transform || actor.transform), position: command.position, rotation: command.rotation }, positionKeyed: true, rotationKeyed: true, scaleKeyed: existing?.scaleKeyed ?? false }].sort((a, b) => a.time - b.time);
                actor.animationTrackEnabled = true;
                objectIds.add(actor.id);
            }
        } else {
            const camera = next.cameras.find(camera => camera.id === command.cameraId);
            if (!camera || camera.locked) throw new Error('机位不存在或已锁定');
            if (!Number.isFinite(command.time) || command.time < 0 || !Number.isFinite(command.fov) || command.fov <= 0 || command.fov >= 180
                || ![command.position, command.target].every(vector => vector.length === 3 && vector.every(Number.isFinite))) throw new Error('机位关键帧参数无效');
            const existing = camera.keyframes.find(frame => frame.id === command.id);
            if (camera.keyframes.some(frame => frame.id !== command.id && frame.time === command.time)) throw new Error('该时刻已有其他关键帧，请明确修改目标');
            const frame = { ...existing, id: command.id, time: command.time, transform: { ...structuredClone(existing?.transform || camera.transform), position: command.position }, target: command.target, fov: command.fov, positionKeyed: true, rotationKeyed: existing?.rotationKeyed ?? false };
            camera.keyframes = [...camera.keyframes.filter(frame => frame.id !== command.id), frame].sort((a, b) => a.time - b.time);
            camera.lookAtMode = 'coordinates';
            delete camera.lookAtObjectId;
            cameraIds.add(camera.id);
        }
        validateDirectorSceneSequence(next);
    }
    return clampDirectorSceneToGround(next, { objectIds, cameraIds });
}

import type { DirectorCamera, DirectorKeyframe, DirectorObject, DirectorScene, DirectorTransform, DirectorVec3 } from "../web/src/types/director";
import { directorGroundSettings } from "./director-ground";


/**
 * 摄影机贴地余量：镜头允许贴近地面取景，但不能钻进地面里穿模。
 * 人物/道具没有余量——脚底、底面正好落在地面高度即视为合法。
 */
export const DIRECTOR_CAMERA_GROUND_CLEARANCE = 0.05;

/** 自由视角（环绕相机）的贴地余量，比机位稍大，避免相机擦着地面出现地平线翻面。 */
export const DIRECTOR_ORBIT_GROUND_CLEARANCE = 0.25;

/**
 * 内置几何体的半高（未乘缩放）。
 *
 * 这些 primitive 的 transform.position 是几何中心而不是底面，直接拿 y 和地面比，
 * 会让一个 1×1×1 的箱子在 y=0 时正好埋掉一半。取值与 director-viewport 里的
 * 几何参数一一对应，改那边记得同步这里。
 */
const DIRECTOR_PRIMITIVE_HALF_HEIGHT: Partial<Record<NonNullable<DirectorObject["primitive"]>, number>> = {
    box: 0.5,
    sphere: 0.6,
    cylinder: 0.6,
    plane: 0.5,
    torus: 0.58,
    cone: 0.6,
    pyramid: 0.6,
    empty: 0,
    character: 0,
};

/**
 * 对象锚点到「物体最低点」的距离，用来把「不许低于地面」翻译成「锚点至少要多高」。
 * 角色与导入模型的枢轴就在脚底/原点（模型 bbox 在场景层不可知），因此偏移为 0。
 */
export function directorObjectGroundBase(object: Pick<DirectorObject, "kind" | "primitive">): number {
    if (object.kind === "actor" || object.kind === "model") return 0;
    return object.primitive ? DIRECTOR_PRIMITIVE_HALF_HEIGHT[object.primitive] ?? 0 : 0;
}

function scaleY(transform: DirectorTransform | undefined): number {
    const value = Math.abs(transform?.scale?.[1] ?? 1);
    return Number.isFinite(value) && value > 0 ? value : 1;
}

/** 对象（含当前缩放）的锚点下限。 */
export function directorObjectGroundFloor(object: Pick<DirectorObject, "kind" | "primitive" | "transform">, groundHeight: number): number {
    return groundHeight + directorObjectGroundBase(object) * scaleY(object.transform);
}

/** 只抬不压：低于地面的 Y 直接抬到地面，其余原样返回。 */
export function clampDirectorGroundY(y: number, floor: number): number {
    if (!Number.isFinite(y)) return floor;
    return y < floor ? floor : y;
}

/**
 * 把 transform 的 Y 抬到给定下限以上；X/Z、旋转、缩放完全不动。
 * 无变化时返回原对象引用，让上层可以靠引用相等跳过无意义的场景写入。
 */
export function clampDirectorTransformToGround(transform: DirectorTransform, floor: number): DirectorTransform {
    const y = transform?.position?.[1];
    if (!Number.isFinite(y) || y >= floor) return transform;
    return { ...transform, position: [transform.position[0], floor, transform.position[2]] as DirectorVec3 };
}

/** 关键帧按各自那一帧的缩放算下限：放大后的箱子比原尺寸更需要抬高。 */
function clampKeyframesToGround(keyframes: DirectorKeyframe[], floorFor: (transform: DirectorTransform) => number): DirectorKeyframe[];
function clampKeyframesToGround(keyframes: DirectorKeyframe[] | undefined, floorFor: (transform: DirectorTransform) => number): DirectorKeyframe[] | undefined;
function clampKeyframesToGround(keyframes: DirectorKeyframe[] | undefined, floorFor: (transform: DirectorTransform) => number): DirectorKeyframe[] | undefined {
    if (!keyframes?.length) return keyframes;
    let changed = false;
    const next = keyframes.map((frame) => {
        const transform = clampDirectorTransformToGround(frame.transform, floorFor(frame.transform));
        if (transform === frame.transform) return frame;
        changed = true;
        return { ...frame, transform };
    });
    return changed ? next : keyframes;
}

function clampObjectToGround(object: DirectorObject, groundHeight: number): DirectorObject {
    const transform = clampDirectorTransformToGround(object.transform, directorObjectGroundFloor(object, groundHeight));
    const floorFor = (frame: DirectorTransform) => groundHeight + directorObjectGroundBase(object) * scaleY(frame);
    const keyframes = clampKeyframesToGround(object.keyframes, floorFor);
    const originalKeyframes = clampKeyframesToGround(object.motionPath?.originalKeyframes, floorFor);
    const motionPath = object.motionPath && originalKeyframes !== object.motionPath.originalKeyframes ? { ...object.motionPath, originalKeyframes: originalKeyframes ?? [] } : object.motionPath;
    if (transform === object.transform && keyframes === object.keyframes && motionPath === object.motionPath) return object;
    return { ...object, transform, keyframes, motionPath };
}

function clampCameraToGround(camera: DirectorCamera, groundHeight: number): DirectorCamera {
    const floor = groundHeight + DIRECTOR_CAMERA_GROUND_CLEARANCE;
    const transform = clampDirectorTransformToGround(camera.transform, floor);
    const keyframes = clampKeyframesToGround(camera.keyframes, () => floor);
    const originalKeyframes = clampKeyframesToGround(camera.drawnPath?.originalKeyframes, () => floor);
    const drawnPath = camera.drawnPath && originalKeyframes !== camera.drawnPath.originalKeyframes ? { ...camera.drawnPath, originalKeyframes: originalKeyframes ?? [] } : camera.drawnPath;
    if (transform === camera.transform && keyframes === camera.keyframes && drawnPath === camera.drawnPath) return camera;
    return { ...camera, transform, keyframes, drawnPath };
}

/**
 * 场景级的「不许钻到地面以下」归一化。
 *
 * 只夹 Y、只抬不压，且严格保持引用：没有任何对象越界时返回原 scene，
 * 因此可以安全地挂在每次提交上而不会制造多余的重渲染或历史记录。
 */
export function clampDirectorSceneToGround(scene: DirectorScene, scope?: { objectIds: ReadonlySet<string>; cameraIds: ReadonlySet<string> }): DirectorScene {
    const groundHeight = directorGroundSettings(scene).height;
    let changed = false;
    const objects = scene.objects.map((object) => {
        if (scope && !scope.objectIds.has(object.id)) return object;
        const next = clampObjectToGround(object, groundHeight);
        if (next !== object) changed = true;
        return next;
    });
    const cameras = scene.cameras.map((camera) => {
        if (scope && !scope.cameraIds.has(camera.id)) return camera;
        const next = clampCameraToGround(camera, groundHeight);
        if (next !== camera) changed = true;
        return next;
    });
    return changed ? { ...scene, objects, cameras } : scene;
}

import type { DirectorScene } from "../web/src/types/director";

export const DIRECTOR_DEFAULT_GROUND = { visible: true, opacity: 0.4, height: 0 } as const;

/** Legacy scenes retain their original opaque floor until the user edits ground settings. */
export function directorGroundSettings(scene: Pick<DirectorScene, "ground">): { visible: boolean; opacity: number; height: number } {
    const ground = scene.ground;
    const opacity = ground?.opacity;
    const height = ground?.height;
    return {
        visible: ground?.visible !== false,
        opacity: Number.isFinite(opacity) ? Math.min(1, Math.max(0, opacity!)) : 1,
        height: Number.isFinite(height) ? Math.min(2, Math.max(-2, height!)) : 0,
    };
}

import type { DirectorScene } from '../web/src/types/director';
import { validateDirectorSceneSequence } from './director-sequence';

export type DirectorChange = { collection?: 'cameras' | 'objects'; id?: string; field?: string; before?: unknown; after?: unknown };
const same = (a: unknown, b: unknown) => JSON.stringify(a) === JSON.stringify(b);
export function directorChanges(before: DirectorScene, after: DirectorScene): DirectorChange[] {
    const changes: DirectorChange[] = [];
    if (!same(before.cameraSequence, after.cameraSequence)) changes.push({ field: 'cameraSequence', before: before.cameraSequence, after: after.cameraSequence });
    for (const collection of ['cameras', 'objects'] as const) {
        for (const item of after[collection]) {
            const original = before[collection].find(entry => entry.id === item.id);
            if (!original) { changes.push({ collection, id: item.id, after: item }); continue; }
            for (const field of new Set([...Object.keys(original), ...Object.keys(item)])) {
                const oldValue = (original as any)[field], newValue = (item as any)[field];
                if (!same(oldValue, newValue)) changes.push({ collection, id: item.id, field, before: oldValue, after: newValue });
            }
        }
    }
    return changes;
}
/** Undo a persisted, server-generated diff, never a model-provided patch. */
export function undoDirectorChanges(scene: DirectorScene, changes: DirectorChange[]): DirectorScene {
    const next = structuredClone(scene);
    for (const change of [...changes].sort((a, b) => Number(Boolean(a.collection)) - Number(Boolean(b.collection)))) {
        const list = change.collection ? next[change.collection] : undefined;
        const target = list ? list.find(item => item.id === change.id) : next;
        if (!target) throw new Error('撤销目标已删除，请手动核对');
        if (!change.field) {
            if (!same(target, change.after)) throw new Error('新增对象已有后续修改，无法自动撤销');
            if (change.collection === 'cameras' && (next.shots.some(shot => shot.cameraId === change.id) || next.cameraSequence?.cuts.some(cut => cut.cameraId === change.id))) throw new Error('新增机位已被其他镜头引用，无法自动撤销');
            list!.splice(list!.findIndex(item => item.id === change.id), 1);
        } else {
            const record = target as unknown as Record<string, unknown>;
            if (!same(record[change.field], change.after)) throw new Error('相关字段已有后续修改，原操作不能自动撤销');
            if (change.before === undefined) delete record[change.field];
            else record[change.field] = structuredClone(change.before);
        }
    }
    validateDirectorSceneSequence(next);
    return next;
}

import type { DirectorCamera, DirectorKeyframe } from "../web/src/types/director";

export type DirectorMotionPresetKind = "orbit" | "half_arc" | "push_in" | "pull_out" | "lift" | "truck" | "spiral";

export function buildDirectorMotionPresetKeyframes(kind: DirectorMotionPresetKind, action: "replace" | "append", camera: DirectorCamera, duration: number, createId: () => string): { keyframes: DirectorKeyframe[]; duration: number } {
    const previous = action === "append" ? [...camera.keyframes].sort((a, b) => a.time - b.time) : [];
    const last = previous.at(-1);
    const start = last?.transform || camera.transform;
    const [x, y, z] = start.position;
    const target = last?.target || camera.target;
    const fov = last?.fov ?? camera.fov;
    const [targetX, targetY, targetZ] = target;
    const forward = [targetX - x, targetY - y, targetZ - z];
    const forwardLength = Math.hypot(...forward) || 1;
    const [forwardX, forwardY, forwardZ] = forward.map((axis) => axis / forwardLength);
    const horizontalLength = Math.hypot(forwardX, forwardZ) || 1;
    const rightX = -forwardZ / horizontalLength;
    const rightZ = forwardX / horizontalLength;
    const startPosition: [number, number, number] = [x, y, z];
    const orbit = (steps: number, radians: number, rise: number): [number, number, number][] => Array.from({ length: steps + 1 }, (_, index) => {
        if (index === 0) return startPosition;
        if (index === steps && radians === 2 * Math.PI && rise === 0) return startPosition;
        const angle = radians * index / steps;
        const radialX = x - targetX;
        const radialZ = z - targetZ;
        return [targetX + radialX * Math.cos(angle) - radialZ * Math.sin(angle), y + rise * index / steps, targetZ + radialX * Math.sin(angle) + radialZ * Math.cos(angle)];
    });
    const positions: [number, number, number][] = kind === "orbit" ? orbit(8, 2 * Math.PI, 0)
        : kind === "half_arc" ? orbit(4, Math.PI, 0)
        : kind === "spiral" ? orbit(8, 2 * Math.PI, 2)
        : kind === "push_in" ? [startPosition, [x + forwardX * 2, y + forwardY * 2, z + forwardZ * 2]]
        : kind === "pull_out" ? [startPosition, [x - forwardX * 2, y - forwardY * 2, z - forwardZ * 2]]
        : kind === "lift" ? [startPosition, [x, y + 2, z]]
        : [startPosition, [x + rightX * 2, y, z + rightZ * 2]];
    const startTime = last?.time || 0;
    const generated = positions.map((position, index) => ({
        id: createId(),
        time: startTime + 2 * index / (positions.length - 1),
        transform: { ...start, position },
        target: [...target] as [number, number, number],
        fov,
        easing: (positions.length > 2 ? "smooth" : "linear") as DirectorKeyframe["easing"],
    }));
    const keyframes = previous.length ? [...previous, ...generated.slice(1)] : generated;
    return { keyframes, duration: Math.max(duration, startTime + 2) };
}

import { defaultShotComposition, type ProductionDocument, type ProductionMedia } from './production';

export type DirectorCutMapping = { shotId: string; cutId: string; title: string; cameraId: string; startFrame: number; endFrame: number; inMs: number; outMs: number };
export const directorProductionShotId = (sceneId: string, sequenceId: string, cutId: string) => `director:${[sceneId, sequenceId, cutId].map(encodeURIComponent).join(':')}`;
export function directorProductionMapping(scene: { id: string; cameras: Array<{ id: string; name: string }>; cameraSequence?: DirectorSequence }, parameters: { sequenceId: string; kind: string; startFrame: number; endFrame: number; fixedCameraId?: string }, durationMs: number): DirectorCutMapping[] {
    const sequence = scene.cameraSequence;
    if (!sequence || sequence.id !== parameters.sequenceId || parameters.kind !== 'video' || parameters.fixedCameraId) throw new Error('请选择按序列录制的参考视频');
    validateDirectorSequence(sequence, scene.cameras.map(camera => camera.id));
    if (!Number.isSafeInteger(parameters.startFrame) || !Number.isSafeInteger(parameters.endFrame) || parameters.startFrame < 0 || parameters.endFrame > sequence.durationFrames || parameters.endFrame <= parameters.startFrame) throw new Error('预演视频帧范围无效');
    const requiredMs = (parameters.endFrame - parameters.startFrame) * 1000 / sequence.fps;
    if (!Number.isFinite(durationMs) || durationMs < requiredMs) throw new Error('实际视频时长不足以覆盖切镜范围，请重新录制后采用');
    return sequence.cuts.flatMap((cut, index) => {
        const startFrame = Math.max(cut.startFrame, parameters.startFrame);
        const endFrame = Math.min(sequence.cuts[index + 1]?.startFrame ?? sequence.durationFrames, parameters.endFrame);
        if (startFrame >= endFrame) return [];
        return [{ shotId: directorProductionShotId(scene.id, sequence.id, cut.id), cutId: cut.id,
            title: `${scene.cameras.find(camera => camera.id === cut.cameraId)!.name} · 第 ${startFrame}–${endFrame} 帧`, cameraId: cut.cameraId,
            startFrame, endFrame, inMs: (startFrame - parameters.startFrame) * 1000 / sequence.fps, outMs: (endFrame - parameters.startFrame) * 1000 / sequence.fps }];
    });
}

/** Stable cut identity retains edited titles/order and unrelated shots across recordings. */
export function adoptDirectorCuts(source: ProductionDocument, mapping: DirectorCutMapping[], media: ProductionMedia): ProductionDocument {
    if (media.kind !== 'video') throw new Error('切镜采用需要视频素材');
    const next = structuredClone(source);
    for (const cut of mapping) {
        let shot = next.shots.find(shot => shot.shotId === cut.shotId);
        if (!shot) {
            shot = { shotId: cut.shotId, title: cut.title, description: '', action: '', prompt: '', references: [], revision: 1 };
            next.shots.push(shot);
        }
        shot.adoptedMedia = { ...media };
        shot.mediaRevision = shot.revision || 1;
        next.composition.shots[cut.shotId] = { ...defaultShotComposition(), ...next.composition.shots[cut.shotId], inMs: cut.inMs, outMs: cut.outMs, durationMs: cut.outMs - cut.inMs };
    }
    return next;
}

import type { DirectorScene } from '../web/src/types/director';

/** Frame-based camera cuts. All animation continues on the shared scene clock. */
export type DirectorSequence = {
    schemaVersion: 1;
    id: string;
    name: string;
    fps: 24 | 25 | 30;
    durationFrames: number;
    cuts: Array<{ id: string; startFrame: number; cameraId: string }>;
};

export function validateDirectorSequence(value: unknown, cameraIds: readonly string[]): asserts value is DirectorSequence {
    const sequence = value as DirectorSequence | null;
    if (!sequence || sequence.schemaVersion !== 1 || typeof sequence.id !== 'string' || !sequence.id || typeof sequence.name !== 'string'
        || ![24, 25, 30].includes(sequence.fps) || !Number.isSafeInteger(sequence.durationFrames) || sequence.durationFrames <= 0
        || !Array.isArray(sequence.cuts) || !sequence.cuts.length) throw new Error('切镜序列格式或版本无效');
    const ids = new Set<string>();
    let previous = -1;
    for (const cut of sequence.cuts) {
        if (!cut || typeof cut.id !== 'string' || !cut.id || ids.has(cut.id) || !cameraIds.includes(cut.cameraId)
            || !Number.isSafeInteger(cut.startFrame) || cut.startFrame <= previous || cut.startFrame >= sequence.durationFrames
            || (previous === -1 && cut.startFrame !== 0)) throw new Error('切点必须从第 0 帧开始、严格递增，并引用有效机位');
        ids.add(cut.id);
        previous = cut.startFrame;
    }
}

/** Call after validation when loading/changing the sequence, not on every animation tick. */
export function sampleDirectorSequence(sequence: DirectorSequence, frame: number, fixedCameraId?: string) {
    if (!Number.isFinite(frame)) throw new Error('预演帧号无效');
    const resolvedFrame = Math.max(0, Math.min(sequence.durationFrames - 1, Math.floor(frame)));
    let cut = sequence.cuts[0];
    for (const next of sequence.cuts) {
        if (next.startFrame > resolvedFrame) break;
        cut = next;
    }
    return { frame: resolvedFrame, timeSeconds: resolvedFrame / sequence.fps, cameraId: fixedCameraId ?? cut.cameraId, cutId: cut.id };
}

/** A shot list describes saved cuts, not rendered media or generated footage. */
export function directorShotList(scene: { id: string; title: string; cameras: Array<{ id: string; name: string }>; cameraSequence?: DirectorSequence }, source: { projectId: string; projectVersion: number }) {
    if (!scene.cameraSequence) throw new Error('场景尚未建立切镜序列');
    validateDirectorSequence(scene.cameraSequence, scene.cameras.map(camera => camera.id));
    if (!source.projectId || !Number.isSafeInteger(source.projectVersion) || source.projectVersion < 1) throw new Error('镜头单需要已保存的项目版本');
    const sequence = scene.cameraSequence;
    return {
        schemaVersion: 1, ...source, sceneId: scene.id, sceneTitle: scene.title,
        sequenceId: sequence.id, sequenceName: sequence.name, fps: sequence.fps,
        durationFrames: sequence.durationFrames, durationSeconds: sequence.durationFrames / sequence.fps,
        interval: '[startFrame,endFrame)',
        cuts: sequence.cuts.map((cut, index) => {
            const endFrame = sequence.cuts[index + 1]?.startFrame ?? sequence.durationFrames;
            return { ...cut, cameraName: scene.cameras.find(camera => camera.id === cut.cameraId)!.name,
                endFrame, startSeconds: cut.startFrame / sequence.fps, endSeconds: endFrame / sequence.fps };
        }),
    };
}

/** Preserve seconds when changing fps; refuse collisions instead of dropping cuts. */
export function retimeDirectorSequence(sequence: DirectorSequence, fps: DirectorSequence['fps'], cameraIds: readonly string[]) {
    const next = { ...sequence, fps, durationFrames: Math.round(sequence.durationFrames * fps / sequence.fps),
        cuts: sequence.cuts.map(cut => ({ ...cut, startFrame: Math.round(cut.startFrame * fps / sequence.fps) })) };
    validateDirectorSequence(next, cameraIds);
    return next;
}

export type DirectorSequenceCommand =
    | { type: 'insert'; cut: DirectorSequence['cuts'][number] }
    | { type: 'update'; id: string; startFrame?: number; cameraId?: string }
    | { type: 'remove'; id: string }
    | { type: 'duration'; durationFrames: number }
    | { type: 'fps'; fps: DirectorSequence['fps'] };

/** Review the actual rounding and sampling impact before committing a timing edit. */
export function previewDirectorSequenceTiming(scene: DirectorScene, command: Extract<DirectorSequenceCommand, { type: 'fps' | 'duration' }>) {
    const sequence = scene.cameraSequence;
    if (!sequence) throw new Error('场景尚未建立序列');
    const fps = command.type === 'fps' ? command.fps : sequence.fps;
    const durationFrames = command.type === 'duration' ? command.durationFrames : Math.round(sequence.durationFrames * fps / sequence.fps);
    const cuts = sequence.cuts.map(cut => ({ ...cut, previousFrame: cut.startFrame,
        startFrame: command.type === 'fps' ? Math.round(cut.startFrame * fps / sequence.fps) : cut.startFrame }));
    const lines = [`${sequence.fps}fps / ${sequence.durationFrames} 帧 → ${fps}fps / ${durationFrames} 帧`];
    for (const cut of cuts) {
        const name = scene.cameras.find(camera => camera.id === cut.cameraId)?.name || cut.cameraId;
        lines.push(`${name}：第 ${cut.previousFrame} 帧 → 第 ${cut.startFrame} 帧（${(cut.previousFrame / sequence.fps).toFixed(3)} 秒 → ${(cut.startFrame / fps).toFixed(3)} 秒）${cut.startFrame >= durationFrames ? '，超出新范围' : ''}`);
    }
    const lastTime = (durationFrames - 1) / fps;
    if (lastTime < (sequence.durationFrames - 1) / sequence.fps) {
        for (const item of [...scene.objects, ...scene.cameras]) {
            const times = item.keyframes.map(frame => frame.time);
            if ('boneTracks' in item) times.push(...(item.boneTracks || []).flatMap(track => track.keyframes.map(frame => frame.time)));
            if ('motionClips' in item) times.push(...(item.motionClips || []).map(clip => clip.start + clip.duration));
            if ('motionPath' in item && item.motionPath) times.push((item.motionPath.startTime || 0) + item.motionPath.duration);
            if (times.some(time => time > lastTime)) lines.push(`${item.name}：部分动作或关键帧位于新的末帧时间 ${lastTime.toFixed(3)} 秒之后；原数据保留，但本次预演不会播放到该处。`);
        }
    }
    try { return { lines, next: editDirectorSequence(sequence, command, scene.cameras.map(camera => camera.id)), error: '' }; }
    catch (cause) { return { lines, next: null, error: cause instanceof Error ? cause.message : '序列调整无效' }; }
}

/** Pure atomic edit: validation failures never mutate the user's original sequence. */
export function editDirectorSequence(sequence: DirectorSequence, command: DirectorSequenceCommand, cameraIds: readonly string[]): DirectorSequence {
    validateDirectorSequence(sequence, cameraIds);
    if (command.type === 'fps') return retimeDirectorSequence(sequence, command.fps, cameraIds);
    const next = structuredClone(sequence);
    if (command.type === 'duration') next.durationFrames = command.durationFrames;
    else if (command.type === 'insert') next.cuts.push({ ...command.cut });
    else {
        const index = next.cuts.findIndex(cut => cut.id === command.id);
        if (index < 0) throw new Error('切点不存在，请重新读取场景');
        if (command.type === 'remove') {
            if (index === 0) throw new Error('第 0 帧切点必须保留，可替换其机位');
            next.cuts.splice(index, 1);
        } else {
            if (index === 0 && command.startFrame !== undefined && command.startFrame !== 0) throw new Error('首个切点必须位于第 0 帧');
            if (command.startFrame !== undefined) next.cuts[index].startFrame = command.startFrame;
            if (command.cameraId !== undefined) next.cuts[index].cameraId = command.cameraId;
        }
    }
    next.cuts.sort((a, b) => a.startFrame - b.startFrame);
    validateDirectorSequence(next, cameraIds);
    return next;
}

export function validateDirectorSceneSequence(scene: { cameraSequence?: unknown; cameras: Array<{ id: string }> }): void {
    if (scene.cameraSequence !== undefined) validateDirectorSequence(scene.cameraSequence, scene.cameras.map(camera => camera.id));
}

/** The capability marks an intentional edit/delete by a sequence-aware client. */
export function validateDirectorSequenceWrite(previous: unknown, next: unknown, capability?: number): void {
    const scenes = (content: unknown): Array<{ id: string; cameraSequence?: unknown; cameras: Array<{ id: string }> }> => {
        if (!content || typeof content !== 'object') return [];
        const value = (content as { directorScenes?: unknown }).directorScenes;
        if (value === undefined) return [];
        if (!Array.isArray(value)) throw new Error('导演场景列表格式无效');
        return value;
    };
    const incoming = scenes(next);
    for (const scene of incoming) {
        if (!scene || typeof scene !== 'object') throw new Error('导演场景格式无效');
        validateDirectorSceneSequence(scene);
    }
    for (const old of scenes(previous)) {
        if (old?.cameraSequence === undefined) continue;
        validateDirectorSceneSequence(old);
        const replacement = incoming.find(scene => scene.id === old.id);
        if (capability !== 1 && JSON.stringify(replacement?.cameraSequence) !== JSON.stringify(old.cameraSequence)) {
            throw new Error('当前客户端未声明切镜序列支持，请刷新页面后再保存；云端序列已保留');
        }
    }
}

type Change = { path: string[]; before?: unknown; after?: unknown };
const equal = (a: unknown, b: unknown) => JSON.stringify(a) === JSON.stringify(b);
const object = (value: unknown): value is Record<string, unknown> => !!value && typeof value === 'object' && !Array.isArray(value);
const scope = (p: ProductionDocument) => ({ composition: p.composition, subtitles: p.subtitles, shotIds: p.shots.map(shot => shot.shotId) });
export function productionChanges(before: ProductionDocument, after: ProductionDocument): Change[] {
    const changes: Change[] = [];
    const diff = (a: unknown, b: unknown, path: string[]) => {
        if (equal(a,b)) return;
        if (object(a) && object(b)) for (const key of new Set([...Object.keys(a), ...Object.keys(b)])) diff(a[key],b[key],[...path,key]);
        else changes.push({ path, before: a, after: b });
    };
    diff(scope(before),scope(after),[]);
    return changes;
}
export function undoProductionChanges(value: unknown, changes: Change[]): ProductionDocument {
    const next = structuredClone(parseProduction(value));
    const editable = scope(next);
    for (const change of changes) {
        let parent: any = editable;
        for (const key of change.path.slice(0,-1)) { parent = parent?.[key]; if (!parent) throw new Error('编排字段已删除，无法自动撤销'); }
        const key = change.path.at(-1)!;
        if (!equal(parent[key],change.after)) throw new Error('编排字段已有后续修改，无法自动撤销');
        if (change.before === undefined) delete parent[key]; else parent[key] = structuredClone(change.before);
    }
    if (editable.shotIds.length !== next.shots.length || editable.shotIds.some(id => !next.shots.some(shot => shot.shotId === id))) throw new Error('镜头列表已变化，无法撤销排序');
    next.shots = editable.shotIds.map(id => next.shots.find(shot => shot.shotId === id)!);
    next.composition = editable.composition; next.subtitles = editable.subtitles;
    return parseProduction(next);
}

export function validateProduction(value: unknown): { valid: boolean; errors: string[] } {

    };

    unique(subtitles, 'id', '字幕');

}

const upsert = <T>(items: T[], next: T, key: (item: T) => string) => items.some(item => key(item) === key(next)) ? items.map(item => key(item) === key(next) ? next : item) : [...items, next];
export function applyProductionOperation(source: ProductionDocument, operation: ProductionOperation): ProductionDocument {
    const p = structuredClone(source);
    switch (operation.type) {
        case 'replace': return parseProduction(structuredClone(operation.production));
        case 'script.edit': p.script = { ...p.script, draft: operation.text, original: operation.original ?? p.script.original, revision: p.script.revision + 1 }; break;
        case 'script.adopt': p.script.adopted = p.script.draft; p.script.adoptedRevision = p.script.revision; break;
        case 'asset.upsert': p.library = upsert(p.library, operation.asset, item => item.id); break;
        case 'asset.remove': if (p.shots.some(shot => shot.references.some(ref => ref.assetId === operation.id))) throw new Error('仍有镜头引用该资料，请先调整引用'); p.library = p.library.filter(item => item.id !== operation.id); break;
        case 'shot.upsert': { const previous = p.shots.find(item => item.shotId === operation.shot.shotId); const changed = previous && ['description', 'action', 'prompt', 'references'].some(key => JSON.stringify(previous[key as keyof ProductionShot]) !== JSON.stringify(operation.shot[key as keyof ProductionShot])); const next = { ...operation.shot, revision: previous ? (previous.revision || 1) + (changed ? 1 : 0) : operation.shot.revision || 1 }; p.shots = upsert(p.shots, next, item => item.shotId); break; }
        case 'shot.remove': p.shots = p.shots.filter(item => item.shotId !== operation.shotId); delete p.composition.shots[operation.shotId]; break;
        case 'shot.reorder': { const byId = new Map(p.shots.map(item => [item.shotId, item])); if (operation.shotIds.length !== p.shots.length || new Set(operation.shotIds).size !== p.shots.length || operation.shotIds.some(key => !byId.has(key))) throw new Error('镜头顺序必须完整且不能重复'); p.shots = operation.shotIds.map(key => byId.get(key)!); break; }
        case 'shot.adopt': { const shot = p.shots.find(item => item.shotId === operation.shotId); if (!shot) throw new Error('镜头已删除，结果仅保留在生产记录'); shot.adoptedMedia = operation.media; shot.mediaRevision = shot.revision || 1; break; }
        case 'narration.upsert': { const previous = p.narration.find(item => item.id === operation.narration.id); const changed = previous && (['text', 'model', 'voice', 'speed'] as const).some(key => previous[key] !== operation.narration[key]); const next = { ...operation.narration, revision: previous ? previous.revision + (changed ? 1 : 0) : operation.narration.revision }; p.narration = upsert(p.narration, next, item => item.id); break; }
        case 'narration.remove': p.narration = p.narration.filter(item => item.id !== operation.id); p.subtitles = p.subtitles.filter(item => item.narrationId !== operation.id); delete p.composition.narration[operation.id]; break;
        case 'subtitles.replace': if (operation.subtitles.some(item => item.narrationId !== operation.narrationId)) throw new Error('字幕旁白归属不一致'); p.subtitles = [...p.subtitles.filter(item => item.narrationId !== operation.narrationId), ...operation.subtitles]; break;
        case 'composition.replace': p.composition = operation.composition; break;
        default: throw new Error('未知制作操作');
    }
    return parseProduction(p);
}

export type TimelineShot = { shotId: string; media: ProductionMedia; startMs: number; endMs: number; durationMs: number; settings: ShotComposition };

/** Pure timing compiler shared by subtitle export, preview and server rendering. No implicit retiming. */
export function compileProductionTimeline(value: ProductionDocument): ProductionTimeline {
    const validation = validateProduction(value);
    const plan: ProductionTimeline = { valid: false, errors: [...validation.errors], durationMs: 0, shots: [], narration: [], subtitles: [] };
    if (!validation.valid) return plan;
    const p = value;
    const included = p.shots.filter(shot => p.composition.shots[shot.shotId]?.included !== false);
    for (let index = 0; index < included.length; index++) {
        const shot = included[index], settings = p.composition.shots[shot.shotId] ?? defaultShotComposition();
        const media = shot.adoptedMedia;
        if (!media) { plan.errors.push(`「${shot.title}」尚未采用画面`); continue; }
        if (shot.mediaRevision !== undefined && shot.mediaRevision !== (shot.revision || 1)) plan.errors.push(`「${shot.title}」采用画面对应旧输入，请重新生成或明确保留原画面`);
        const duration = media.kind === 'video' ? (settings.outMs ?? media.durationMs ?? 0) - (settings.inMs ?? 0) : settings.durationMs ?? 0;
        if (!(duration > 0)) { plan.errors.push(`「${shot.title}」需要有效的画面时长`); continue; }
        if (media.kind === 'video' && (!media.durationMs || (settings.outMs ?? media.durationMs) > media.durationMs)) plan.errors.push(`「${shot.title}」需探测视频时长或修正裁剪区间`);
        const previous = plan.shots.at(-1), overlap = previous ? settings.transitionMs : 0;
        if (overlap >= duration || (previous && overlap >= previous.durationMs)) plan.errors.push(`「${shot.title}」转场必须短于相邻镜头`);
        if (previous && (previous.settings.transitionMs + overlap >= previous.durationMs) && plan.shots.length > 1) plan.errors.push(`「${previous.shotId}」前后转场占满镜头，请缩短转场`);
        const startMs = plan.durationMs - overlap, endMs = startMs + duration;
        plan.shots.push({ shotId: shot.shotId, media, startMs, endMs, durationMs: duration, settings }); plan.durationMs = endMs;
    }
    if (!included.length) plan.errors.push('请至少加入一个镜头');
    const positions = new Map(included.map((shot, index) => [shot.shotId, index]));
    for (const line of p.narration) {
        const placement = p.composition.narration[line.id];
        if (placement?.enabled === false) continue;
        if (!line.media || !line.media.durationMs) { plan.errors.push(`旁白「${line.text.slice(0, 16)}」缺少采用音频或真实时长`); continue; }
        if (line.mediaRevision !== undefined && line.mediaRevision !== line.revision) plan.errors.push(`旁白「${line.text.slice(0, 16)}」音频对应旧文本或音色，请重新配音或明确采用原音频`);
        if (!placement) { plan.errors.push(`旁白「${line.text.slice(0, 16)}」尚未安排位置`); continue; }
        const indices = line.shotIds.map(key => positions.get(key));
        if (!indices.length || indices.some((v, i) => v === undefined || (i > 0 && v !== indices[i - 1]! + 1)) || placement.anchorShotId !== line.shotIds[0]) { plan.errors.push(`旁白「${line.text.slice(0, 16)}」关联镜头不再连续或锚点已改变，请重新安排`); continue; }
        const first = plan.shots.find(shot => shot.shotId === placement.anchorShotId), last = plan.shots.find(shot => shot.shotId === line.shotIds.at(-1));
        if (!first || !last) continue;
        const startMs = first.startMs + placement.offsetMs, endMs = startMs + line.media.durationMs;
        if (endMs > last.endMs || startMs < first.startMs) plan.errors.push(`旁白「${line.text.slice(0, 16)}」超出关联画面，请调整时长或位置`);
        plan.narration.push({ narrationId: line.id, media: line.media, startMs, endMs, volume: placement.volume });
        let previousEnd = 0;
        for (const cue of p.subtitles.filter(item => item.narrationId === line.id).sort((a, b) => a.startMs - b.startMs)) {
            if (cue.endMs <= cue.startMs || cue.endMs > line.media.durationMs || cue.startMs < previousEnd) plan.errors.push(`旁白「${line.text.slice(0, 16)}」的字幕存在重叠或越界`);
            previousEnd = cue.endMs;
            plan.subtitles.push({ ...cue, timelineStartMs: startMs + cue.startMs, timelineEndMs: startMs + cue.endMs });
        }
    }
    const narrations = [...plan.narration].sort((a, b) => a.startMs - b.startMs);
    for (let i = 1; i < narrations.length; i++) if (narrations[i].startMs < narrations[i - 1].endMs) plan.errors.push('旁白段在成片中重叠，请重新安排');
    const bgm = p.composition.bgm;
    if (bgm) { const length = (bgm.outMs ?? bgm.media.durationMs ?? 0) - bgm.inMs; if (!bgm.media.durationMs || length <= 0 || (bgm.outMs ?? 0) > bgm.media.durationMs) plan.errors.push('背景音乐区间无效或缺少真实时长'); }
    plan.errors = [...new Set(plan.errors)]; plan.valid = plan.errors.length === 0;
    return plan;
}

export function productionSrt(plan: ProductionTimeline): string {
    if (!plan.valid) throw new Error(plan.errors.join('；'));
    const stamp = (ms: number) => { const time = Math.round(ms); return `${Math.floor(time / 3600000).toString().padStart(2, '0')}:${Math.floor(time / 60000 % 60).toString().padStart(2, '0')}:${Math.floor(time / 1000 % 60).toString().padStart(2, '0')},${(time % 1000).toString().padStart(3, '0')}`; };
    return [...plan.subtitles].sort((a, b) => a.timelineStartMs - b.timelineStartMs).map((cue, i) => `${i + 1}\n${stamp(cue.timelineStartMs)} --> ${stamp(cue.timelineEndMs)}\n${cue.text.replace(/\r/g, '')}\n`).join('\n');
}

import { type InputHTMLAttributes, type ReactNode, useEffect, useRef, useState } from "react";
import { createPortal } from "react-dom";
import { App as AntApp } from "antd";
import { Eye, EyeOff, KeyRound, LoaderCircle, LockKeyhole, Mail, Ticket, X } from "lucide-react";
import { useTranslation } from "react-i18next";
import { useLocation, useNavigate } from "react-router-dom";

import { BrandName, BrandWordmark } from "@/components/brand/brand-wordmark";
import { changeAppLocale, listLocales, type AppLocale } from "@/i18n";
import { useAuthStore } from "@/stores/use-auth-store";
import { useAuthDialogStore, type AuthMode } from "@/stores/use-auth-dialog-store";
import "./auth-dialog.css";

const EMAIL_RE = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
const AUTH_ROUTES = ["/login", "/register", "/forgot-password"];

/**
 * 即梦式登录弹窗：浮在当前页面之上，遮罩 + 居中卡片，不跳路由。
 * 登录 / 注册 / 重置三种模式共用一套外壳，切换在弹窗内完成。
 */
export function AuthDialog() {
    const { message } = AntApp.useApp();
    const { i18n, t } = useTranslation();
    const navigate = useNavigate();
    const location = useLocation();

    const mode = useAuthDialogStore((state) => state.mode);
    const redirect = useAuthDialogStore((state) => state.redirect);
    const presetInvite = useAuthDialogStore((state) => state.invite);
    const setMode = useAuthDialogStore((state) => state.setMode);
    const close = useAuthDialogStore((state) => state.close);

    const login = useAuthStore((state) => state.login);
    const register = useAuthStore((state) => state.register);
    const resetPassword = useAuthStore((state) => state.resetPassword);
    const sendCode = useAuthStore((state) => state.sendCode);

    const [email, setEmail] = useState("");
    const [password, setPassword] = useState("");
    const [confirm, setConfirm] = useState("");
    const [code, setCode] = useState("");
    const [inviteCode, setInviteCode] = useState("");
    const [loading, setLoading] = useState(false);
    const [sending, setSending] = useState(false);
    const [cooldown, setCooldown] = useState(0);
    const [passwordVisible, setPasswordVisible] = useState(false);
    const [confirmVisible, setConfirmVisible] = useState(false);

    const cardRef = useRef<HTMLDivElement>(null);
    const open = mode !== null;

    // 打开时预填邀请码；切换模式保留已输入的邮箱，避免重复输入。
    useEffect(() => {
        if (!open) return;
        if (presetInvite) setInviteCode(presetInvite);
    }, [open, presetInvite]);

    useEffect(() => {
        if (cooldown <= 0) return;
        const timer = window.setTimeout(() => setCooldown((value) => value - 1), 1000);
        return () => window.clearTimeout(timer);
    }, [cooldown]);

    // 关闭入口要清掉验证码与密码，邮箱留着方便再打开
    function closeDialog() {
        close();
        setPassword("");
        setConfirm("");
        setCode("");
        setCooldown(0);
        setPasswordVisible(false);
        setConfirmVisible(false);
        // 通过 /login 等地址进来的，关闭后回首页（保留来源页更自然）
        if (AUTH_ROUTES.includes(location.pathname)) {
            navigate("/", { replace: true });
        }
    }

    useEffect(() => {
        if (!open) return;
        const onKeyDown = (event: KeyboardEvent) => {
            if (event.key === "Escape") closeDialog();
        };
        const previousOverflow = document.body.style.overflow;
        document.body.style.overflow = "hidden";
        document.addEventListener("keydown", onKeyDown);
        // 打开时把焦点交给第一个输入框，键盘用户不用先摸到卡片
        const timer = window.setTimeout(() => {
            cardRef.current?.querySelector<HTMLInputElement>("input")?.focus();
        }, 60);
        return () => {
            document.body.style.overflow = previousOverflow;
            document.removeEventListener("keydown", onKeyDown);
            window.clearTimeout(timer);
        };
        // eslint-disable-next-line react-hooks/exhaustive-deps
    }, [open]);

    if (!open) return null;

    const currentMode: AuthMode = mode;
    const needsCode = currentMode !== "login";
    const locale = i18n.resolvedLanguage as AppLocale;
    const authLocales = listLocales();
    const localeIndex = Math.max(0, authLocales.findIndex((item) => item.code === locale));
    const nextLocaleItem = authLocales[(localeIndex + 1) % authLocales.length];

    const copy =
        currentMode === "login"
            ? { title: t("auth.welcomeBack"), subtitle: t("auth.loginSubtitle"), action: t("auth.login") }
            : currentMode === "register"
              ? { title: t("auth.createWorkspace"), subtitle: t("auth.registerSubtitle"), action: t("auth.register") }
              : { title: t("auth.resetAccess"), subtitle: t("auth.resetSubtitle"), action: t("auth.reset") };

    async function handleSendCode() {
        if (!EMAIL_RE.test(email)) {
            message.error(t("auth.invalidEmail"));
            return;
        }
        setSending(true);
        try {
            await sendCode(email, currentMode === "register" ? "register" : "reset");
            message.success(t("auth.codeSent"));
            setCooldown(60);
        } catch (error) {
            message.error(error instanceof Error ? error.message : String(error));
        } finally {
            setSending(false);
        }
    }

    async function handleSubmit() {
        if (!EMAIL_RE.test(email)) {
            message.error(t("auth.invalidEmail"));
            return;
        }
        if (password.length < 8) {
            message.error(t("auth.passwordTooShort"));
            return;
        }
        if (currentMode !== "login" && password !== confirm) {
            message.error(t("auth.passwordMismatch"));
            return;
        }
        if (needsCode && !code.trim()) {
            message.error(t("auth.codeRequired"));
            return;
        }

        setLoading(true);
        try {
            if (currentMode === "login") {
                await login(email, password);
                message.success(t("auth.loginSuccess"));
                close();
                navigate(redirect || "/", { replace: true });
            } else if (currentMode === "register") {
                await register(email, code.trim(), password, inviteCode.trim().toUpperCase());
                message.success(t("auth.registerSuccess"));
                close();
                navigate(redirect || "/", { replace: true });
            } else {
                await resetPassword(email, code.trim(), password);
                message.success(t("auth.resetSuccess"));
                setPassword("");
                setConfirm("");
                setCode("");
                setMode("login");
            }
        } catch (error) {
            message.error(error instanceof Error ? error.message : String(error));
        } finally {
            setLoading(false);
        }
    }

    return createPortal(
        <div className="auth-modal">
            <div className="auth-modal-mask" onClick={closeDialog} />
            <div className="auth-modal-card" role="dialog" aria-modal="true" aria-labelledby="auth-modal-title" ref={cardRef}>
                <header className="auth-modal-head">
                    <span className="auth-modal-brand">
                        <img className="auth-modal-symbol" src="/brand/sundas-symbol.svg" alt="" />
                        <BrandWordmark height={13} decorative />
                    </span>
                    <div className="auth-modal-head-actions">
                        <button
                            type="button"
                            className="auth-modal-locale"
                            onClick={() => void changeAppLocale(nextLocaleItem.code)}
                            aria-label={t("topNav.switchLanguage", { language: nextLocaleItem.label })}
                        >
                            {nextLocaleItem.short}
                        </button>
                        <button type="button" className="auth-modal-close" onClick={closeDialog} aria-label={t("auth.backHome")}>
                            <X />
                        </button>
                    </div>
                </header>

                <div className="auth-modal-heading">
                    <h2 id="auth-modal-title" className="auth-modal-title">
                        <BrandName text={copy.title} imageHeight="0.78em" />
                    </h2>
                    <p className="auth-modal-subtitle">{copy.subtitle}</p>
                </div>

                <form
                    className="auth-modal-fields"
                    onSubmit={(event) => {
                        event.preventDefault();
                        void handleSubmit();
                    }}
                >
                    <AuthField
                        label={t("auth.email")}
                        icon={<Mail />}
                        type="email"
                        name="email"
                        placeholder={t("auth.emailPlaceholder")}
                        value={email}
                        autoComplete="email"
                        onChange={(event) => setEmail(event.target.value)}
                    />
                    <AuthField
                        label={t("auth.password")}
                        icon={<LockKeyhole />}
                        type={passwordVisible ? "text" : "password"}
                        name="password"
                        placeholder={t("auth.passwordPlaceholder")}
                        value={password}
                        autoComplete={currentMode === "login" ? "current-password" : "new-password"}
                        onChange={(event) => setPassword(event.target.value)}
                        action={
                            <button type="button" className="auth-modal-eye" onClick={() => setPasswordVisible((value) => !value)} aria-label={passwordVisible ? t("auth.hidePassword") : t("auth.showPassword")}>
                                {passwordVisible ? <EyeOff /> : <Eye />}
                            </button>
                        }
                    />
                    {currentMode !== "login" ? (
                        <AuthField
                            label={t("auth.confirmPassword")}
                            icon={<LockKeyhole />}
                            type={confirmVisible ? "text" : "password"}
                            name="confirmPassword"
                            placeholder={t("auth.confirmPassword")}
                            value={confirm}
                            autoComplete="new-password"
                            onChange={(event) => setConfirm(event.target.value)}
                            action={
                                <button type="button" className="auth-modal-eye" onClick={() => setConfirmVisible((value) => !value)} aria-label={confirmVisible ? t("auth.hidePassword") : t("auth.showPassword")}>
                                    {confirmVisible ? <EyeOff /> : <Eye />}
                                </button>
                            }
                        />
                    ) : null}
                    {needsCode ? (
                        <label className="auth-modal-field">
                            <span className="auth-modal-label">{t("auth.code")}</span>
                            <span className="auth-modal-code-row">

import { useEffect, useRef } from "react";

import "./dot-field.css";

/**
 * 点阵底纹（参照 liblib.tv 的页面背景）。
 *
 * 效果：一整片均匀的暗点，光标移上去时附近的点被「点亮」并微微放大，
 * 以光标为圆心柔和衰减，移开后回落到原亮度；光点带缓动拖尾，所以移动时是流动的而不是硬贴。
 *
 * 实现拆成两层 canvas：
 *   底层整片点阵只在尺寸变化时画一次（几千个点合成一条 Path，一次 fill）；
 *   上层每帧只重画光标半径内的点，并按亮度分桶批量 fill，
 *   所以逐帧开销只跟「光斑内的点数」有关，与页面大小无关。
 */

/* 网格间距：参照图实测 18px，点径约 2px */
const SPACING = 19;
const BASE_RADIUS = 1;
const PEAK_RADIUS = 1.85;
const BASE_ALPHA = 0.12;
const PEAK_ALPHA = 0.46;
/* 光斑半径：超过这个距离的点保持基础亮度 */
const FALLOFF = 200;
/* 光标周围的柔光，让点亮区域像一小片被打亮的面而不是孤立亮点 */
const HALO = 0.02;
const TAU = Math.PI * 2;
/* 亮度分档：把光斑内的点按衰减值分桶，每桶一次 fill */
const BUCKETS = 14;

export function DotField({ className = "" }: { className?: string }) {
    const hostRef = useRef<HTMLDivElement>(null);
    const baseRef = useRef<HTMLCanvasElement>(null);
    const glowRef = useRef<HTMLCanvasElement>(null);

    useEffect(() => {
        const host = hostRef.current;
        const baseCanvas = baseRef.current;
        const glowCanvas = glowRef.current;
        if (!host || !baseCanvas || !glowCanvas) return;
        const baseCtx = baseCanvas.getContext("2d");
        const glowCtx = glowCanvas.getContext("2d");
        if (!baseCtx || !glowCtx) return;

        const motion = window.matchMedia("(prefers-reduced-motion: reduce)");
        let width = 0;
        let height = 0;
        let dpr = 1;
        let frame = 0;

        /* x/y 是缓动后的光斑中心，tx/ty 是真实光标；strength 控制整体亮灭 */
        const pointer = { x: -9999, y: -9999, tx: -9999, ty: -9999, strength: 0, target: 0 };

        const paintBase = () => {
            baseCtx.setTransform(1, 0, 0, 1, 0, 0);
            baseCtx.clearRect(0, 0, baseCanvas.width, baseCanvas.height);
            if (!width || !height) return;
            baseCtx.setTransform(dpr, 0, 0, dpr, 0, 0);
            baseCtx.fillStyle = `rgba(255, 255, 255, ${BASE_ALPHA})`;
            baseCtx.beginPath();
            for (let y = SPACING / 2; y < height; y += SPACING) {
                for (let x = SPACING / 2; x < width; x += SPACING) {
                    baseCtx.moveTo(x + BASE_RADIUS, y);
                    baseCtx.arc(x, y, BASE_RADIUS, 0, TAU);
                }
            }
            baseCtx.fill();
        };

        const paintGlow = () => {
            glowCtx.setTransform(1, 0, 0, 1, 0, 0);
            glowCtx.clearRect(0, 0, glowCanvas.width, glowCanvas.height);
            const strength = pointer.strength;
            if (!width || !height || strength <= 0.004) return;
            glowCtx.setTransform(dpr, 0, 0, dpr, 0, 0);

            const cx = pointer.x;
            const cy = pointer.y;

            /* 柔光：被点亮的那一小片面 */
            const halo = glowCtx.createRadialGradient(cx, cy, 0, cx, cy, FALLOFF);
            halo.addColorStop(0, `rgba(255, 255, 255, ${HALO * strength})`);
            halo.addColorStop(0.5, `rgba(255, 255, 255, ${HALO * 0.38 * strength})`);
            halo.addColorStop(1, "rgba(255, 255, 255, 0)");
            glowCtx.fillStyle = halo;
            glowCtx.fillRect(cx - FALLOFF, cy - FALLOFF, FALLOFF * 2, FALLOFF * 2);

            /* 逐点算衰减，按档位归桶；smoothstep 让中心是一片接近满亮的区域，边缘才快速淡出 */
            const paths = Array.from({ length: BUCKETS }, () => new Path2D());
            const firstCol = Math.floor((cx - FALLOFF) / SPACING);
            const lastCol = Math.ceil((cx + FALLOFF) / SPACING);
            const firstRow = Math.floor((cy - FALLOFF) / SPACING);
            const lastRow = Math.ceil((cy + FALLOFF) / SPACING);
            const spans: boolean[] = new Array(BUCKETS).fill(false);

            for (let row = firstRow; row <= lastRow; row += 1) {
                const y = SPACING / 2 + row * SPACING;
                if (y < 0 || y > height) continue;
                for (let col = firstCol; col <= lastCol; col += 1) {
                    const x = SPACING / 2 + col * SPACING;
                    if (x < 0 || x > width) continue;
                    const distance = Math.hypot(x - cx, y - cy);
                    if (distance >= FALLOFF) continue;
                    const energy = 1 - distance / FALLOFF;
                    const eased = energy * energy * (3 - 2 * energy);
                    const slot = Math.min(BUCKETS - 1, Math.round(eased * (BUCKETS - 1)));
                    const radius = BASE_RADIUS + (PEAK_RADIUS - BASE_RADIUS) * eased * strength;
                    const path = paths[slot];
                    path.moveTo(x + radius, y);
                    path.arc(x, y, radius, 0, TAU);
                    spans[slot] = true;
                }
            }

            /* 由暗到亮依次铺，亮的覆盖在暗的上面 */
            for (let slot = 0; slot < BUCKETS; slot += 1) {
                if (!spans[slot]) continue;
                const eased = slot / (BUCKETS - 1);
                const alpha = BASE_ALPHA + (PEAK_ALPHA - BASE_ALPHA) * eased * strength;
                glowCtx.fillStyle = `rgba(255, 255, 255, ${alpha.toFixed(4)})`;
                glowCtx.fill(paths[slot]);
            }
        };

        const tick = () => {
            frame = 0;
            const snap = motion.matches;
            const follow = snap ? 1 : 0.18;
            const rise = snap ? 1 : 0.1;
            pointer.x += (pointer.tx - pointer.x) * follow;
            pointer.y += (pointer.ty - pointer.y) * follow;
            pointer.strength += (pointer.target - pointer.strength) * rise;
            paintGlow();
            const following = Math.abs(pointer.tx - pointer.x) > 0.4 || Math.abs(pointer.ty - pointer.y) > 0.4;
            const changing = Math.abs(pointer.target - pointer.strength) > 0.003;
            if (following || changing) frame = requestAnimationFrame(tick);
        };

        const wake = () => {
            if (!frame) frame = requestAnimationFrame(tick);
        };

        const onMove = (event: PointerEvent) => {
            if (event.pointerType === "touch") return;
            const rect = host.getBoundingClientRect();
            const x = event.clientX - rect.left;
            const y = event.clientY - rect.top;
            if (x < 0 || y < 0 || x > rect.width || y > rect.height) {
                if (pointer.target !== 0) {
                    pointer.target = 0;
                    wake();
                }
                return;
            }
            /* 熄着进来的话直接落位，不要从上一个位置飞过来 */
            if (pointer.strength < 0.05) {
                pointer.x = x;
                pointer.y = y;
            }
            pointer.tx = x;
            pointer.ty = y;
            pointer.target = 1;
            wake();
        };

        const onLeave = () => {
            if (pointer.target === 0) return;
            pointer.target = 0;
            wake();
        };

        const resize = () => {
            const rect = host.getBoundingClientRect();
            width = Math.round(rect.width);
            height = Math.round(rect.height);
            dpr = Math.min(window.devicePixelRatio || 1, 2);
            for (const canvas of [baseCanvas, glowCanvas]) {
                canvas.width = Math.round(width * dpr);
                canvas.height = Math.round(height * dpr);
            }
            paintBase();
            glowCtx.setTransform(1, 0, 0, 1, 0, 0);
            glowCtx.clearRect(0, 0, glowCanvas.width, glowCanvas.height);
        };

        const observer = new ResizeObserver(resize);
        observer.observe(host);
        resize();

        window.addEventListener("pointermove", onMove, { passive: true });
        window.addEventListener("blur", onLeave);
        document.addEventListener("pointerleave", onLeave);
        motion.addEventListener("change", wake);

        return () => {
            if (frame) cancelAnimationFrame(frame);
            observer.disconnect();
            window.removeEventListener("pointermove", onMove);
            window.removeEventListener("blur", onLeave);
            document.removeEventListener("pointerleave", onLeave);
            motion.removeEventListener("change", wake);
        };
    }, []);

    return (
        <div ref={hostRef} aria-hidden="true" className={`jm-dot-field ${className}`.trim()}>
            <canvas ref={baseRef} />
            <canvas ref={glowRef} />
        </div>
    );
}

    const theme = canvasThemes[useThemeStore((state) => state.theme)];

        };

const stepNames: Record<CharacterStepName, string> = { prepare: "准备镜头", image: "生成替换参考帧", video: "生成替换视频", assemble: "回插并输出成片" };
const stateNames = { queued: "等待处理", running: "处理中", succeeded: "已完成", failed: "失败", review: "待核对" };

async function cloudKey(node: CanvasNodeData, owner: string) {
    const content = node.metadata?.content || "";
    const localKey = node.metadata?.storageKey;
    const direct = mediaKey(content);
    if (direct) return direct;
    const input = node.type === CanvasNodeType.Image
        ? await imageToDataUrl({ dataUrl: content, storageKey: localKey })
        : localKey ? await getMediaBlob(localKey) : content;
    if (!input) throw new Error("素材内容不可用，请重新上传");
    const uploaded = await uploadMediaFile(input, node.type, { expectedOwner: owner });
    const key = mediaKey(uploaded.url);
    if (!key) throw new Error("素材尚未同步到服务端，请检查网络后重试");
    return key;
}

function mediaKey(value: string) {
    try { const url = new URL(value, window.location.origin); return url.origin === window.location.origin && url.pathname.startsWith("/media/private/") ? decodeURIComponent(url.pathname.slice(7)) : ""; }
    catch { return ""; }
}
