// Keep previously opened pages while refreshing the newest page after a task
// update. Requests are serialized so a refresh and an older-page load cannot
// overwrite one another.
export class GenerationHistoryPager {
    private records = new Map<string, RestoredLog>();
    private activeIds = new Set<string>();
    private hidden: string[] = [];
    private nextPage = 1;
    private more = true;
    private queue: Promise<unknown> = Promise.resolve();

    constructor(private owner: string, private type: 'image' | 'video') {}

    private enqueue<T>(run: () => Promise<T>): Promise<T> {
        const result = this.queue.then(run, run);
        this.queue = result.catch(() => undefined);
        return result;
    }

    private async fetch(page: number) {
        const data = await fetchGenerationHistory(this.owner, this.type, page);
        this.hidden = data.hiddenTaskIds;
        for (const id of this.hidden) this.records.delete(id);
        for (const log of data.logs) this.records.set(log.id, log);
        if (page >= this.nextPage) {
            this.nextPage = page + 1;
            this.more = data.hasMore;
        }
        return this.snapshot();
    }

    refresh() { return this.enqueue(async () => {
        await this.fetch(1);
        const active = new Set<string>();
        for (let page = 1; ; page++) {
            const result = await fetchGenerationHistory(this.owner, this.type, page, { active: true });
            for (const log of result.logs) { active.add(log.id); this.records.set(log.id, log); }
            if (!result.hasMore) break;
        }
        for (const id of this.activeIds) {
            if (!active.has(id)) await this.readRecord(id);
        }
        this.activeIds = active;
        return this.snapshot();
    }); }
    private async readRecord(id: string) {
        const result = await fetchGenerationHistory(this.owner, this.type, 1, { id });
        this.hidden = result.hiddenTaskIds;
        for (const hidden of this.hidden) this.records.delete(hidden);
        if (!result.logs.length) this.records.delete(id);
        for (const log of result.logs) this.records.set(log.id, log);
        return this.snapshot();
    }
    loadRecord(id: string) { return this.enqueue(() => this.readRecord(id)); }
    loadMore() { return this.enqueue(() => this.more ? this.fetch(this.nextPage) : Promise.resolve(this.snapshot())); }
    snapshot() { return { logs: [...this.records.values()], hiddenTaskIds: this.hidden, hasMore: this.more }; }
}

function resolveSize(ratio: string): string {
    const parsedRatio = parseImageRatio(ratio);
    const preset = imageSizePresets["1k"][ratio];
    if (preset) return preset;
    const isLandscape = parsedRatio.width >= parsedRatio.height;
    const longRatio = isLandscape ? parsedRatio.width / parsedRatio.height : parsedRatio.height / parsedRatio.width;
    const shortSide = DEFAULT_IMAGE_SHORT_SIDE;
    const longSide = Math.round((shortSide * longRatio) / IMAGE_SIZE_STEP) * IMAGE_SIZE_STEP;
    const width = isLandscape ? longSide : shortSide;
    const height = isLandscape ? shortSide : longSide;
    validateImageSize(width, height);
    return `${width}x${height}`;
}

function parseRatioValue(value: string) {
    const parts = value.split(":");
    if (parts.length !== 2) throw new Error(apiText("invalidImageSizeFormat"));
    const w = Number(parts[0]);
    const h = Number(parts[1]);
    if (!Number.isFinite(w) || !Number.isFinite(h) || w <= 0 || h <= 0) throw new Error(apiText("positiveImageRatio"));

}

function validateImageSize(width: number, height: number) {
    if (!Number.isInteger(width) || !Number.isInteger(height) || width <= 0 || height <= 0) throw new Error(apiText("positiveImageDimensions"));
    if (width % IMAGE_SIZE_STEP !== 0 || height % IMAGE_SIZE_STEP !== 0) throw new Error(apiText("imageDimensionStep"));
    if (Math.max(width, height) > IMAGE_MAX_EDGE) throw new Error(apiText("imageEdgeLimit"));
    if (Math.max(width, height) / Math.min(width, height) > IMAGE_MAX_RATIO) throw new Error(apiText("imageRatioLimit"));

}

function closestGeminiAspectRatio(value: string) {
    const ratio = parseImageRatio(value);
    const target = ratio.width / ratio.height;

function readApiErrorMessage(value: unknown): string {
    if (!value) return "";
    if (typeof value === "string") {
        // The value may be serialized JSON, such as error.message, or a plain-text error.
        try {
            const parsed = JSON.parse(value);
            const inner = readApiErrorMessage(parsed) || value;
            // Treat an empty parsed object such as "{}" as having no useful message.
            if (inner === value && typeof parsed === "object" && Object.keys(parsed).length === 0) return "";
            return friendlyErrorMessage(inner);
        } catch {
            // Detect HTML error pages.
            if (/<[a-z][\s\S]*>/i.test(value)) return apiText("htmlError", { preview: `${value.slice(0, 80)}...` });
            return friendlyErrorMessage(value);
        }
    }
    if (typeof value !== "object") return "";
    const payload = value as { msg?: unknown; message?: unknown; error?: unknown; detail?: unknown };
    // error may be a string or an object containing a message.
    const errorMsg =
        typeof payload.error === "string"
            ? payload.error
            : (payload.error as { message?: unknown })?.message;
    return friendlyErrorMessage(
        readApiErrorMessage(payload.msg) ||
        readApiErrorMessage(payload.message) ||
        readApiErrorMessage(errorMsg) ||
        readApiErrorMessage(payload.detail) ||
        ""
    );
}

function readAxiosError(error: unknown, fallback: string) {
    if (axios.isCancel(error)) return apiText("requestCanceled");
    if (axios.isAxiosError(error)) {
        if (error.code === "ECONNABORTED" || error.code === "ETIMEDOUT") return apiText("imageTimeout");
        if (!error.response && error.code === "ERR_NETWORK") return apiText("requestFailed");
        const responseData = error.response?.data;
        // Prefer the API error from the response body.
        const apiMsg = readApiErrorMessage(responseData);
        if (apiMsg) return apiMsg;
        // Infer the error from the HTTP status when the response body has no usable message.
        const statusMsg = readStatusError(error.response?.status, fallback);
        if (statusMsg) return statusMsg;
        // Fall back to Axios's own error message.
        return error.message || fallback;
    }
    if (error instanceof DOMException && error.name === "AbortError") return apiText("requestCanceled");
    return error instanceof Error ? readApiErrorMessage(error.message) || error.message : fallback;
}

function readStatusError(status: number | undefined, fallback: string) {

}

function consumeResponseStreamBlock(block: string, state: ResponseStreamState, onDelta?: (text: string) => void) {
    const data = block
        .split(/\r?\n/)
        .filter((line) => line.startsWith("data:"))
        .map((line) => line.slice(5).replace(/^ /, ""))
        .join("\n")
        .trim();
    if (!data) return;
    if (data === "[DONE]") { state.completed = true; return; }
    const event = JSON.parse(data) as Record<string, unknown>;
    const type = stringValue(event.type);
    if (type === "response.completed") state.completed = true;
    const errorMessage = responseErrorMessage(event);
    if (errorMessage) state.error = errorMessage;
    if (type === "response.output_text.delta" && typeof event.delta === "string") {
        state.text += event.delta;
        onDelta?.(state.text);
    }
    if (type === "response.output_text.done" && !state.text && typeof event.text === "string") {
        state.text = event.text;
        onDelta?.(state.text);
    }
    if (type === "response.completed" && isRecord(event.response)) {
        state.payload = event.response as ResponseApiPayload;
    } else if (Array.isArray(event.output)) {
        state.payload = event as ResponseApiPayload;
    }
}

function consumeResponseStreamText(state: ResponseStreamState, text: string, onDelta?: (text: string) => void, flush = false) {
    state.buffer += text;
    for (;;) {
        const match = state.buffer.match(/\r?\n\r?\n/);
        if (!match) break;
        const index = match.index ?? 0;
        consumeResponseStreamBlock(state.buffer.slice(0, index), state, onDelta);
        state.buffer = state.buffer.slice(index + match[0].length);
    }
    if (flush && state.buffer.trim()) {
        consumeResponseStreamBlock(state.buffer, state, onDelta);
        state.buffer = "";
    }
}

async function requestStreamingResponse(config: AiConfig, body: Record<string, unknown>, onDelta?: (text: string) => void, options?: RequestOptions): Promise<ToolResponseResult> {
    const response = await fetch(aiApiUrl(config, "/responses"), {
        method: "POST",
        headers: { ...aiHeaders(config, "application/json"), Accept: "text/event-stream", ...expectedAccountHeaders(aiApiUrl(config, "/responses"), options?.expectedUserId), ...(options?.surface ? { "X-Sundas-Surface": options.surface } : {}) },
        body: JSON.stringify({ ...body, stream: true }),
        signal: options?.signal,
    });
    if (!response.ok) throw new Error(await readFetchError(response, apiText("requestFailed")));
    if (!response.body) {
        const payload = (await response.json()) as ResponseApiPayload;
        validateResponsePayload(payload);
        return parseToolResponse(payload);
    }

    const reader = response.body.getReader();
    const decoder = new TextDecoder();
    const state: ResponseStreamState = { buffer: "", text: "", completed: false };
    try {
        for (;;) {
            const { done, value } = await reader.read();
            if (done) break;
            consumeResponseStreamText(state, decoder.decode(value, { stream: true }), onDelta);
            if (state.error) throw new Error(state.error);
        }
        consumeResponseStreamText(state, decoder.decode(), onDelta, true);
        if (state.error) throw new Error(state.error);
        if (!state.completed) throw new Error(apiText("streamIncomplete"));
        if (!state.payload) return { content: state.text, toolCalls: [] };
        validateResponsePayload(state.payload);
        const result = parseToolResponse(state.payload);
        return { ...result, content: state.text || result.content };
    } catch (error) {
        void reader.cancel().catch(() => undefined);
        throw error;
    } finally {
        reader.releaseLock();
    }
}

    const systemText = [
        config.systemPrompt.trim(),
        ...messages.flatMap((message) => (!("type" in message) && message.role === "system" ? [geminiTextContent(message.content)] : [])),
    ]
        .filter(Boolean)
        .join("\n\n");
    const contents = toGeminiContents(messages.filter((message) => ("type" in message ? true : message.role !== "system")));
    return {
        contents,
        ...(systemText ? { systemInstruction: { parts: [{ text: systemText }] } } : {}),
        ...extra,
    };
}

function toGeminiContents(messages: ResponseInputMessage[]): GeminiContent[] {
    const callNameById = new Map<string, string>();
    return messages.flatMap((message): GeminiContent[] => {
        if ("type" in message) {
            callNameById.set(message.call_id, message.name);
            return [{ role: "model", parts: [{ functionCall: { id: message.call_id, name: message.name, args: jsonObject(message.arguments) }, ...(message.thoughtSignature ? { thoughtSignature: message.thoughtSignature } : {}) }] }];
        }
        if (message.role === "tool") {
            const name = callNameById.get(message.tool_call_id) || "tool_result";
            return [{ role: "user", parts: [{ functionResponse: { id: message.tool_call_id, name, response: { result: jsonValue(message.content) } } }] }];
        }
        return [{ role: message.role === "assistant" ? "model" : "user", parts: toGeminiParts(message.content) }];
    });
}

function toGeminiParts(content: ResponseMessageContent): GeminiPart[] {
    if (!Array.isArray(content)) return [{ text: String(content || "") }];
    return content.map((item) => (item.type === "text" ? { text: item.text } : toGeminiImagePart(item.image_url.url)));
}

function toGeminiImagePart(url: string): GeminiPart {
    const match = url.match(/^data:([^;,]+);base64,(.+)$/);
    if (match) return { inlineData: { mimeType: match[1], data: match[2] } };
    return { fileData: { fileUri: url, mimeType: "image/png" } };
}

function geminiTextContent(content: ResponseMessageContent) {
    if (!Array.isArray(content)) return String(content || "");
    return content.map((item) => (item.type === "text" ? item.text : item.image_url.url)).join("\n");
}

function jsonObject(value: string): Record<string, unknown> {
    const parsed = jsonValue(value);
    return isRecord(parsed) ? parsed : {};
}

function jsonValue(value: string): unknown {
    try {
        return JSON.parse(value);
    } catch {
        return value;
    }
}

function toGeminiToolOptions(tools: ResponseFunctionTool[], toolChoice: ToolChoice) {
    if (!tools.length) return {};
    const functionDeclarations = tools.map((tool) => ({
        name: tool.function.name,
        description: tool.function.description,
        parameters: tool.function.parameters,
    }));
    const functionCallingConfig =
        typeof toolChoice === "object"
            ? { mode: "ANY", allowedFunctionNames: [toolChoice.name] }
            : { mode: toolChoice === "required" ? "ANY" : "AUTO" };
    return {
        tools: [{ functionDeclarations }],
        toolConfig: { functionCallingConfig },
    };
}

async function requestGeminiStreamingResponse(config: AiConfig, body: Record<string, unknown>, onDelta?: (text: string) => void, options?: RequestOptions): Promise<ToolResponseResult> {
    const response = await fetch(`${geminiApiUrl(config, "streamGenerateContent")}?alt=sse`, {
        method: "POST",
        headers: geminiHeaders(config),
        body: JSON.stringify(body),
        signal: options?.signal,
    });
    if (!response.ok) throw new Error(await readFetchError(response, apiText("requestFailed")));
    if (!response.body) {
        const payload = (await response.json()) as GeminiPayload;
        return parseGeminiToolResponse(payload);
    }

    const reader = response.body.getReader();
    const decoder = new TextDecoder();
    const state: GeminiStreamState = { buffer: "", text: "", toolCalls: [] };
    for (;;) {
        const { done, value } = await reader.read();
        if (done) break;
        consumeGeminiStreamText(state, decoder.decode(value, { stream: true }), onDelta);
        if (state.error) throw new Error(state.error);
    }
    consumeGeminiStreamText(state, decoder.decode(), onDelta, true);
    if (state.error) throw new Error(state.error);
    return { content: state.text, toolCalls: state.toolCalls };
}

function consumeGeminiStreamText(state: GeminiStreamState, text: string, onDelta?: (text: string) => void, flush = false) {
    state.buffer += text;
    for (;;) {
        const match = state.buffer.match(/\r?\n\r?\n/);
        if (!match) break;
        const index = match.index ?? 0;
        consumeGeminiStreamBlock(state.buffer.slice(0, index), state, onDelta);
        state.buffer = state.buffer.slice(index + match[0].length);
    }
    if (flush && state.buffer.trim()) {
        consumeGeminiStreamBlock(state.buffer, state, onDelta);
        state.buffer = "";
    }
}

function consumeGeminiStreamBlock(block: string, state: GeminiStreamState, onDelta?: (text: string) => void) {
    const data = block
        .split(/\r?\n/)
        .filter((line) => line.startsWith("data:"))
        .map((line) => line.slice(5).replace(/^ /, ""))
        .join("\n")
        .trim();
    if (!data || data === "[DONE]") return;
    const result = parseGeminiToolResponse(JSON.parse(data) as GeminiPayload);
    if (result.content) {
        state.text += result.content;
        onDelta?.(state.text);
    }
    state.toolCalls.push(...result.toolCalls);
}

function parseGeminiToolResponse(payload: GeminiPayload): ToolResponseResult {
    validateGeminiPayload(payload);
    const parts = payload.candidates?.flatMap((candidate) => candidate.content?.parts || []) || [];
    const content = parts.map((part) => part.text || "").join("");
    const toolCalls = parts
        .map((part) => part.functionCall)
        .filter((call): call is NonNullable<GeminiPart["functionCall"]> => Boolean(call?.name))
        .map((call) => {
            const part = parts.find((item) => item.functionCall === call);
            const thoughtSignature = part?.thoughtSignature || part?.thought_signature;
            return {
                id: call.id || nanoid(),
                type: "function" as const,
                function: { name: call.name || "", arguments: JSON.stringify(call.args || {}) },
                ...(thoughtSignature ? { thoughtSignature } : {}),
            };
        });
    return { content, toolCalls };
}

function parseGeminiImagePayload(payload: GeminiPayload) {

    const images =
        payload.candidates
            ?.flatMap((candidate) => candidate.content?.parts || [])

export async function tryDirectMediaUpload(blob: Blob, options: UploadOptions): Promise<Response | null> {
    const headers = { 'Content-Type': 'application/json', 'X-Sundas-User': options.owner };
    const config = await fetch('/api/uploads/config', { credentials: 'include', signal: options.signal, headers: { 'X-Sundas-User': options.owner } });
    if (!config.ok) return config;
    if (!(await config.json()).direct) return null;
    // 图片已由上传入口尝试标准化；未支持的特殊格式仍走现有服务器转换。
    if (/image\/(heic|heif|avif|tiff|bmp)/i.test(blob.type)) return null;
    const prepared = await fetch('/api/uploads/prepare', { method: 'POST', credentials: 'include', headers,
        body: JSON.stringify({ bytes: blob.size }), signal: options.signal });
    if (!prepared.ok) return prepared;
    const upload = await prepared.json() as { id: string; url: string; headers: Record<string, string> };
    const sent = await new Promise<number>((resolve, reject) => {
        const xhr = new XMLHttpRequest();
        const abort = () => xhr.abort();
        xhr.open('PUT', upload.url);
        // OSS 不接收本站 Cookie 或用户会话头。
        xhr.withCredentials = false;
        Object.entries(upload.headers).forEach(([key, value]) => xhr.setRequestHeader(key, value));
        xhr.upload.onprogress = event => { if (event.lengthComputable) options.onProgress?.(Math.min(95, Math.round(event.loaded / event.total * 95))); };
        xhr.onload = () => resolve(xhr.status);
        xhr.onerror = () => reject(new TypeError('文件上传失败，请检查网络后重试'));
        xhr.onabort = () => reject(new DOMException('Upload aborted', 'AbortError'));
        xhr.onloadend = () => options.signal.removeEventListener('abort', abort);
        if (options.signal.aborted) { reject(new DOMException('Upload aborted', 'AbortError')); return; }
        options.signal.addEventListener('abort', abort, { once: true });
        options.onProgress?.(0);
        xhr.send(blob);
    });
    if (sent < 200 || sent >= 300) return new Response(JSON.stringify({ message: '文件上传失败，请重新上传' }), { status: 400, headers: { 'Content-Type': 'application/json' } });
    return fetch(`/api/uploads/${encodeURIComponent(upload.id)}/complete`, { method: 'POST', credentials: 'include', headers, body: '{}', signal: options.signal });
}

function createPluginHttp(config: AiConfig, options?: RequestOptions): PluginHttp {
    const run = async (method: "get" | "post", path: string, body: unknown, opts?: PluginHttpOptions) => {
        const isForm = typeof FormData !== "undefined" && body instanceof FormData;
        const response = await axios.request({
            method,
            url: pluginUrl(config, path),
            data: method === "post" ? body : undefined,
            params: opts?.params,
            headers: pluginHeaders({ Authorization: `Bearer ${config.apiKey}`, ...opts?.headers }, method === "post" && !isForm && body !== undefined),
            responseType: opts?.responseType || "json",
            signal: options?.signal,
        });
        return response.data;
    };
    return {
        url: (path) => pluginUrl(config, path),
        post: (path, body, opts) => run("post", path, body, opts),
        get: (path, opts) => run("get", path, undefined, opts),
    };
}

function createPluginRequest(config: AiConfig, options?: RequestOptions) {
    return async (requestConfig: AxiosRequestConfig & { url: string }) => {
        const response = await axios.request({ ...requestConfig, url: pluginUrl(config, requestConfig.url), signal: options?.signal });
        return response.data;
    };
}

function sleep(ms: number, signal?: AbortSignal) {
    return new Promise<void>((resolve, reject) => {
        if (signal?.aborted) {
            reject(new DOMException("Aborted", "AbortError"));
            return;
        }
        const timer = setTimeout(resolve, ms);
        signal?.addEventListener(
            "abort",
            () => {
                clearTimeout(timer);
                reject(new DOMException("Aborted", "AbortError"));
            },
            { once: true },
        );
    });
}

function createPoll(signal?: AbortSignal) {
    return async function poll<T, R>(request: () => Promise<T>, extract: (value: T) => R | null | undefined | false, options?: PluginPollOptions): Promise<R> {
        const intervalMs = options?.intervalMs ?? 2500;
        const timeoutMs = options?.timeoutMs ?? 300000;
        const deadline = performance.now() + timeoutMs;
        for (;;) {
            if (signal?.aborted) throw new DOMException("Aborted", "AbortError");
            const result = extract(await request());
            if (result !== null && result !== undefined && result !== false) return result;
            if (performance.now() >= deadline) throw new Error(i18n.t("modelPlugin.pollTimeout"));
            await sleep(intervalMs, signal);
        }
    };
}

/**
 * Run a user-authored model call script. Locals are injected (see PLUGIN_VARIABLES); templates wrap them in an async function.
 * The script still runs as an async function body and must `return` the result.
 */
export async function runModelPlugin<T = unknown>(args: RunPluginArgs): Promise<T> {
    const { config } = args;
    const http = createPluginHttp(config, { signal: args.signal });
    const request = createPluginRequest(config, { signal: args.signal });
    const poll = createPoll(args.signal);
    const runner = new Function(
        "prompt",
        "images",
        "videos",
        "audios",
        "messages",
        "params",
        "model",
        "baseUrl",
        "apiKey",
        "systemPrompt",
        "reasoningEffort",
        "http",
        "request",
        "poll",
        "sleep",
        "signal",
        "onDelta",
        `"use strict"; return (async () => {\n${args.script}\n})();`,
    ) as (...fnArgs: unknown[]) => Promise<T>;
    try {
        return await runner(
            args.prompt || "",
            args.images || [],
            args.videos || [],
            args.audios || [],
            args.messages || [],
            args.params || {},
            config.model,
            config.baseUrl,
            config.apiKey,
            config.systemPrompt || "",
            config.reasoningEffort,
            http,
            request,
            poll,
            (ms: number) => sleep(ms, args.signal),
            args.signal,
            args.onDelta,
        );
    } catch (error) {
        if (error instanceof DOMException && error.name === "AbortError") throw error;
        if (axios.isCancel(error)) throw error;
        const message = error instanceof Error ? error.message : String(error);
        throw new Error(i18n.t("modelPlugin.executionFailed", { message }));
    }
}

export function getPluginAuthoringPrompt(capability: ModelCapability, modelName: string, draft = "") {
    const variables = getPluginVariables().filter((variable) => !variable.capabilities || variable.capabilities.includes(capability));
    const lines = [
        i18n.t("modelPlugin.authoring.intro", { capability: i18n.t(`config.channelEditor.capabilities.${capability}`), model: modelName || i18n.t("modelPlugin.authoring.anyModel") }),
        "",
        i18n.t("modelPlugin.authoring.shape"),
        "",
        i18n.t("modelPlugin.authoring.returnTitle"),
        getPluginReturn(capability),
        "",
        i18n.t("modelPlugin.authoring.variablesTitle"),
        ...variables.map((variable) => `- ${variable.name} (${variable.type}): ${variable.desc}`),
        "",
        i18n.t("modelPlugin.authoring.rulesTitle"),
        i18n.t("modelPlugin.authoring.rules"),
    ];
    const templates = getPluginTemplates()[capability];
    if (templates.length) {
        lines.push("", i18n.t("modelPlugin.authoring.examplesTitle"));
        for (const template of templates) {
            lines.push("", `${template.label}`, template.script);
        }
    }
    if (draft.trim()) {
        lines.push("", i18n.t("modelPlugin.authoring.draftTitle"), draft.trim());
    }
    return lines.join("\n");
}

export function compactApiParams(params: ApiParams) {
    return Object.fromEntries(Object.entries(params).filter(([, value]) => value !== "" && value !== undefined && (!Array.isArray(value) || value.length > 0))) as ApiParams;
}

export function serializeApiParams(params?: ApiParams) {
    const queryParams = new URLSearchParams();
    for (const [key, value] of Object.entries(params || {})) {
        if (value === undefined) continue;
        if (Array.isArray(value)) value.forEach((item) => queryParams.append(key, String(item)));
        else queryParams.set(key, String(value));
    }
    return queryParams;

import type { VideoGenerationResult, VideoGenerationTaskState } from './video';

export async function waitForVideoResult(poll: () => Promise<VideoGenerationTaskState>, signal?: AbortSignal): Promise<VideoGenerationResult> {
    const startedAt = Date.now();
    while (true) {
        if (signal?.aborted) throw new DOMException("Aborted", "AbortError");
        let state: VideoGenerationTaskState | undefined;
        try { state = await poll(); }
        catch (error) {
            if (signal?.aborted) throw new DOMException("Aborted", "AbortError");
            // 查询异常不代表生成失败，继续查询同一任务；不创建新任务或改变计费。
        }
        if (state?.status === "completed") return state.result;
        if (state?.status === "failed") {
            const error = new Error(state.error); error.name = "VideoTaskFailed"; throw error;
        }
        if (state?.status === "unavailable") throw new Error(state.error);
        if (Date.now() - startedAt >= 60 * 60 * 1000) throw new Error("已等待超过 60 分钟，任务结果尚未确认，积分以后台任务状态为准；请查询原任务，不要重复生成。");
        await delay(5000, signal);
    }
}

function delay(ms: number, signal?: AbortSignal) {
    return new Promise<void>((resolve, reject) => {
        if (signal?.aborted) {
            reject(new DOMException("Aborted", "AbortError"));
            return;
        }
        const abort = () => {
            clearTimeout(timer);
            reject(new DOMException("Aborted", "AbortError"));
        };
        const timer = setTimeout(() => { signal?.removeEventListener("abort", abort); resolve(); }, ms);
        signal?.addEventListener("abort", abort, { once: true });
    });
}

function videoPluginResult(result: unknown): VideoGenerationResult {
    if (result instanceof Blob) return { blob: result };
    if (typeof result === "string") return { url: result, mimeType: "video/mp4" };

        const record = result as Record<string, unknown>;
        if (record.blob instanceof Blob) return { blob: record.blob };
        const url = [record.url, record.video_url, record.result_url].find((value) => typeof value === "string" && value) as string | undefined;
        if (url) return { url, mimeType: "video/mp4" };
    }
    throw new Error(apiText("scriptNoVideo"));
}

    }

}

function mediaKeyFromUrl(value?: string) {
    if (!value) return "";
    const path = value.startsWith("http") ? (() => { try { return new URL(value).pathname; } catch { return ""; } })() : value.split(/[?#]/)[0];
    const marker = "/media/";
    const index = path.indexOf(marker);
    if (index < 0) return "";
    const key = decodeURIComponent(path.slice(index + marker.length));
    return key.startsWith("private/") || key.startsWith("upload/") ? key : "";
}

type PreparedReference = { file: File } | { key: string; name: string; mime: string };

/**
 * 参考视频/音频已经存在本站媒体库时只传对象键，由服务端直接取文件。
 * 旧做法是把文件整包下载回浏览器再塞进提交请求：手机上常见 100-200MB 的视频会让请求体翻倍，
 * 既超出 nginx 的请求体上限（直接 413），上行慢的手机还会在上传阶段超时或被系统回收标签页。
 */
async function prepareReferenceMedia(
    item: { name: string; type?: string; url?: string; storageKey?: string },
    fallbackName: string,
    errorKey: "invalidReferenceVideo" | "invalidReferenceAudio",
    options?: RequestOptions,
): Promise<PreparedReference> {
    const key = item.storageKey ? "" : mediaKeyFromUrl(item.url);
    if (key) return { key, name: key.split("/").pop() || item.name || fallbackName, mime: item.type || "application/octet-stream" };
    return { file: await referenceMediaToFile(item, fallbackName, errorKey, options) };
}

async function createOpenAIVideoTask(config: AiConfig, model: string, prompt: string, references: ReferenceImage[], options?: VideoMediaOptions): Promise<VideoGenerationTask> {
    const prepare = async () => {
        const images: PreparedReference[] = await Promise.all(references.map(async (image) => {
            const key = mediaKeyFromUrl(image.url) || mediaKeyFromUrl(image.dataUrl);
            if (key) return { key, name: image.name, mime: image.type || "image/png" };
            return { file: await dataUrlToFile({ ...image, dataUrl: await imageToDataUrl(image) }) };
        }));
        const videos = await Promise.all((options?.videos || []).map((video) => prepareReferenceMedia(video, "ref.mp4", "invalidReferenceVideo", options)));
        const audios = await Promise.all((options?.audios || []).map((audio) => prepareReferenceMedia(audio, "ref.mp3", "invalidReferenceAudio", options)));
        return { images, videos, audios };
    };
    const { images, videos, audios } = await prepare().catch((error: unknown) => {
        if (error instanceof Error && error.name === "AbortError") throw error;
        const rejected = new Error(error instanceof Error ? error.message : apiText("videoTaskCreateFailed"));
        rejected.name = "VideoSubmissionRejected";
        throw rejected;
    });
    const mode = resolveVideoMode(config.videoMode, images.length);
    const ratio = inferVideoRatio(config.size || "16:9");
    const body = new FormData();
    body.append("model", modelOptionName(model));
    if (config.videoRouteId && model.startsWith('server::')) body.append("routeId", config.videoRouteId);
    if (options?.clientTaskId) body.append("clientTaskId", options.clientTaskId);
    body.append("prompt", prompt);
    body.append("seconds", normalizeVideoSeconds(config.videoSeconds, model));
    body.append("size", normalizeVideoSize(config.size, config.vquality) || "1280x720");
    if (ratio && ratio !== "auto") {
        body.append("ratio", ratio);
    }
    body.append("resolution_name", normalizeVideoResolution(config.vquality));
    body.append("generate_audio", String(boolConfig(config.videoGenerateAudio, true)));
    body.append("watermark", String(boolConfig(config.videoWatermark, false)));
    body.append("mode", mode);
    const referenceKeys: { field: string; key: string; name: string; mime: string }[] = [];
    const appendImage = (field: string, item: PreparedReference, filename: string) => {
        if ("key" in item) referenceKeys.push({ field, key: item.key, name: filename, mime: item.mime });
        else body.append(field, item.file, filename);
    };
    // 视频/音频沿用媒体库里的原始文件名：保留真实扩展名，且每个对象键天然唯一。
    const appendMedia = (field: string, item: PreparedReference, fallbackName: string) => {
        if ("key" in item) referenceKeys.push({ field, key: item.key, name: item.name || fallbackName, mime: item.mime });
        else body.append(field, item.file);
    };
    if (mode === "frames") {
        if (images[0]) appendImage("first_frame", images[0], "first.png");
        if (images[1]) appendImage("last_frame", images[1], "last.png");
    } else {
        images.forEach((item, idx) => appendImage("image[]", item, `ref_${idx + 1}.png`));
    }
    videos.forEach((item, idx) => appendMedia("video[]", item, `ref_video_${idx + 1}.mp4`));
    audios.forEach((item, idx) => appendMedia("audio[]", item, `ref_audio_${idx + 1}.mp3`));
    // 所有参考素材登记完再写 referenceKeys，服务端按字段名把它们还原成对应的文件。
    if (referenceKeys.length) body.append("referenceKeys", JSON.stringify(referenceKeys));
    try {
        const send = (confirmed = false) => axios.post<ApiVideoResponse>(aiApiUrl(config, "/videos"), body, { headers: { ...aiHeaders(config, undefined, options?.expectedUserId), ...(options?.surface ? { 'X-Sundas-Surface': options.surface } : {}), ...(confirmed ? { 'X-Confirm-Duplicate': 'true' } : {}) }, signal: options?.signal });
        let response;
        try { response = await send(); }
        catch (error) {
            if (!axios.isAxiosError(error) || error.response?.data?.errorCode !== 'DUPLICATE_VIDEO_IN_PROGRESS') throw error;
            if (!window.confirm('已经有一条相同的内容正在生成中，继续提交会再次消耗积分。是否继续？')) {
                const cancelled = new Error('已取消重复提交');
                cancelled.name = 'VideoSubmissionRejected';
                throw cancelled;
            }
            response = await send(true);
        }
        const created = unwrapVideoResponse(response.data);
        if (!created.id) throw new Error(apiText("noVideoTaskId"));
        return { id: created.id, provider: "openai", model };
    } catch (error) {
        if (isVideoSubmissionRejected(error)) throw error;
        if (axios.isAxiosError(error) && error.response?.status === 402) {
            const insufficientCredits = new Error("积分不足，请充值后再生成视频");
            insufficientCredits.name = "VideoInsufficientCredits";
            throw insufficientCredits;
        }
        const message = new Error(readVideoSubmissionError(error));
        if (axios.isAxiosError(error) && error.response && [400, 401, 403, 404, 409, 413, 415, 422, 429].includes(error.response.status)) message.name = "VideoSubmissionRejected";
        throw message;
    }
}

/**
 * 提交失败时要给出能据此行动的原因。
 * 手机端最常见的是请求体过大被网关拒绝（413）和慢速上行中断（无响应），
 * 这两类错误此前只剩一句“视频任务创建失败”，用户无法判断该怎么处理。
 */
function readVideoSubmissionError(error: unknown) {
    if (axios.isAxiosError(error)) {
        const status = error.response?.status;
        if (status === 413) return "提交内容过大，服务器拒绝了本次请求（413）。请压缩或减少参考视频/图片后重试。";
        if (status === 409) return readApiErrorMessage(error.response?.data) || "登录账户已变化，请刷新页面后重试。";
        if (status === 415) return readApiErrorMessage(error.response?.data) || "参考素材格式不受支持，请换成常见的 JPG/PNG、MP4 或 MP3 后重试。";
        if (status === 429) return apiText("rateLimited");
        if (status === 502 || status === 503 || status === 504) return `网关返回 ${status}，提交结果未确认。请先刷新生成记录，确认没有已创建的任务后再重试。`;
        if (!error.response) return error.code === "ECONNABORTED" ? "提交超时，网络较慢时请减少参考素材或改用更小的文件后重试。" : "网络异常，提交没有完成，请检查网络后重试。";
    }
    return readAxiosError(error, apiText("videoTaskCreateFailed"));
}

async function pollOpenAIVideoTask(config: AiConfig, task: VideoGenerationTask, options?: RequestOptions): Promise<VideoGenerationTaskState> {

        if (url) return { status: "completed", result: await videoResultFromUrl(url, options) };
        if (video.status === "completed") {
            const content = await axios.get<Blob>(aiApiUrl(config, `/videos/${task.id}/content`), { headers: aiHeaders(config, undefined, options?.expectedUserId), responseType: "blob", signal: options?.signal });

async function referenceMediaToFile(item: { name: string; type?: string; url?: string; storageKey?: string }, fallbackName: string, errorKey: "invalidReferenceVideo" | "invalidReferenceAudio", options?: RequestOptions) {
    let blob = item.storageKey ? await getMediaBlob(item.storageKey) : null;
    if (!blob) {
        const url = item.storageKey ? await resolveMediaUrl(item.storageKey, item.url || "") : item.url || "";
        if (!url) throw new Error(apiText(errorKey));

    if (!blob.size) throw new Error(apiText(errorKey));

function unwrapEnvelope<T>(payload: ApiEnvelope<T>, emptyMessage: string): T {
    if (!payload) throw new Error(emptyMessage);
    if (typeof payload === "object" && "code" in payload && payload.code !== undefined) {
        if (payload.code !== 0 && payload.code !== "0") throw new Error(readApiErrorMessage(payload) || apiText("requestFailed"));
        if (!payload.data) throw new Error(emptyMessage);
        return payload.data;
    }
    return payload as T;
}

function readApiErrorMessage(value: unknown): string {
    if (!value) return "";
    if (typeof value === "string") {
        try {
            const parsed = JSON.parse(value);
            const inner = readApiErrorMessage(parsed) || value;
            if (inner === value && typeof parsed === "object" && Object.keys(parsed).length === 0) return "";
            return inner;
        } catch {
            if (/<[a-z][\s\S]*>/i.test(value)) return apiText("htmlError", { preview: `${value.slice(0, 80)}...` });
            return friendlyErrorMessage(value);
        }
    }
    if (typeof value !== "object") return "";
    const payload = value as { msg?: unknown; message?: unknown; error?: unknown; detail?: unknown };
    // error may be a string or an object containing a message.
    const errorMsg =
        typeof payload.error === "string"
            ? payload.error
            : (payload.error as { message?: unknown })?.message;
    return friendlyErrorMessage(
        readApiErrorMessage(payload.msg) ||
        readApiErrorMessage(payload.message) ||
        readApiErrorMessage(errorMsg) ||
        readApiErrorMessage(payload.detail) ||
        ""
    );
}

function readAxiosError(error: unknown, fallback: string) {
    if (axios.isCancel(error)) return apiText("requestCanceled");
    if (axios.isAxiosError<{ error?: { message?: string }; msg?: string; message?: string; code?: number | string }>(error)) {
        if (!error.response && error.code === "ERR_NETWORK") return apiText("requestFailed");
        const responseData = error.response?.data;
        return readApiErrorMessage(responseData) || statusMessage(error.response?.status, fallback);
    }
    if (error instanceof DOMException && error.name === "AbortError") return apiText("requestCanceled");
    return error instanceof Error ? readApiErrorMessage(error.message) || error.message : fallback;
}

function statusMessage(status: number | undefined, fallback: string) {
    if (status === 401 || status === 403) return apiText("authenticationFailed");
    if (status === 429) return apiText("rateLimited");
    return status ? `${fallback}（${status}）` : fallback;
}

async function assertVideoBlob(blob: Blob) {
    if (!blob.type.includes("json")) return;
    let payload: { code?: number; msg?: string; error?: { message?: string } };
    try {
        payload = JSON.parse(await blob.text()) as { code?: number; msg?: string; error?: { message?: string } };
    } catch {
        return;
    }
    if (typeof payload.code === "number" && payload.code !== 0) throw new Error(readApiErrorMessage(payload) || apiText("videoDownloadFailed"));
    if (payload.error?.message) throw new Error(readApiErrorMessage(payload.error.message) || payload.error.message);
}

function isImageUpload(blob: Blob, prefix: string) {
    return blob.type.toLowerCase().startsWith("image/") || ["image", "reference", "promo", "prompts"].includes(prefix);

async function normalizeImageForUpload(blob: Blob, prefix: string, force = false) {
    if (!isImageUpload(blob, prefix)) return blob;
    if (!force && UPLOAD_IMAGE_TYPES.has(blob.type.toLowerCase())) return blob;

    let source: ImageBitmap | HTMLImageElement | undefined;
    let bitmap: ImageBitmap | undefined;
    let image: HTMLImageElement | undefined;
    let sourceUrl: string | undefined;
    try {
        if (typeof createImageBitmap === "function") {
            try { bitmap = await createImageBitmap(blob); source = bitmap; } catch { bitmap = undefined; }
        }
        if (!source) {
            sourceUrl = URL.createObjectURL(blob);
            image = new Image();
            image.decoding = "async";
            await new Promise<void>((resolve, reject) => {
                image!.onload = () => resolve();
                image!.onerror = () => reject(new Error("image decode failed"));
                image!.src = sourceUrl!;
            });
            source = image;
        }
        const width = bitmap ? bitmap.width : image?.naturalWidth ?? 0;
        const height = bitmap ? bitmap.height : image?.naturalHeight ?? 0;
        if (!width || !height) throw new Error("image has no dimensions");
        const canvas = document.createElement("canvas");
        canvas.width = width;
        canvas.height = height;
        const context = canvas.getContext("2d");
        if (!context) throw new Error("canvas unavailable");
        context.fillStyle = "#ffffff";
        context.fillRect(0, 0, width, height);
        context.drawImage(source, 0, 0, width, height);
        const jpeg = await new Promise<Blob | null>((resolve) => canvas.toBlob(resolve, "image/jpeg", 0.92));
        if (!jpeg) throw new Error("image conversion failed");
        return new Blob([jpeg], { type: "image/jpeg" });
    } finally {
        if (bitmap) bitmap.close();
        if (sourceUrl) URL.revokeObjectURL(sourceUrl);
    }
}

function uploadWithProgress(url: string, form: FormData, owner: string, signal: AbortSignal, onProgress: (progress: number) => void) {
    return new Promise<Response>((resolve, reject) => {
        const xhr = new XMLHttpRequest();
        const cleanup = () => signal.removeEventListener("abort", abort);
        const abort = () => xhr.abort();
        xhr.open("POST", url);
        xhr.withCredentials = true;
        xhr.setRequestHeader("X-Sundas-User", owner);
        xhr.upload.onprogress = (event) => {
            if (event.lengthComputable && event.total > 0) onProgress(Math.min(99, Math.round((event.loaded / event.total) * 100)));
        };
        xhr.onload = () => resolve(new Response(xhr.responseText, { status: xhr.status, headers: { "content-type": xhr.getResponseHeader("content-type") || "" } }));
        xhr.onerror = () => reject(new TypeError("Network request failed"));
        xhr.onabort = () => reject(new DOMException("Upload aborted", "AbortError"));
        xhr.ontimeout = () => reject(new DOMException("Upload timed out", "AbortError"));
        xhr.onloadend = cleanup;
        if (signal.aborted) {
            cleanup();
            reject(new DOMException("Upload aborted", "AbortError"));
            return;
        }
        signal.addEventListener("abort", abort, { once: true });
        onProgress(0);
        xhr.send(form);
    });
}

export async function uploadBlobToServer(blob: Blob, prefix: string, expectedOwner = currentMediaOwner(), signal?: AbortSignal, reportError = false, normalize = true, onProgress?: (progress: number) => void) {
    if (!blob.size) return null;
    const owner = expectedOwner;
    assertMediaOwner(owner, signal);
    if (!owner) return null;
    const notifyUnsupported = () => notification.error({ key: "image-format-unsupported", message: i18n.t("common.unsupportedImageFormatTitle"), description: i18n.t("common.unsupportedImageFormatDescription") });
    let upload: Blob = blob;
    if (normalize) {
        try { upload = await normalizeImageForUpload(blob, prefix); }
        catch { notifyUnsupported(); return null; }
    }
    const controller = new AbortController();
    const abort = () => controller.abort();
    if (signal?.aborted) abort();
    else signal?.addEventListener("abort", abort, { once: true });
    const timer = window.setTimeout(() => controller.abort(), 180_000);
    const send = async (body: Blob) => {
        if (!['promo', 'prompts', 'prompt-video'].includes(prefix)) {
            const { tryDirectMediaUpload } = await import('@/services/api/media-upload');
            const direct = await tryDirectMediaUpload(body, { owner, signal: controller.signal, onProgress });
            assertMediaOwner(owner, controller.signal);
            if (direct) return direct;
        }
        const form = new FormData();
        form.append("file", body, `${prefix}.${EXT[body.type] ?? "bin"}`);
        const url = prefix === 'promo' ? '/api/admin/promo/upload' : ['prompts', 'prompt-video'].includes(prefix) ? '/api/admin/prompts/upload' : '/api/uploads';
        return onProgress ? uploadWithProgress(url, form, owner, controller.signal, onProgress) : fetch(url, { method: "POST", body: form, credentials: "include", headers: { "X-Sundas-User": owner }, signal: controller.signal });
    };
    assertMediaOwner(owner, signal);
    let response: Response;
    try {
        response = await send(upload);
        if (response.status === 400 && !response.headers.get("content-type")?.includes("application/json")) {
            response = await send(upload);
        }
        if (normalize && response.status === 415 && isImageUpload(blob, prefix)) {
            try {
                const converted = await normalizeImageForUpload(blob, prefix, true);
                if (converted !== upload) response = await send(converted);
            } catch { notifyUnsupported(); return null; }
        }
    } catch (error) {
        if (signal?.aborted) throw error;
        if (controller.signal.aborted) throw new DOMException("Upload timed out", "AbortError");
        throw error;
    } finally {
        window.clearTimeout(timer);
        signal?.removeEventListener("abort", abort);
    }
    if (!response.ok) {
        const body = await response.json().catch(() => null) as { message?: unknown; errorCode?: unknown } | null;
        if (body?.errorCode === "STORAGE_QUOTA_EXCEEDED") {
            void useAuthStore.getState().refreshWallet().catch(() => undefined);
            throw new StorageQuotaError(typeof body.message === "string" ? body.message : i18n.t("home.materialUploadFailed"));
        }
        if (reportError) {
            throw new Error(typeof body?.message === "string" ? body.message : i18n.t("home.materialUploadFailed"));
        }
        return null;
    }
    const result = (await response.json()) as { url: string; mime: string; bytes: number };
    assertMediaOwner(owner, signal);
    return result;
}

type PrepareEditor = () => void | Promise<void>;

/**
 * 桌面更新前的编辑器准备钩子（sundas 精简版）。
 *
 * BeefTV 桌面端在更新前要等编辑器落盘；sundas 是纯 Web，没有原生宿主，
 * 但导演台的保存协调器仍然注册这个钩子来表达「我还有未落盘的内容」。
 * 这里保留注册/注销与会话级查询，让调用方代码零改动、语义不丢。
 */
const editors = new Set<{ prepare: PrepareEditor; isSaving: () => boolean; mounted: boolean }>();

export function registerDesktopUpdatePreparation(prepare: PrepareEditor, isSaving: () => boolean = () => false) {
    const editor = { prepare, isSaving, mounted: true };
    editors.add(editor);
    return () => {
        editor.mounted = false;
        if (!isSaving()) editors.delete(editor);
    };
}

export async function prepareDesktopEditorsForUpdate() {
    for (const editor of [...editors]) if (editor.mounted) await editor.prepare();
}

export function hasActiveDesktopEditorSaves() {
    let saving = false;
    for (const editor of editors) {
        if (editor.isSaving()) saving = true;
        else if (!editor.mounted) editors.delete(editor);
    }
    return saving;
}

/**
 * 客户端诊断（sundas 精简版）。
 *
 * BeefTV 有一套带本地事件存储、上传与去重的诊断管道；sundas 还没有对应后端，
 * 这里保留 recordDiagnosticEvent 的签名，开发环境落到 console，生产环境静默丢弃。
 * 目的只有一个：让从 BeefTV 平移过来的调用点保持可编译、可观测、且不引入新后端依赖。
 */
export type DiagnosticEventInput = {
    level: "info" | "warning" | "error";
    category?: string;
    code?: string;
    message: string;
    detail?: Record<string, unknown>;
    timestamp?: string;
};

export function recordDiagnosticEvent(input: DiagnosticEventInput) {
    if (!import.meta.env.DEV) return;
    const line = `[diagnostic:${input.level}] ${input.code || input.category || "runtime"} — ${input.message}`;
    if (input.level === "error") console.warn(line, input.detail ?? "");
    else console.info(line, input.detail ?? "");
}

import { nanoid } from "nanoid";

import { assertLocalMediaOwner, assertMediaOwner, currentMediaOwner, isOwnMediaUrl, localMediaKey, notifyLocalMediaOnly, StorageQuotaError, uploadBlobToServer } from "@/services/cloud-media";
import { lazyStore } from "@/lib/localforage-store";
import { withLocalProxy } from "@/stores/use-config-store";
import { MediaObjectUrls, registerSessionMediaCache } from "@/lib/media-object-urls";

export type UploadedFile = { mediaId?: string; taskId?: string; objectKey?: string; url: string; storageKey: string; bytes: number; mimeType: string; width?: number; height?: number; durationMs?: number };

const store = lazyStore("media_files");
const objectUrls = registerSessionMediaCache(new MediaObjectUrls());

export class InvalidMediaResponseError extends Error {
    constructor() { super("媒体地址返回了错误或无效内容，请检查来源后重试"); this.name = "InvalidMediaResponseError"; }
}

export async function uploadMediaFile(input: string | Blob, prefix = "file", options?: { expectedOwner?: string | null; signal?: AbortSignal }): Promise<UploadedFile> {
    const owner = options?.expectedOwner === undefined ? currentMediaOwner() : options.expectedOwner;
    assertMediaOwner(owner, options?.signal);
    if (typeof input === "string" && isOwnMediaUrl(input)) {
        const meta = await readMediaMetaByUrl(input);
        assertMediaOwner(owner, options?.signal);
        return { url: input, storageKey: "", bytes: 0, mimeType: "", ...meta };
    }

    let blob: Blob;
    if (typeof input === "string") {
        const response = await fetch(withLocalProxy(input), { signal: options?.signal });
        const contentType = response.headers.get("content-type")?.split(";")[0].trim().toLowerCase() ?? "";
        if (!response.ok || contentType.startsWith("text/") || contentType === "application/json" || contentType.endsWith("+json")) throw new InvalidMediaResponseError();
        blob = await response.blob();
        if (!blob.size) throw new InvalidMediaResponseError();
    } else blob = input;
    assertMediaOwner(owner, options?.signal);
    const uploaded = await uploadBlobToServer(blob, prefix, owner, options?.signal).catch((error) => { if (error instanceof StorageQuotaError) throw error; return null; });
    assertMediaOwner(owner, options?.signal);
    const probeUrl = URL.createObjectURL(blob);
    const meta = blob.type.startsWith("video/") ? await readVideoMeta(probeUrl) : blob.type.startsWith("audio/") ? await readAudioMeta(probeUrl) : {};
    URL.revokeObjectURL(probeUrl);
    assertMediaOwner(owner, options?.signal);

    if (uploaded) {
        return { url: uploaded.url, storageKey: "", bytes: blob.size, mimeType: blob.type || "application/octet-stream", ...meta };
    }

    const storageKey = localMediaKey(prefix, owner, nanoid());
    await store.setItem(storageKey, blob);
    try { assertMediaOwner(owner, options?.signal); } catch (error) { await store.removeItem(storageKey); throw error; }
    notifyLocalMediaOnly(owner);
    const url = URL.createObjectURL(blob);
    objectUrls.set(storageKey, url);
    return { url, storageKey, bytes: blob.size, mimeType: blob.type || "application/octet-stream", ...meta };
}

function readMediaMetaByUrl(url: string) {
    if (/\.(mp4|webm|mov|m4v)(\?|$)/i.test(url)) return readVideoMeta(url);
    if (/\.(mp3|wav|ogg|m4a|aac|flac)(\?|$)/i.test(url)) return readAudioMeta(url);
    return Promise.resolve({});
}

export async function resolveMediaUrl(storageKey?: string, fallback = "", hydration?: { owner: string; isActive: () => boolean }) {
    if (!storageKey) return fallback;
    const owner = hydration?.owner ?? currentMediaOwner();
    assertLocalMediaOwner(storageKey, owner);
    if (hydration && !hydration.isActive()) throw new DOMException("账户数据加载已中断", "AbortError");
    const url = await objectUrls.resolve(storageKey, async () => {
        const blob = await store.getItem<Blob>(storageKey);
        if (hydration) { if (!hydration.isActive()) throw new DOMException("账户数据加载已中断", "AbortError"); }
        else assertMediaOwner(owner);
        return blob;
    });
    if (hydration) { if (!hydration.isActive()) throw new DOMException("账户数据加载已中断", "AbortError"); }
    else assertMediaOwner(owner);
    return url || fallback;
}

export async function getMediaBlob(storageKey: string) {
    const owner = currentMediaOwner();
    assertLocalMediaOwner(storageKey, owner);
    const blob = await store.getItem<Blob>(storageKey);
    assertMediaOwner(owner);
    return blob;
}

export async function setMediaBlob(storageKey: string, blob: Blob) {
    await store.setItem(storageKey, blob);
    const url = URL.createObjectURL(blob);
    objectUrls.set(storageKey, url);
    return url;
}

export async function deleteStoredMedia(keys: Iterable<string>) {
    await Promise.all(
        Array.from(new Set(keys)).map(async (key) => {
            objectUrls.delete(key);
            await store.removeItem(key);
        }),
    );
}

export async function cleanupUnusedMedia(usedData: unknown) {
    const usedKeys = collectMediaStorageKeys(usedData);
    const unused: string[] = [];
    await store.iterate((_value, key) => {
        if (!usedKeys.has(key)) unused.push(key);
    });
    await deleteStoredMedia(unused);
}

export function collectMediaStorageKeys(value: unknown, keys = new Set<string>()) {
    if (!value || typeof value !== "object") return keys;
    if ("storageKey" in value && typeof value.storageKey === "string" && value.storageKey.includes(":")) keys.add(value.storageKey);
    Object.values(value).forEach((item) => (Array.isArray(item) ? item.forEach((child) => collectMediaStorageKeys(child, keys)) : collectMediaStorageKeys(item, keys)));
    return keys;
}

function readVideoMeta(url: string) {
    return new Promise<{ width: number; height: number; durationMs?: number }>((resolve) => {
        const video = document.createElement("video");
        const done = () => {
            const meta = { width: video.videoWidth || 1280, height: video.videoHeight || 720, durationMs: Number.isFinite(video.duration) ? Math.round(video.duration * 1000) : undefined };
            video.onloadedmetadata = null;
            video.onerror = null;
            video.removeAttribute("src");
            video.load();
            resolve(meta);
        };
        video.onloadedmetadata = done;
        video.onerror = done;
        video.preload = "metadata";
        video.src = url;
    });
}
